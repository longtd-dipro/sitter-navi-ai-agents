import {
  Injectable,
  CanActivate,
  ExecutionContext,
  UnauthorizedException,
  ForbiddenException,
} from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { Reflector } from '@nestjs/core';

import { ApiKeyService } from '../modules/api-key/api-key.service';

export const AUTH_API_KEY_METADATA = '__AuthApiKey__';

/**
 * Metadata key for required scopes on a controller/method.
 * Usage: @SetMetadata(API_KEY_SCOPES_METADATA, ['user:read', 'order:read'])
 */
export const API_KEY_SCOPES_METADATA = '__ApiKeyScopes__';

@Injectable()
export class ApiKeyGuard implements CanActivate {
  constructor(
    private readonly reflector: Reflector,
    private readonly configService: ConfigService,
    private readonly apiKeyService: ApiKeyService,
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const rawKey = request.headers['x-api-key'];

    if (!rawKey) {
      throw new UnauthorizedException('Missing API key');
    }

    // 1. Try DB-backed validation first
    const result = await this.apiKeyService.validate(rawKey);

    if (result.valid && result.apiKey) {
      // Attach API client info to request for downstream use
      request.apiClient = {
        id: result.apiKey.id,
        name: result.apiKey.name,
        scopes: result.apiKey.scopes,
      };

      // Check scopes if required
      const requiredScopes = this.reflector.getAllAndOverride<string[]>(API_KEY_SCOPES_METADATA, [
        context.getHandler(),
        context.getClass(),
      ]);

      if (requiredScopes && requiredScopes.length > 0) {
        if (!this.apiKeyService.hasAllScopes(result.apiKey, requiredScopes)) {
          throw new ForbiddenException(
            `Insufficient scopes. Required: ${requiredScopes.join(', ')}`,
          );
        }
      }

      return true;
    }

    // 2. Fallback: check legacy env-based key (backward compatible)
    const legacyKey = this.configService.get<string>('apiKey');
    if (legacyKey && rawKey === legacyKey) {
      request.apiClient = { id: null, name: 'legacy', scopes: null };
      return true;
    }

    throw new UnauthorizedException('Invalid API key');
  }
}
