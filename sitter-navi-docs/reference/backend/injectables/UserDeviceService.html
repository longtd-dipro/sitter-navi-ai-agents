import { Injectable, Logger } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { In, Repository } from 'typeorm';
import { UserDevice } from './user-device.entity';
import { RegisterDeviceDto } from './dtos/register-device.dto';

/**
 * Manages user device tokens for push notifications.
 *
 * - Register/unregister devices.
 * - Query active device tokens by user ID.
 * - Batch query for multiple users.
 */
@Injectable()
export class UserDeviceService {
  private readonly logger = new Logger(UserDeviceService.name);

  constructor(
    @InjectRepository(UserDevice)
    private readonly repo: Repository<UserDevice>,
  ) {}

  /**
   * Register or update a device token.
   *
   * - If the token already exists for this user → update lastUsedAt.
   * - If the token exists but belongs to a different user → transfer ownership.
   * - If the token is new → create record.
   */
  async registerDevice(userId: string, dto: RegisterDeviceDto): Promise<UserDevice> {
    const existing = await this.repo.findOneBy({
      deviceToken: dto.deviceToken,
    });

    if (existing) {
      // Update existing record
      existing.userId = userId;
      existing.platform = dto.platform;
      existing.deviceName = dto.deviceName ?? existing.deviceName;
      existing.isActive = true;
      existing.lastUsedAt = new Date();

      return this.repo.save(existing);
    }

    // Create new record
    const device = this.repo.create({
      userId,
      deviceToken: dto.deviceToken,
      platform: dto.platform,
      deviceName: dto.deviceName ?? null,
      isActive: true,
      lastUsedAt: new Date(),
    });

    return this.repo.save(device);
  }

  /**
   * Unregister a device (soft-deactivate).
   */
  async unregisterDevice(deviceToken: string): Promise<void> {
    await this.repo.update({ deviceToken }, { isActive: false, lastUsedAt: new Date() });
    this.logger.debug(`Device unregistered: ${deviceToken}`);
  }

  /**
   * Get all active device tokens for a user.
   */
  async getActiveDevices(userId: string): Promise<string[]> {
    const devices = await this.repo.find({
      where: { userId, isActive: true },
      select: ['deviceToken'],
    });
    return devices.map((d) => d.deviceToken);
  }

  /**
   * Get active device tokens for multiple users at once.
   *
   * @returns Map of userId → deviceToken[]
   */
  async getActiveDevicesBatch(userIds: string[]): Promise<Map<string, string[]>> {
    const devices = await this.repo.find({
      where: { userId: In(userIds), isActive: true },
      select: ['userId', 'deviceToken'],
    });

    const map = new Map<string, string[]>();
    for (const d of devices) {
      const tokens = map.get(d.userId) ?? [];
      tokens.push(d.deviceToken);
      map.set(d.userId, tokens);
    }
    return map;
  }

  /**
   * Clean up devices that haven't been used for N days.
   */
  async cleanupInactive(days: number): Promise<number> {
    const result = await this.repo
      .createQueryBuilder()
      .delete()
      .where('isActive = :isActive', { isActive: false })
      .andWhere(`lastUsedAt < NOW() - INTERVAL :days DAY`, { days })
      .execute();

    const count = result.affected ?? 0;
    if (count > 0) {
      this.logger.log(`Cleaned up ${count} inactive device tokens`);
    }
    return count;
  }
}
