import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { IsNull, Repository } from 'typeorm';

import { Participant } from '@entities';

import {
  ParticipantRepository,
  ParticipantDto,
  CreateParticipantInput,
  RestoreParticipantInput,
  CursorPaginationInput,
  CursorPaginatedResult,
} from '../contracts';
import { DuplicateKeyError, EntityNotFoundError } from '../errors';
import { wrapTypeOrmError } from './typeorm-conversation.repository';

// ── Helpers ────────────────────────────────────────────────────────────

export function toParticipantDto(entity: Participant): ParticipantDto {
  return {
    id: entity.id,
    conversationId: entity.conversationId,
    userId: entity.userId,
    role: entity.role,
    unreadCount: entity.unreadCount,
    joinedAt: entity.joinedAt,
    createdAt: entity.createdAt,
    updatedAt: entity.updatedAt,
    deletedAt: entity.deletedAt,
  };
}

// ── Repository ──────────────────────────────────────────────────────────

@Injectable()
export class TypeOrmParticipantRepository implements ParticipantRepository {
  constructor(
    @InjectRepository(Participant)
    private readonly repo: Repository<Participant>,
  ) {}

  async findByGroupAndUser(groupId: string, userId: string): Promise<ParticipantDto | null> {
    const entity = await this.repo.findOne({
      where: { conversationId: groupId, userId, deletedAt: IsNull() },
    });
    return entity ? toParticipantDto(entity) : null;
  }

  async findActiveByGroup(groupId: string): Promise<ParticipantDto[]> {
    const entities = await this.repo.find({
      where: { conversationId: groupId, deletedAt: IsNull() },
    });
    return entities.map(toParticipantDto);
  }

  async findActiveByUser(userId: string): Promise<ParticipantDto[]> {
    const entities = await this.repo.find({
      where: { userId, deletedAt: IsNull() },
    });
    return entities.map(toParticipantDto);
  }

  async countActiveByGroup(groupId: string): Promise<number> {
    return this.repo.count({
      where: { conversationId: groupId, deletedAt: IsNull() },
    });
  }

  async create(data: CreateParticipantInput): Promise<ParticipantDto> {
    try {
      const entity = this.repo.create({
        conversationId: data.conversationId,
        userId: data.userId,
        role: data.role as any,
        unreadCount: data.unreadCount ?? 0,
      });
      const saved = await this.repo.save(entity as Participant);
      return toParticipantDto(saved);
    } catch (error) {
      throw wrapTypeOrmError(error);
    }
  }

  async createMany(data: CreateParticipantInput[]): Promise<ParticipantDto[]> {
    try {
      const entities = data.map((d) =>
        this.repo.create({
          conversationId: d.conversationId,
          userId: d.userId,
          role: d.role as any,
          unreadCount: d.unreadCount ?? 0,
        }),
      );
      const saved = await this.repo.save(entities as Participant[]);
      return saved.map(toParticipantDto);
    } catch (error) {
      throw wrapTypeOrmError(error);
    }
  }

  async softDelete(groupId: string, userId: string): Promise<void> {
    await this.repo.softDelete({
      conversationId: groupId,
      userId,
      deletedAt: IsNull(),
    });
  }

  async restore(groupId: string, userId: string, data: RestoreParticipantInput): Promise<void> {
    const updateData: Record<string, unknown> = { deletedAt: null };
    if (data.role !== undefined) updateData.role = data.role;
    if (data.unreadCount !== undefined) updateData.unreadCount = data.unreadCount;

    await this.repo.update({ conversationId: groupId, userId }, updateData as any);
  }

  async updateRole(groupId: string, userId: string, role: string): Promise<void> {
    await this.repo.update(
      { conversationId: groupId, userId, deletedAt: IsNull() },
      { role: role as any },
    );
  }

  async incrementUnreadExcept(groupId: string, excludeUserId: string): Promise<void> {
    await this.repo
      .createQueryBuilder()
      .update(Participant)
      .set({ unreadCount: () => 'unread_count + 1' } as any)
      .where('conversation_id = :groupId', { groupId })
      .andWhere('user_id != :excludeUserId', { excludeUserId })
      .andWhere('deleted_at IS NULL')
      .execute();
  }

  async resetUnread(groupId: string, userId: string): Promise<void> {
    await this.repo.update(
      { conversationId: groupId, userId, deletedAt: IsNull() },
      { unreadCount: 0 },
    );
  }

  async findSoftDeleted(groupId: string, userId: string): Promise<ParticipantDto | null> {
    const entity = await this.repo.findOne({
      where: { conversationId: groupId, userId },
      withDeleted: true,
    });
    return entity ? toParticipantDto(entity) : null;
  }

  async findPaginated(
    groupId: string,
    pagination: CursorPaginationInput,
  ): Promise<CursorPaginatedResult<ParticipantDto>> {
    const pageSize = Math.min(Math.max(pagination.pageSize ?? 20, 1), 100);

    const qb = this.repo
      .createQueryBuilder('gp')
      .where('gp.conversation_id = :groupId', { groupId })
      .andWhere('gp.deleted_at IS NULL')
      .orderBy('gp.id', 'DESC')
      .take(pageSize + 1);

    if (pagination.cursor) {
      qb.andWhere('gp.id < :cursor', { cursor: pagination.cursor });
    }

    const entities = await qb.getMany();
    const hasMore = entities.length > pageSize;
    const data = hasMore ? entities.slice(0, pageSize) : entities;

    return {
      data: data.map(toParticipantDto),
      nextCursor: hasMore && data.length > 0 ? data[data.length - 1].id : undefined,
      hasMore,
    };
  }
}
