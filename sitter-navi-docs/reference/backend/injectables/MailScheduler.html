Description

Recovery-only scheduler for the email module.

When BullMQ/Redis is healthy, emails are dispatched immediately. When Redis is down, this scheduler picks up stuck PENDING records.

Uses @DistributedLock to ensure only one instance runs recovery, with automatic Redis → PostgreSQL advisory lock fallback.

BullMQ future: when BullMQ worker handles retries, this scheduler only needs to recover truly stuck messages (PENDING > N minutes).

import { Injectable, Logger } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { DistributedLock } from '@infrastructure/lock';
import { OutboxEmailService } from './outbox-email.service';
import { IEmailDispatcher, EMAIL_DISPATCHER } from './email-dispatcher.interface';
import { Inject } from '@nestjs/common';

/**
 * Recovery-only scheduler for the email module.
 *
 * When BullMQ/Redis is healthy, emails are dispatched immediately.
 * When Redis is down, this scheduler picks up stuck PENDING records.
 *
 * Uses @DistributedLock to ensure only one instance runs recovery,
 * with automatic Redis → PostgreSQL advisory lock fallback.
 *
 * BullMQ future: when BullMQ worker handles retries, this scheduler
 * only needs to recover truly stuck messages (PENDING > N minutes).
 */
@Injectable()
export class MailScheduler {
  private readonly logger = new Logger(MailScheduler.name);

  /** Minutes after which a PENDING record is considered "stuck" */
  private readonly stuckThresholdMinutes = 2;

  /** Days after which sent/permanently-failed records are cleaned up */
  private readonly cleanupDays = 30;

  constructor(
    private readonly outboxService: OutboxEmailService,
    @Inject(EMAIL_DISPATCHER)
    private readonly dispatcher: IEmailDispatcher,
  ) {}

  /**
   * Recovery job: finds emails stuck in PENDING state and dispatches them.
   *
   * Runs every 30 seconds with a distributed lock.
   * Lock uses Redis (primary) → PostgreSQL advisory lock (fallback).
   */
  @Cron(CronExpression.EVERY_30_SECONDS)
  @DistributedLock({ key: 'mail:recovery', ttlMs: 25_000 })
  async recoverStuckEmails(): Promise<void> {
    const stuck = await this.outboxService.getStuckPending(this.stuckThresholdMinutes);

    if (stuck.length === 0) {
      return;
    }

    this.logger.log(`Recovering ${stuck.length} stuck pending emails`);

    // Dispatch in concurrent batches to avoid sequential TokenBucket wait
    const concurrency = 5;
    for (let i = 0; i < stuck.length; i += concurrency) {
      const batch = stuck.slice(i, i + concurrency);
      await Promise.all(batch.map((record) => this.dispatchOne(record)));
    }
  }

  private async dispatchOne(record: any): Promise<void> {
    // If the record has a templateName and empty body, it was pushed as raw data
    // and needs rendering. Otherwise, use the already-rendered subject/body.
    const job =
      record.templateName && !record.body
        ? {
            type: 'template' as const,
            outboxId: record.id,
            templateName: record.templateName,
            to: record.toAddress,
            data: {}, // Original data not stored — dispatcher renders with empty context
          }
        : {
            type: 'custom' as const,
            outboxId: record.id,
            to: record.toAddress,
            subject: record.subject || '',
            html: record.body || '',
          };

    const result = await this.dispatcher.dispatch(job);

    if (result.success) {
      await this.outboxService.markSent(record.id, result.persist as any);
    } else {
      await this.outboxService.markFailed(
        record.id,
        result.error ?? 'Recovery dispatch failed',
        result.permanent ?? false,
        result.persist as any,
      );
    }
  }

  /**
   * Cleanup job: removes old sent and permanently-failed records.
   * Runs daily at 3:00 AM.
   */
  @Cron('0 3 * * *')
  @DistributedLock({ key: 'mail:cleanup', ttlMs: 60_000 })
  async cleanup(): Promise<void> {
    await this.outboxService.cleanup(this.cleanupDays);
  }
}
