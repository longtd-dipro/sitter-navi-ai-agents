Description

Allocates a unique Snowflake instance_id per pod via a DB-backed lease, replacing the random fallback used during local development.

Why: random ids are 12-bit, so on ECS/EKS with many pods restarting there's a non-trivial probability of two pods picking the same id and silently emitting duplicate Snowflake IDs. A DB lease guarantees exclusivity for the lifetime of the pod.

Implementation notes:

  • We acquire a Postgres session-level advisory lock around the slot selection so concurrent boots don't race on the same row.
  • The lease is renewed by a heartbeat that extends expires_at. If the heartbeat ever finds its row hijacked (different holder), it re-acquires a fresh slot — that path is rare but covers split-brain after long pauses (network partitions, host suspends, etc.).
  • On graceful shutdown we set expires_at = now() so the slot is immediately reclaimable. Hard kills fall back to TTL expiry.

import { Injectable, Logger, OnApplicationBootstrap, OnApplicationShutdown } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { InjectDataSource } from '@nestjs/typeorm';
import { hostname } from 'os';
import { DataSource } from 'typeorm';
import { SnowflakeInstanceEntity } from '../../../entities/snowflake-instance.entity';
import { initSnowflake } from '@utils/snowflake';

/**
 * Maximum value of the Snowflake `instance_id` field (12 bits).
 * `nodejs-snowflake` reserves 12 bits for the machine id, so valid ids
 * are 0..4095 inclusive.
 */
const MAX_INSTANCE_ID = 4095;

/** Postgres advisory lock key. Arbitrary constant scoped to this app. */
const ADVISORY_LOCK_KEY = 7320581123;

/** Defaults intentionally live close to the code that uses them. */
const DEFAULT_LEASE_MS = 60_000;
const DEFAULT_HEARTBEAT_MS = 20_000;
const ACQUIRE_RETRY_COUNT = 5;
const ACQUIRE_RETRY_DELAY_MS = 250;

/**
 * Allocates a unique Snowflake `instance_id` per pod via a DB-backed
 * lease, replacing the random fallback used during local development.
 *
 * Why: random ids are 12-bit, so on ECS/EKS with many pods restarting
 * there's a non-trivial probability of two pods picking the same id and
 * silently emitting duplicate Snowflake IDs. A DB lease guarantees
 * exclusivity for the lifetime of the pod.
 *
 * Implementation notes:
 *  - We acquire a Postgres session-level advisory lock around the slot
 *    selection so concurrent boots don't race on the same row.
 *  - The lease is renewed by a heartbeat that extends `expires_at`. If
 *    the heartbeat ever finds its row hijacked (different `holder`), it
 *    re-acquires a fresh slot — that path is rare but covers split-brain
 *    after long pauses (network partitions, host suspends, etc.).
 *  - On graceful shutdown we set `expires_at = now()` so the slot is
 *    immediately reclaimable. Hard kills fall back to TTL expiry.
 */
@Injectable()
export class SnowflakeAllocatorService implements OnApplicationBootstrap, OnApplicationShutdown {
  private readonly logger = new Logger(SnowflakeAllocatorService.name);
  private readonly leaseMs: number;
  private readonly heartbeatMs: number;
  private readonly holder: string;

  private instanceId: number | null = null;
  private heartbeatTimer: NodeJS.Timeout | null = null;

  constructor(
    @InjectDataSource() private readonly dataSource: DataSource,
    private readonly configService: ConfigService,
  ) {
    this.leaseMs = this.configService.get<number>('snowflake.leaseMs') ?? DEFAULT_LEASE_MS;
    this.heartbeatMs =
      this.configService.get<number>('snowflake.heartbeatMs') ?? DEFAULT_HEARTBEAT_MS;
    this.holder = `${hostname()}#${process.pid}`;
  }

  async onApplicationBootstrap(): Promise<void> {
    const id = await this.acquireWithRetry();
    this.instanceId = id;
    initSnowflake(id);
    this.logger.log(
      `Acquired snowflake instance_id=${id} as holder=${this.holder} (lease=${this.leaseMs}ms)`,
    );
    this.startHeartbeat();
  }

  async onApplicationShutdown(): Promise<void> {
    this.stopHeartbeat();
    if (this.instanceId === null) return;
    try {
      // Expire our own row immediately so other pods can reclaim the
      // slot on the next boot without waiting for the lease TTL.
      await this.dataSource
        .createQueryBuilder()
        .update(SnowflakeInstanceEntity)
        .set({ expiresAt: () => 'NOW()' })
        .where('id = :id AND holder = :holder', {
          id: this.instanceId,
          holder: this.holder,
        })
        .execute();
      this.logger.log(`Released snowflake instance_id=${this.instanceId}`);
    } catch (err) {
      this.logger.warn(
        `Failed to release snowflake instance_id=${this.instanceId}: ${(err as Error).message}`,
      );
    }
  }

  /** Exposed for diagnostics / health checks. */
  getInstanceId(): number | null {
    return this.instanceId;
  }

  private async acquireWithRetry(): Promise<number> {
    let lastErr: unknown;
    for (let attempt = 1; attempt <= ACQUIRE_RETRY_COUNT; attempt++) {
      try {
        return await this.acquireSlot();
      } catch (err) {
        lastErr = err;
        this.logger.warn(
          `acquireSlot attempt ${attempt}/${ACQUIRE_RETRY_COUNT} failed: ${(err as Error).message}`,
        );
        await delay(ACQUIRE_RETRY_DELAY_MS * attempt);
      }
    }
    throw new Error(
      `Unable to acquire snowflake instance_id after ${ACQUIRE_RETRY_COUNT} attempts: ${(lastErr as Error)?.message}`,
    );
  }

  /**
   * Atomically pick the lowest free instance_id and write a lease row
   * for it. The advisory lock serializes concurrent pods so two of them
   * can't both observe the same slot as free.
   */
  private async acquireSlot(): Promise<number> {
    return this.dataSource.transaction(async (manager) => {
      await manager.query('SELECT pg_advisory_xact_lock($1)', [ADVISORY_LOCK_KEY]);

      // Free slot = no row OR row whose lease has expired. We scan ids
      // 0..MAX_INSTANCE_ID and pick the smallest gap. The `LEFT JOIN`
      // pattern keeps the query a single round-trip even though the
      // slot space is sparse.
      const rows = await manager.query<{ id: number }[]>(
        `WITH ids AS (SELECT generate_series(0, $1)::smallint AS id)
         SELECT ids.id
         FROM ids
         LEFT JOIN snowflake_instance s ON s.id = ids.id
         WHERE s.id IS NULL OR s.expires_at <= NOW()
         ORDER BY ids.id
         LIMIT 1`,
        [MAX_INSTANCE_ID],
      );

      if (!rows.length) {
        throw new Error('No free snowflake instance_id available (all 4096 slots leased).');
      }
      const candidateId = rows[0].id;

      const expiresAt = new Date(Date.now() + this.leaseMs);
      // Upsert: if a stale row exists for the same id we overwrite it,
      // otherwise we insert fresh. The advisory lock guarantees no
      // concurrent writer for this id.
      await manager.query(
        `INSERT INTO snowflake_instance (id, holder, acquired_at, expires_at)
         VALUES ($1, $2, NOW(), $3)
         ON CONFLICT (id) DO UPDATE
           SET holder = EXCLUDED.holder,
               acquired_at = EXCLUDED.acquired_at,
               expires_at = EXCLUDED.expires_at,
               updated_at = NOW()`,
        [candidateId, this.holder, expiresAt],
      );

      return candidateId;
    });
  }

  private startHeartbeat(): void {
    this.heartbeatTimer = setInterval(() => {
      void this.heartbeat();
    }, this.heartbeatMs);
    // Don't keep the event loop alive solely for the heartbeat.
    this.heartbeatTimer.unref?.();
  }

  private stopHeartbeat(): void {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer);
      this.heartbeatTimer = null;
    }
  }

  private async heartbeat(): Promise<void> {
    if (this.instanceId === null) return;
    const expiresAt = new Date(Date.now() + this.leaseMs);
    try {
      const result = await this.dataSource
        .createQueryBuilder()
        .update(SnowflakeInstanceEntity)
        .set({ expiresAt })
        .where('id = :id AND holder = :holder', {
          id: this.instanceId,
          holder: this.holder,
        })
        .execute();

      // affected = 0 means our row was hijacked (lease expired and a
      // peer re-claimed it). Re-acquire a fresh slot so we stop emitting
      // ids under the old machine bits.
      if ((result.affected ?? 0) === 0) {
        this.logger.warn(`Heartbeat lost lease on instance_id=${this.instanceId}, re-acquiring.`);
        const newId = await this.acquireWithRetry();
        this.instanceId = newId;
        initSnowflake(newId);
      }
    } catch (err) {
      // We deliberately swallow heartbeat errors: the lease has a TTL,
      // and the next tick will retry. Surfacing the error as fatal would
      // be worse than the brief drift window we already accept.
      this.logger.warn(
        `Snowflake heartbeat failed for instance_id=${this.instanceId}: ${(err as Error).message}`,
      );
    }
  }
}

function delay(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}
