Description

RefreshTokenService — owns the lifecycle of opaque (non-JWT) refresh tokens: issuance, rotation and revocation.

Design:

  • The token is an opaque high-entropy random string. The server keeps no decodable claims in it — validity is determined purely by the DB row keyed on the token's SHA-256 hash.
  • Rotation: every successful refresh consumes (revokes) the presented token and issues a fresh one. A refresh token is therefore single-use — replaying a rotated token fails, which also surfaces token theft (the legitimate client and the attacker can't both hold a valid token).

This service deliberately does NOT mint access tokens — that stays with import('./auth.service').AuthService. Keeping the dependency one-directional (AuthService → RefreshTokenService for issuance) avoids a circular module dependency; the refresh endpoint composes the two.

import { Injectable, UnauthorizedException } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { createHash, randomBytes } from 'crypto';
import ms from 'ms';

import { RefreshTokenEntity } from '../refresh-token.entity';
import { RefreshTokenRepository } from '../refresh-token.repository';

/** Bytes of entropy in a raw refresh token (384 bits → 64 base64url chars). */
const TOKEN_ENTROPY_BYTES = 48;

/** Fallback TTL if `jwt.rfExpireTime` is unset/unparseable (30 days). */
const DEFAULT_TTL_MS = 30 * 24 * 60 * 60 * 1000;

/** Lightweight client fingerprint stored alongside a token (audit only). */
export interface RefreshTokenMeta {
  userAgent?: string;
  ip?: string;
}

/** Result of issuing/rotating a refresh token — the raw value is shown once. */
export interface IssuedRefreshToken {
  /** Owning user id. */
  userId: string;
  /** Raw opaque token — returned to the client, never persisted. */
  refreshToken: string;
  /** Absolute expiry of the raw token. */
  refreshTokenExpiresAt: Date;
}

/**
 * RefreshTokenService — owns the lifecycle of opaque (non-JWT) refresh
 * tokens: issuance, rotation and revocation.
 *
 * Design:
 *  - The token is an opaque high-entropy random string. The server keeps
 *    no decodable claims in it — validity is determined purely by the DB
 *    row keyed on the token's SHA-256 hash.
 *  - **Rotation:** every successful refresh consumes (revokes) the
 *    presented token and issues a fresh one. A refresh token is therefore
 *    single-use — replaying a rotated token fails, which also surfaces
 *    token theft (the legitimate client and the attacker can't both hold
 *    a valid token).
 *
 * This service deliberately does NOT mint access tokens — that stays with
 * {@link import('./auth.service').AuthService}. Keeping the dependency
 * one-directional (AuthService → RefreshTokenService for issuance) avoids
 * a circular module dependency; the refresh endpoint composes the two.
 */
@Injectable()
export class RefreshTokenService {
  constructor(
    private readonly refreshTokenRepository: RefreshTokenRepository,
    private readonly config: ConfigService,
  ) {}

  /**
   * Issue a brand-new refresh token for a user and persist its hash.
   * Returns the raw token (shown once) plus its expiry.
   */
  async issueForUser(userId: string, meta: RefreshTokenMeta = {}): Promise<IssuedRefreshToken> {
    const refreshToken = this.generateRawToken();
    const refreshTokenExpiresAt = new Date(Date.now() + this.getTtlMs());

    await this.refreshTokenRepository.save({
      userId,
      tokenHash: this.hashToken(refreshToken),
      expiresAt: refreshTokenExpiresAt,
      revokedAt: null,
      meta,
    });

    return { userId, refreshToken, refreshTokenExpiresAt };
  }

  /**
   * Rotate a refresh token: validate the presented one, revoke it, and
   * issue a replacement bound to the same user. Throws
   * {@link UnauthorizedException} when the token is unknown, already
   * rotated/revoked, or expired.
   *
   * Policy: when `jwt.rotateRefreshToken` is `false`, rotation is disabled
   * — the presented token is validated and returned as-is (only the access
   * token is refreshed). This trades away single-use theft detection for a
   * longer-lived, reusable refresh token.
   */
  async rotate(rawToken: string, meta: RefreshTokenMeta = {}): Promise<IssuedRefreshToken> {
    const current = await this.resolveActive(rawToken);

    if (!this.config.get<boolean>('jwt.rotateRefreshToken')) {
      return {
        userId: current.userId,
        refreshToken: rawToken,
        refreshTokenExpiresAt: current.expiresAt,
      };
    }

    await this.refreshTokenRepository.revokeById(current.id, new Date());

    return this.issueForUser(current.userId, meta);
  }

  /**
   * Revoke a single refresh token (logout on the current device). Unknown
   * or already-revoked tokens are treated as a no-op — logout is
   * idempotent and must not leak whether a token existed.
   */
  async revoke(rawToken: string): Promise<void> {
    const token = await this.refreshTokenRepository.findActiveByTokenHash(this.hashToken(rawToken));
    if (token) {
      await this.refreshTokenRepository.revokeById(token.id, new Date());
    }
  }

  /** Revoke every active refresh token for a user (logout everywhere). */
  async revokeAllForUser(userId: string): Promise<void> {
    await this.refreshTokenRepository.revokeAllByUserId(userId, new Date());
  }

  /**
   * Resolve a raw token to its active, unexpired row. An expired (but not
   * yet revoked) row is revoked on the spot before rejecting, so it can't
   * be retried.
   */
  private async resolveActive(rawToken: string): Promise<RefreshTokenEntity> {
    const token = await this.refreshTokenRepository.findActiveByTokenHash(this.hashToken(rawToken));

    if (!token) {
      throw new UnauthorizedException('auth.invalid_refresh_token');
    }

    if (token.expiresAt.getTime() <= Date.now()) {
      await this.refreshTokenRepository.revokeById(token.id, new Date());
      throw new UnauthorizedException('auth.refresh_token_expired');
    }

    return token;
  }

  /** Opaque, URL-safe random token. The `rt_` prefix aids log/debug triage. */
  private generateRawToken(): string {
    return `rt_${randomBytes(TOKEN_ENTROPY_BYTES).toString('base64url')}`;
  }

  /** Deterministic SHA-256 hash used as the DB lookup key. */
  private hashToken(rawToken: string): string {
    return createHash('sha256').update(rawToken).digest('hex');
  }

  /** Refresh-token TTL in ms, parsed from `jwt.rfExpireTime` (e.g. "360000s"). */
  private getTtlMs(): number {
    const configured = this.config.get<string>('jwt.rfExpireTime');
    if (!configured) {
      return DEFAULT_TTL_MS;
    }
    const parsed = ms(configured as ms.StringValue);
    return typeof parsed === 'number' && parsed > 0 ? parsed : DEFAULT_TTL_MS;
  }
}
