@Injectable()
export class ConversationService {
  private readonly logger = new Logger(ConversationService.name);

  constructor(
    @InjectRepository(Conversation)
    private readonly groupConversationRepo: Repository<Conversation>,
    @InjectRepository(Participant)
    private readonly participantRepo: Repository<Participant>,
    @InjectRepository(Message)
    private readonly messageRepo: Repository<Message>,
    @InjectRepository(MessageAttachment)
    private readonly groupMessageAttachmentRepo: Repository<MessageAttachment>,
    @InjectRepository(MessageStatus)
    private readonly messageStatusRepo: Repository<MessageStatus>,

    private readonly userRepo: Repository<UserEntity>,

    private readonly dataSource: DataSource,

    private readonly storageService: StorageService,
    private readonly redisCacheService: RedisCacheService,
  ) {}

  // ─────────────────────────────────────────────────────────────────────────
  // Cache invalidation helpers
  // ─────────────────────────────────────────────────────────────────────────

  /**
   * Invalidate the participants cache for a group.
   * Called when members are added, removed, or leave.
   */
  private async invalidateParticipantsCache(groupId: string): Promise<void> {
    try {
      await this.redisCacheService.clearCacheByKey(getParticipantsCacheKey(groupId));
    } catch {
      // Redis unavailable, cache will expire naturally
    }
  }

  /**
   * Invalidate the group info cache.
   * Called when group metadata is updated.
   */
  private async invalidateGroupInfoCache(groupId: string): Promise<void> {
    try {
      await this.redisCacheService.clearCacheByKey(getGroupInfoCacheKey(groupId));
    } catch {
      // Redis unavailable, cache will expire naturally
    }
  }

  /**
   * Invalidate the user's group list cache.
   * Called when a user is added to or removed from a group.
   */
  private async invalidateUserGroupsCache(userId: string): Promise<void> {
    try {
      await this.redisCacheService.clearCacheByKey(getUserGroupsCacheKey(userId));
    } catch {
      // Redis unavailable, cache will expire naturally
    }
  }

  /**
   * Set the unread count for a user in a group via Redis.
   * Used for explicit reset (conversation-opened).
   */
  async setUnreadCount(groupId: string, userId: string, count: number): Promise<void> {
    try {
      const key = getUnreadCountCacheKey(groupId, userId);
      await this.redisCacheService.setValueWithTTL(key, count, UNREAD_COUNT_CACHE_TTL);
    } catch {
      // Redis unavailable, DB is the source of truth
    }
  }

  /**
   * Increment unread count for a user in a group via Redis INCR.
   * Used when a new message is sent.
   * Sets TTL to prevent unbounded key growth.
   */
  private async incrementUnreadCount(groupId: string, userId: string): Promise<void> {
    try {
      const key = getUnreadCountCacheKey(groupId, userId);
      await this.redisCacheService.incrBy(key, 1);
      // Ensure TTL is set (INCR preserves existing TTL, but set it for new keys)
      await this.redisCacheService.expire(key, UNREAD_COUNT_CACHE_TTL);
    } catch {
      // Redis unavailable, DB is the source of truth
    }
  }

  /**
   * Batch increment unread counts for multiple users via Redis pipeline.
   * Uses a single Redis round-trip for all INCR + EXPIRE commands.
   */
  private async batchIncrementUnreadCounts(groupId: string, userIds: string[]): Promise<void> {
    if (userIds.length === 0) return;
    try {
      const pipe = this.redisCacheService.pipeline();
      for (const userId of userIds) {
        const key = getUnreadCountCacheKey(groupId, userId);
        pipe.incrby(key, 1);
        pipe.expire(key, UNREAD_COUNT_CACHE_TTL);
      }
      await pipe.exec();
    } catch {
      // Redis unavailable, DB is the source of truth
    }
  }

  /**
   * Reset unread count for a user in a group (conversation opened).
   * Updates both Redis and DB.
   *
   * Requirements: 9.6
   */
  async resetUnreadCount(userId: string, groupId: string): Promise<void> {
    // Reset in database with WHERE condition — avoids race condition (no findOne + update gap)
    const result = await this.participantRepo
      .createQueryBuilder()
      .update(Participant)
      .set({ unreadCount: 0 })
      .where('conversation_id = :groupId', { groupId })
      .andWhere('user_id = :userId', { userId })
      .andWhere('deleted_at IS NULL')
      .execute();

    if (!result.affected || result.affected === 0) {
      return; // Participant not found or already deleted — silently ignore
    }

    // Reset in Redis (with TTL)
    await this.setUnreadCount(groupId, userId, 0);

    this.logger.debug(`Unread count reset for user ${userId} in group ${groupId}`);
  }

  /**
   * Get total unread message count across all conversations for a user.
   *
   * Sums the `unread_count` column from group_participant for the user.
   * This is a fast SUM query on indexed column (user_id + deleted_at).
   *
   * Requirements: total unread badge across all conversations
   */
  async getTotalUnreadCount(userId: string): Promise<{ totalUnreadCount: number }> {
    const result = await this.participantRepo
      .createQueryBuilder('p')
      .select('COALESCE(SUM(p.unreadCount), 0)', 'total')
      .where('p.userId = :userId', { userId })
      .andWhere('p.deletedAt IS NULL')
      .getRawOne();

    return { totalUnreadCount: Number(result.total) || 0 };
  }

  /**
   * Batch get total unread counts for multiple users.
   *
   * Uses a single GROUP BY query instead of N individual SUM queries.
   * Returns Map<userId, totalUnreadCount>.
   *
   * Requirements: real-time total unread badge for multiple recipients
   */
  async getBatchTotalUnreadCounts(userIds: string[]): Promise<Map<string, number>> {
    if (userIds.length === 0) return new Map();

    const results = await this.participantRepo
      .createQueryBuilder('p')
      .select('p.userId', 'userId')
      .addSelect('COALESCE(SUM(p.unreadCount), 0)', 'total')
      .where('p.userId IN (:...userIds)', { userIds })
      .andWhere('p.deletedAt IS NULL')
      .groupBy('p.userId')
      .getRawMany();

    const map = new Map<string, number>();
    for (const row of results) {
      map.set(row.userId, Number(row.total) || 0);
    }
    // Ensure all requested userIds have an entry (default 0)
    for (const uid of userIds) {
      if (!map.has(uid)) map.set(uid, 0);
    }
    return map;
  }

  /**
   * Cache group info in Redis.
   */
  private async cacheGroupInfo(group: Conversation): Promise<void> {
    try {
      const key = getGroupInfoCacheKey(group.id);
      const info = {
        id: group.id,
        name: group.name,
        avatarFileKey: group.avatarFileKey,
        ownerId: group.ownerId,
        lastMessageId: group.lastMessageId,
        lastMessageAt: group.lastMessageAt,
        createdAt: group.createdAt,
      };
      await this.redisCacheService.setValueWithTTL(key, JSON.stringify(info), GROUP_INFO_CACHE_TTL);
    } catch {
      // Redis unavailable, continue without caching
    }
  }

  /**
   * Get cached group info from Redis.
   * Returns null on cache miss or Redis unavailability.
   */
  private async getCachedGroupInfo(groupId: string): Promise<Record<string, any> | null> {
    try {
      const key = getGroupInfoCacheKey(groupId);
      const cached = await this.redisCacheService.getValue(key);
      if (cached) {
        return JSON.parse(cached);
      }
    } catch {
      // Redis unavailable or parse error
    }
    return null;
  }

  /**
   * Create a new group conversation.
   *
   * - Deduplicates participantIds
   * - Validates total count (5-10 including creator)
   * - Validates all participant IDs exist as active users
   * - Creates Conversation + Participant records in a transaction
   * - Emits GroupCreatedEvent for Socket.IO broadcast
   *
   * Requirements: 1.1, 1.2, 1.3, 1.4, 1.6, 1.7, 1.8
   */
  @DistributedLock({
    key: (args: any[]) => `conversation:create:${args[0]}`,
    ttlMs: 10000,
    onFail: 'throw',
  })
  async createGroup(userId: string, dto: CreateGroupDto): Promise<ConversationResponseDto> {
    // 1. Trim and validate group name
    const trimmedName = dto.name.trim();
    if (!trimmedName || trimmedName.length > 100) {
      throw new BadRequestException(
        'Group name must be between 1 and 100 characters after trimming',
      );
    }

    // 2. Deduplicate participantIds and exclude creator if included
    const uniqueParticipantIds = [...new Set(dto.participantIds.filter((id) => id !== userId))];

    // 3. Validate total count (including creator): must be 2-50
    const totalCount = uniqueParticipantIds.length + 1; // +1 for creator
    if (totalCount < 2) {
      throw new BadRequestException(
        `Minimum participant count is 2 (including creator). Got ${totalCount}`,
      );
    }
    if (totalCount > 50) {
      throw new BadRequestException(
        `Maximum participant count is 50 (including creator). Got ${totalCount}`,
      );
    }

    // 4. Validate all participant IDs exist as active users
    const allUserIds = [userId, ...uniqueParticipantIds];
    const existingUsers = await this.userRepo.find({
      where: { id: In(allUserIds), deletedAt: IsNull() },
      select: ['id'],
    });

    const existingUserIds = new Set(existingUsers.map((u) => u.id));
    const invalidIds = allUserIds.filter((id) => !existingUserIds.has(id));

    if (invalidIds.length > 0) {
      throw new BadRequestException(
        `The following participant IDs do not correspond to existing active users: ${invalidIds.join(', ')}`,
      );
    }

    // 5. Determine conversation type based on total participant count
    const conversationType = totalCount === 2 ? ConversationType.DIRECT : ConversationType.GROUP;

    // 6. Create Conversation and Participant records in a transaction
    const result = await this.dataSource.transaction(async (manager) => {
      // Create the group conversation
      const groupConversation = manager.create(Conversation, {
        name: trimmedName,
        type: conversationType,
        ownerId: userId,
        avatarFileKey: null,
        lastMessageId: null,
        lastMessageAt: null,
      });
      const savedGroup = await manager.save(Conversation, groupConversation);

      // Create participant records: owner + members
      const participantRecords: Partial<Participant>[] = [];

      // Owner record
      participantRecords.push({
        conversationId: savedGroup.id,
        userId: userId,
        role: ParticipantRole.OWNER,
        unreadCount: 0,
      });

      // Member records
      for (const participantId of uniqueParticipantIds) {
        participantRecords.push({
          conversationId: savedGroup.id,
          userId: participantId,
          role: ParticipantRole.MEMBER,
          unreadCount: 0,
        });
      }

      const participants = manager.create(Participant, participantRecords);
      await manager.save(Participant, participants);

      return savedGroup;
    });

    // 6. Emit domain event for Socket.IO broadcast
    const participants = [
      { userId, role: ParticipantRole.OWNER },
      ...uniqueParticipantIds.map((id) => ({
        userId: id,
        role: ParticipantRole.MEMBER,
      })),
    ];

    this.eventEmitter.emit(
      GroupCreatedEvent.eventName,
      new GroupCreatedEvent(result.id, trimmedName, participants, userId),
    );

    // Invalidate user groups cache for all participants
    for (const p of participants) {
      await this.invalidateUserGroupsCache(p.userId);
    }

    this.logger.log(`Group "${trimmedName}" created with ID ${result.id} by user ${userId}`);

    // 7. Return response DTO
    const response = new ConversationResponseDto();
    response.id = result.id;
    response.name = result.name;
    response.type = conversationType;
    response.avatarUrl = null;
    response.lastMessagePreview = null;
    response.unreadCount = 0;
    response.lastMessageAt = null;

    return response;
  }

  /**
   * Find an existing message by clientMessageId for deduplication.
   *
   * Queries Message WHERE conversation_id = groupId
   * AND sender_id = senderId AND client_message_id = clientMessageId.
   * Returns the message response DTO if found, null otherwise.
   *
   * Requirements: 3.13, 3.14
   */
  async findExistingByClientMessageId(

    const existingMessage = await this.messageRepo.findOne({

    if (!existingMessage) {
      return null;
    }

    // Get sender display name
    const sender = await this.userRepo.findOne({

    response.content = existingMessage.content;

  }

  /**
   * Send a message to a group conversation.
   *
   * - Validates participant membership
   * - Validates clientMessageId (required, UUID format)
   * - Performs deduplication check (returns existing message if duplicate)
   * - Validates content (1-5000 chars for text, non-empty content or fileKeys)
   * - Validates fileKeys exist in uploaded_file records
   * - Persists Message and MessageAttachment records
   * - Commits file keys via StorageService
   * - Updates conversation last_message_at/last_message_id
   * - Increments unread counts for other participants via Redis
   * - Emits NewMessageEvent for Socket.IO broadcast
   *
   * Requirements: 3.1, 3.2, 3.3, 3.5, 3.6, 3.7, 3.8, 3.9, 3.10, 3.12, 3.13, 3.14, 4.2, 4.3, 4.4, 9.5
   */
  async sendMessage(
    userId: string,
    groupId: string,
    dto: SendMessageDto,
  ): Promise<MessageResponseDto & { isNew: boolean }> {
    // 1. Get all active participants + sender name in 1 query (was 3 separate SELECTs)
    const participantsWithNames = await this.participantRepo
      .createQueryBuilder('gp')
      .leftJoin(UserEntity, 'u', 'u.id = gp.userId')
      .select(['gp.userId', 'gp.role', 'u.fullNameKanji'])
      .where('gp.conversationId = :groupId', { groupId })
      .andWhere('gp.deletedAt IS NULL')
      .getRawMany();

    // Validate sender is a participant
    const senderRow = participantsWithNames.find((r) => r.gp_user_id === userId);
    if (!senderRow) {
      throw new ForbiddenException('You are not a participant of this group conversation');
    }

    const senderName = senderRow.u_full_name_kanji || '';
    const otherUserIds = participantsWithNames
      .filter((r) => r.gp_user_id !== userId)
      .map((r) => r.gp_user_id);
    const allParticipantIds = participantsWithNames.map((r) => r.gp_user_id);

    // 2. Validate clientMessageId is a valid UUID
    const uuidRegex = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;
    if (!dto.clientMessageId || !uuidRegex.test(dto.clientMessageId)) {
      throw new BadRequestException('clientMessageId is required and must be a valid UUID');
    }

    // 3. Dedup: now handled by INSERT ON CONFLICT (step 7) — no separate SELECT needed.

    // 4. Validate content based on message type
    if (dto.messageType === MessageType.TEXT) {
      if (!dto.content || dto.content.length < 1 || dto.content.length > 2000) {
        throw new BadRequestException('Text message content must be between 1 and 2000 characters');
      }
    }

    // Validate that message has content or file attachments
    const hasContent = dto.content && dto.content.trim().length > 0;
    const hasFiles = dto.fileKeys && dto.fileKeys.length > 0;

    if (!hasContent && !hasFiles) {
      throw new BadRequestException('Message must have content or file attachments');
    }

    // 5. Validate fileKeys exist in uploaded_file records
    if (hasFiles) {
      if (dto.fileKeys!.length > 10) {
        throw new BadRequestException('Maximum 10 file attachments per message');
      }

      const uploadedFiles = await this.dataSource.getRepository(UploadedFile).find({
        where: { fileKey: In(dto.fileKeys!), deletedAt: IsNull() },
      });

      const foundFileKeys = new Set(uploadedFiles.map((f) => f.fileKey));
      const invalidFileKeys = dto.fileKeys!.filter((key) => !foundFileKeys.has(key));

      if (invalidFileKeys.length > 0) {
        throw new BadRequestException(
          `The following file keys are invalid: ${invalidFileKeys.join(', ')}`,
        );
      }
    }

    // 6. Sender name already fetched in step 1 (merged query)

    // 7. INSERT message with ON CONFLICT dedup — eliminates the separate SELECT in step 3.
    //    If a row is returned → new message. If nothing → dedup hit (return existing).
    const messageId = generateSnowflakeId();
    const now = new Date();

    const insertRaw: Array<{
      id: string;
      created_at: string;
    }> = await this.dataSource.query(
      `INSERT INTO conversation.message
         (id, conversation_id, sender_id, client_message_id, content, message_type, created_at, updated_at)
       VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
       ON CONFLICT (conversation_id, sender_id, client_message_id) DO NOTHING
       RETURNING id, created_at`,
      [
        messageId,
        groupId,
        userId,
        dto.clientMessageId,
        dto.content || null,
        dto.messageType,
        now,
        now,
      ],
    );

    // Dedup hit — message already exists
    if (insertRaw.length === 0) {
      const existing = await this.findExistingByClientMessageId(
        groupId,
        userId,
        dto.clientMessageId,
      );
      if (existing) return existing;
      // Edge case: ON CONFLICT didn't insert but also can't find? Retry as new.
      // This shouldn't happen — fall through to normal flow by re-generating ID.
    }

    // 7b. Persist attachments + update conversation in a transaction
    let attachments: MessageAttachment[] = [];

    if (hasFiles) {
      await this.dataSource.transaction(async (manager) => {
        const uploadedFiles = await this.dataSource.getRepository(UploadedFile).find({
          where: { fileKey: In(dto.fileKeys!), deletedAt: IsNull() },
        });
        const fileMap = new Map(uploadedFiles.map((f) => [f.fileKey, f]));

        const attachmentRecords = dto.fileKeys!.map((fileKey) => {
          const file = fileMap.get(fileKey);
          return manager.create(MessageAttachment, {
            messageId,
            fileKey,
            fileName: file?.fileKey?.split('/').pop() || fileKey,
            fileSize: '0',
            contentType: 'application/octet-stream',
          });
        });

        attachments = await manager.save(MessageAttachment, attachmentRecords);
      });
    }

    // 7c. Update conversation last_message — Redis immediately (fast path), DB async (fire-and-forget).
    //     The conversation list reads from Redis cache first; DB is the fallback.
    //     If Redis is down, DB update still completes asynchronously.
    const lastMessageUpdate = {
      lastMessageId: messageId,
      lastMessageAt: now,
    };

    // Redis cache update (immediate, fire-and-forget)
    this.redisCacheService
      .setValueWithTTL(
        getGroupInfoCacheKey(groupId),
        JSON.stringify(lastMessageUpdate),
        GROUP_INFO_CACHE_TTL,
      )
      .catch(() => {});

    // DB update (async, fire-and-forget — completes within cache TTL)
    this.groupConversationRepo
      .update(groupId, lastMessageUpdate)
      .catch((err) => this.logger.warn(`Async conversation update failed: ${err.message}`));

    // 8. Commit file keys via StorageService
    if (hasFiles) {
      await this.storageService.commitFiles(dto.fileKeys!);
    }

    // 9. Sequential: unread increment first, then MessageStatus batch INSERT
    //    Kept sequential (not Promise.all) — under high concurrency with saturated DB pool,
    //    parallel execution causes connection contention and INCREASES latency.
    if (otherUserIds.length > 0) {
      await this.participantRepo
        .createQueryBuilder()
        .update(Participant)
        .set({ unreadCount: () => 'unread_count + 1' })
        .where('conversation_id = :groupId', { groupId })
        .andWhere('user_id IN (:...otherUserIds)', { otherUserIds })
        .andWhere('deleted_at IS NULL')
        .execute();
    }

    // Redis pipeline INCR (1 round-trip regardless of N)
    await this.batchIncrementUnreadCounts(groupId, otherUserIds);

    // 9b. Create MessageStatus records for all participants (SENT by default) — batch INSERT
    await this.dataSource
      .createQueryBuilder()
      .insert()
      .into(MessageStatus)
      .values(
        allParticipantIds.map((participantId) => ({
          id: generateSnowflakeId(),
          messageId: messageId,
          userId: participantId,
          conversationId: groupId,
          status: MessageStatusEnum.SENT,
        })),
      )
      .execute();

    // 9c. Emit MessageSentEvent to sender (acknowledgment with server messageId)
    this.eventEmitter.emit(
      MessageSentEvent.eventName,
      new MessageSentEvent(groupId, messageId, userId, dto.clientMessageId, now),
    );

    // 10. Emit NewMessageEvent for Socket.IO broadcast
    this.eventEmitter.emit(
      NewMessageEvent.eventName,
      new NewMessageEvent(
        groupId,
        messageId,
        userId,
        senderName,
        dto.content || null,
        dto.messageType,
        attachments.map((att) => ({
          fileKey: att.fileKey,
          fileName: att.fileName,
          fileSize: Number(att.fileSize),
          contentType: att.contentType,
        })),
        now,
      ),
    );

    this.logger.log(`Message ${messageId} sent to group ${groupId} by user ${userId}`);

    // 11. Build and return response DTO
    const response = new MessageResponseDto();
    response.id = messageId;
    response.senderId = userId;
    response.senderDisplayName = senderName;
    response.content = dto.content || '';
    response.messageType = dto.messageType as MessageType;
    response.attachments = attachments.map((att) => {
      const attachmentDto = new MessageAttachmentResponseDto();
      attachmentDto.fileKey = att.fileKey;
      attachmentDto.fileName = att.fileName;
      attachmentDto.fileSize = Number(att.fileSize);
      attachmentDto.contentType = att.contentType;
      return attachmentDto;
    });
    response.createdAt = now;
    response.status = MessageStatusEnum.SENT; // Status for sender's own message

    return { ...response, isNew: true };
  }

  /**
   * Mark a message as delivered for a specific recipient.
   *
   * Called when the recipient's device acknowledges receipt via WebSocket.
   * Updates the MessageStatus record from SENT → DELIVERED.
   * Emits MessageDeliveredEvent to notify the sender.
   *
   * Requirements: real-time delivery acknowledgment
   */
  async markDelivered(messageId: string, recipientId: string, groupId: string): Promise<void> {
    const now = new Date();

    const result = await this.messageStatusRepo.update(
      {
        messageId,
        userId: recipientId,
        conversationId: groupId,
        status: MessageStatusEnum.SENT,
      },
      {
        status: MessageStatusEnum.DELIVERED,
        deliveredAt: now,
      },
    );

    if (result.affected && result.affected > 0) {
      // Get the sender of this message to notify them
      const message = await this.messageRepo.findOne({
        where: { id: messageId },
        select: ['senderId'],
      });

      if (message) {
        this.eventEmitter.emit(
          MessageDeliveredEvent.eventName,
          new MessageDeliveredEvent(groupId, messageId, recipientId, message.senderId, now),
        );
      }
    }
  }

  /**
   * Mark messages as read for a specific user in a group.
   *
   * Called when the user opens a conversation via WebSocket.
   * Updates all MessageStatus records from SENT/DELIVERED → READ.
   * Emits MessageReadEvent for each message to notify the senders.
   *
   * Requirements: real-time read acknowledgment
   */
  async markRead(userId: string, groupId: string, messageIds?: string[]): Promise<void> {
    const now = new Date();

    const whereConditions: any = {
      userId,
      conversationId: groupId,
      status: In([MessageStatusEnum.SENT, MessageStatusEnum.DELIVERED]),
    };

    if (messageIds && messageIds.length > 0) {
      whereConditions.messageId = In(messageIds);
    }

    // Get the records that will be updated (to know which senders to notify)
    const pendingStatuses = await this.messageStatusRepo.find({
      where: whereConditions,
      select: ['messageId', 'userId'],
    });

    if (pendingStatuses.length === 0) {
      return;
    }

    // Update all pending statuses to READ
    await this.messageStatusRepo.update(whereConditions, {
      status: MessageStatusEnum.READ,
      readAt: now,
    });

    // Batch fetch all message senders in one query (avoids N+1)
    const pendingMessageIds = pendingStatuses.map((s) => s.messageId);
    const messages = await this.messageRepo.find({
      where: { id: In(pendingMessageIds) },
      select: ['id', 'senderId'],
    });
    const senderMap = new Map(messages.map((m) => [m.id, m.senderId]));

    // Emit MessageReadEvent for each message (grouped by sender to avoid spam)
    const notifiedSenders = new Set<string>();
    for (const status of pendingStatuses) {
      const senderId = senderMap.get(status.messageId);
      if (!senderId || notifiedSenders.has(senderId)) continue;

      notifiedSenders.add(senderId);
      this.eventEmitter.emit(
        MessageReadEvent.eventName,
        new MessageReadEvent(groupId, status.messageId, userId, senderId, now),
      );
    }
  }

  /**
   * Add a member to a group conversation.
   *
   * - Validates the acting user has owner/admin role
   * - Checks group capacity (max 10 active participants)
   * - Handles re-add of soft-deleted participant (restore record)
   * - Rejects if user is already an active participant
   * - Generates system message
   * - Emits MemberAddedEvent
   *
   * Requirements: 2.1, 2.2, 2.6, 2.7, 2.8
   */
  @DistributedLock({
    key: (args: any[]) => `conversation:member:${args[1]}`,
    ttlMs: 10000,
    onFail: 'throw',
  })
  async addMember(userId: string, groupId: string, dto: AddMemberDto): Promise<void> {
    // 1. Validate the group exists
    const group = await this.groupConversationRepo.findOne({
      where: { id: groupId, deletedAt: IsNull() },
    });
    if (!group) {
      throw new NotFoundException('Group not found');
    }

    // 2. Block adding members to direct conversations
    if (group.type === ConversationType.DIRECT) {
      throw new BadRequestException(
        'Cannot add members to a direct conversation. Please create a new group instead.',
      );
    }

    // 3. Validate the acting user is owner or admin
    const actorParticipant = await this.participantRepo.findOne({
      where: { conversationId: groupId, userId, deletedAt: IsNull() },
    });
    if (!actorParticipant) {
      throw new ForbiddenException('You are not a participant of this group');
    }
    if (
      actorParticipant.role !== ParticipantRole.OWNER &&
      actorParticipant.role !== ParticipantRole.ADMIN
    ) {
      throw new ForbiddenException('Only owner or admin can add members to the group');
    }

    // 3. Validate the target user exists
    const targetUser = await this.userRepo.findOne({
      where: { id: dto.userId, deletedAt: IsNull() },
    });
    if (!targetUser) {
      throw new BadRequestException(`User ID ${dto.userId} does not exist or is not active`);
    }

    // 4. Check if user is already an active participant
    const existingActiveParticipant = await this.participantRepo.findOne({
      where: {
        conversationId: groupId,
        userId: dto.userId,
        deletedAt: IsNull(),
      },
    });
    if (existingActiveParticipant) {
      throw new BadRequestException(
        `User ${dto.userId} is already an active participant in this group`,
      );
    }

    // 5. Check group capacity (max 10 active participants)
    const activeCount = await this.participantRepo.count({
      where: { conversationId: groupId, deletedAt: IsNull() },
    });
    if (activeCount >= 50) {
      throw new BadRequestException('Group is full. Maximum participant count is 50');
    }

    // 6. Check for soft-deleted participant (re-add case)
    const softDeletedParticipant = await this.participantRepo
      .createQueryBuilder('gp')
      .withDeleted()
      .where('gp.conversation_id = :groupId', { groupId })
      .andWhere('gp.user_id = :userId', { userId: dto.userId })
      .andWhere('gp.deleted_at IS NOT NULL')
      .getOne();

    // Fetch actor user info for system message
    const actorUser = await this.userRepo.findOne({ where: { id: userId } });

    await this.dataSource.transaction(async (manager) => {
      if (softDeletedParticipant) {
        // Re-add: restore the soft-deleted record
        softDeletedParticipant.deletedAt = null;
        softDeletedParticipant.role = ParticipantRole.MEMBER;
        softDeletedParticipant.joinedAt = new Date();
        softDeletedParticipant.unreadCount = 0;
        await manager.save(Participant, softDeletedParticipant);
      } else {
        // New participant
        const newParticipant = manager.create(Participant, {
          conversationId: groupId,
          userId: dto.userId,
          role: ParticipantRole.MEMBER,
          unreadCount: 0,
        });
        await manager.save(Participant, newParticipant);
      }

      // 7. Generate system message
      const actorName = actorUser?.fullNameKanji || `User ${userId}`;
      const targetName = targetUser.fullNameKanji || `User ${dto.userId}`;
      const systemMessage = manager.create(Message, {
        conversationId: groupId,
        senderId: userId,
        clientMessageId: generateUUID(),
        content: `${targetName} was added by ${actorName}`,
        messageType: MessageType.SYSTEM,
      });
      await manager.save(Message, systemMessage);
    });

    // 8. Emit domain event
    const displayName = targetUser.fullNameKanji || `User ${dto.userId}`;
    this.eventEmitter.emit(
      MemberAddedEvent.eventName,
      new MemberAddedEvent(groupId, dto.userId, displayName, userId),
    );

    // 9. Invalidate caches
    await this.invalidateParticipantsCache(groupId);
    await this.invalidateUserGroupsCache(dto.userId);

    this.logger.log(`User ${dto.userId} added to group ${groupId} by user ${userId}`);
  }

  /**
   * Remove a member from a group conversation.
   *
   * - Validates the acting user has owner/admin role
   * - Validates role hierarchy (admin cannot remove owner/admin)
   * - Soft-deletes the participant record
   * - Generates system message
   * - Emits MemberRemovedEvent
   *
   * Requirements: 2.3, 2.6, 2.9
   */
  @DistributedLock({
    key: (args: any[]) => `conversation:member:${args[1]}`,
    ttlMs: 10000,
    onFail: 'throw',
  })
  async removeMember(userId: string, groupId: string, targetUserId: string): Promise<void> {
    // 1. Validate the group exists
    const group = await this.groupConversationRepo.findOne({
      where: { id: groupId, deletedAt: IsNull() },
    });
    if (!group) {
      throw new NotFoundException('Group not found');
    }

    // 2. Validate the acting user is owner or admin
    const actorParticipant = await this.participantRepo.findOne({
      where: { conversationId: groupId, userId, deletedAt: IsNull() },
    });
    if (!actorParticipant) {
      throw new ForbiddenException('You are not a participant of this group');
    }
    if (
      actorParticipant.role !== ParticipantRole.OWNER &&
      actorParticipant.role !== ParticipantRole.ADMIN
    ) {
      throw new ForbiddenException('Only owner or admin can remove members from the group');
    }

    // 3. Validate the target participant exists and is active
    const targetParticipant = await this.participantRepo.findOne({
      where: {
        conversationId: groupId,
        userId: targetUserId,
        deletedAt: IsNull(),
      },
    });
    if (!targetParticipant) {
      throw new BadRequestException(
        `User ${targetUserId} is not an active participant in this group`,
      );
    }

    // 4. Validate role hierarchy: admin cannot remove owner or another admin
    if (actorParticipant.role === ParticipantRole.ADMIN) {
      if (
        targetParticipant.role === ParticipantRole.OWNER ||
        targetParticipant.role === ParticipantRole.ADMIN
      ) {
        throw new ForbiddenException('Admin cannot remove the owner or another admin');
      }
    }

    // 5. Cannot remove yourself (use leaveGroup instead)
    if (targetUserId === userId) {
      throw new BadRequestException('Cannot remove yourself. Use leave group instead');
    }

    await this.dataSource.transaction(async (manager) => {
      // 6. Soft-delete the participant
      await manager.softRemove(Participant, targetParticipant);

      // 7. Generate system message
      const targetUser = await this.userRepo.findOne({
        where: { id: targetUserId },
      });
      const targetName = targetUser?.fullNameKanji || `User ${targetUserId}`;
      const actorUser = await this.userRepo.findOne({
        where: { id: userId },
      });
      const actorName = actorUser?.fullNameKanji || `User ${userId}`;

      const systemMessage = manager.create(Message, {
        conversationId: groupId,
        senderId: userId,
        clientMessageId: generateUUID(),
        content: `${targetName} was removed by ${actorName}`,
        messageType: MessageType.SYSTEM,
      });
      await manager.save(Message, systemMessage);
    });

    // 8. Emit domain event
    this.eventEmitter.emit(
      MemberRemovedEvent.eventName,
      new MemberRemovedEvent(groupId, targetUserId, userId),
    );

    // 9. Invalidate caches
    await this.invalidateParticipantsCache(groupId);
    await this.invalidateUserGroupsCache(targetUserId);

    this.logger.log(`User ${targetUserId} removed from group ${groupId} by user ${userId}`);
  }

  /**
   * Leave a group conversation.
   *
   * - Soft-deletes the participant record
   * - Handles ownership transfer if owner leaves (earliest admin, then earliest member)
   * - Auto-dissolves group if remaining active participants < 2
   * - Generates system message
   * - Emits MemberLeftEvent (and GroupDissolvedEvent if auto-dissolved)
   *
   * Requirements: 7.1, 7.2, 7.3, 7.6
   */
  @DistributedLock({
    key: (args: any[]) => `conversation:leave:${args[1]}`,
    ttlMs: 10000,
    onFail: 'throw',
  })
  async leaveGroup(userId: string, groupId: string): Promise<void> {
    // 1. Validate the group exists
    const group = await this.groupConversationRepo.findOne({
      where: { id: groupId, deletedAt: IsNull() },
    });
    if (!group) {
      throw new NotFoundException('Group not found');
    }

    // 2. Validate the user is an active participant
    const participant = await this.participantRepo.findOne({
      where: { conversationId: groupId, userId, deletedAt: IsNull() },
    });
    if (!participant) {
      throw new ForbiddenException('You are not a participant of this group');
    }

    const isOwner = participant.role === ParticipantRole.OWNER;

    await this.dataSource.transaction(async (manager) => {
      // 3. Soft-delete the participant
      await manager.softRemove(Participant, participant);

      // 4. Generate system message for leaving
      const leavingUser = await this.userRepo.findOne({
        where: { id: userId },
      });
      const leavingName = leavingUser?.fullNameKanji || `User ${userId}`;

      const systemMessage = manager.create(Message, {
        conversationId: groupId,
        senderId: userId,
        clientMessageId: generateUUID(),
        content: `${leavingName} has left the group`,
        messageType: MessageType.SYSTEM,
      });
      await manager.save(Message, systemMessage);

      // 5. Count remaining active participants
      const remainingCount = await manager.count(Participant, {
        where: { conversationId: groupId, deletedAt: IsNull() },
      });

      // 6. Auto-dissolve if remaining < 2
      if (remainingCount < 2) {
        await this.dissolveGroupInternal(manager, group, groupId);
        return;
      }

      // 7. Handle ownership transfer if owner is leaving
      if (isOwner) {
        await this.transferOwnership(manager, groupId, group);
      }
    });

    // 8. Emit domain event
    this.eventEmitter.emit(MemberLeftEvent.eventName, new MemberLeftEvent(groupId, userId));

    // 9. Invalidate caches
    await this.invalidateParticipantsCache(groupId);
    await this.invalidateUserGroupsCache(userId);

    // Check if group was auto-dissolved (remaining < 2)
    const remainingAfter = await this.participantRepo.count({
      where: { conversationId: groupId, deletedAt: IsNull() },
    });
    if (remainingAfter < 2) {
      const remainingParticipants = await this.participantRepo.find({
        where: { conversationId: groupId },
        withDeleted: true,
        select: ['userId'],
      });

      // Invalidate caches for all participants on dissolution
      await this.invalidateGroupInfoCache(groupId);
      for (const p of remainingParticipants) {
        await this.invalidateUserGroupsCache(p.userId);
      }

      this.eventEmitter.emit(
        GroupDissolvedEvent.eventName,
        new GroupDissolvedEvent(
          groupId,
          remainingParticipants.map((p) => p.userId),
        ),
      );
    }

    this.logger.log(`User ${userId} left group ${groupId}`);
  }

  /**
   * Get message history for a group conversation with cursor-based pagination.
   *
   * - Validates participant membership
   * - Queries messages WHERE conversation_id = groupId AND id < cursor (if provided)
   * - ORDER BY id DESC, LIMIT dto.limit
   * - Includes sender display name by joining with UserEntity
   * - Includes attachment metadata by joining with MessageAttachment
   * - Computes nextCursor and hasMore (queries limit+1 to check)
   *
   * Requirements: 5.1, 5.2, 5.3, 5.4, 5.5, 5.6, 5.7
   */
  async getMessages(
    userId: string,
    groupId: string,
    dto: GetMessagesDto,
  ): Promise<PaginatedMessagesResponseDto> {
    // 1. Validate participant membership
    const participant = await this.participantRepo.findOne({
      where: {
        conversationId: groupId,
        userId: userId,
        deletedAt: IsNull(),
      },
    });

    if (!participant) {
      throw new ForbiddenException('You are not a participant of this group conversation');
    }

    // 2. Build query with cursor-based pagination
    const limit = dto.limit ?? 50;

    const queryBuilder = this.messageRepo
      .createQueryBuilder('message')
      .leftJoinAndSelect('message.attachments', 'attachment')
      .leftJoin(UserEntity, 'sender', 'sender.id = message.sender_id')
      .addSelect('sender.fullNameKanji', 'senderFullName')
      .where('message.conversation_id = :groupId', { groupId });

    // Apply cursor filter if provided
    if (dto.cursor) {
      queryBuilder.andWhere('message.id < :cursor', { cursor: dto.cursor });
    }

    // Order by id DESC and fetch limit+1 to determine hasMore
    queryBuilder.orderBy('message.id', 'DESC').take(limit + 1);

    const rawResults = await queryBuilder.getRawAndEntities();
    const messages = rawResults.entities;
    const rawRows = rawResults.raw;

    // 3. Determine hasMore and trim to limit
    const hasMore = messages.length > limit;
    const paginatedMessages = messages.slice(0, limit);

    // 4. Build a map of sender display names from raw results
    const senderNameMap = new Map<string, string>();
    for (const raw of rawRows) {
      const senderId = raw.message_sender_id || raw.sender_id;
      const senderName = raw.senderFullName || raw.sender_full_name || '';
      if (senderId && !senderNameMap.has(senderId)) {
        senderNameMap.set(senderId, senderName);
      }
    }

    // If some sender names are missing from the join, fetch them separately
    const missingSenderIds = paginatedMessages
      .filter((m) => !senderNameMap.has(m.senderId))
      .map((m) => m.senderId);

    if (missingSenderIds.length > 0) {
      const uniqueMissing = [...new Set(missingSenderIds)];
      const senders = await this.userRepo.find({
        where: { id: In(uniqueMissing) },
        select: ['id', 'fullNameKanji'],
      });
      for (const sender of senders) {
        senderNameMap.set(sender.id, sender.fullNameKanji || '');
      }
    }

    // 5. Batch fetch message status for the current user's sent messages
    const messageIds = paginatedMessages.map((m) => m.id);
    let statusMap = new Map<string, string>();
    if (messageIds.length > 0) {
      const statuses = await this.messageStatusRepo.find({
        where: { messageId: In(messageIds), userId },
        select: ['messageId', 'status'],
      });
      for (const s of statuses) {
        statusMap.set(s.messageId, s.status);
      }
    }

    // 6. Map to response DTOs
    const data: MessageResponseDto[] = paginatedMessages.map((message) => {
      const response = new MessageResponseDto();
      response.id = message.id;
      response.senderId = message.senderId;
      response.senderDisplayName = senderNameMap.get(message.senderId) || '';
      response.content = message.content;
      response.messageType = message.messageType as MessageType;
      response.attachments = (message.attachments || []).map((att) => {
        const attachmentDto = new MessageAttachmentResponseDto();
        attachmentDto.fileKey = att.fileKey;
        attachmentDto.fileName = att.fileName;
        attachmentDto.fileSize = Number(att.fileSize);
        attachmentDto.contentType = att.contentType;
        return attachmentDto;
      });
      response.createdAt = message.createdAt;
      // Include status only for the current user's own sent messages
      response.status = message.senderId === userId ? (statusMap.get(message.id) ?? 'sent') : null;
      return response;
    });

    // 7. Compute nextCursor
    const nextCursor =
      hasMore && paginatedMessages.length > 0
        ? paginatedMessages[paginatedMessages.length - 1].id
        : null;

    // 7. Return paginated response
    const result = new PaginatedMessagesResponseDto();
    result.data = data;
    result.nextCursor = nextCursor;
    result.hasMore = hasMore;

    return result;
  }

  /**
   * Get group detail with participant list and avatar presigned GET URL.
   *
   * - Validates participant membership
   * - Joins participants with UserEntity for display names
   * - Generates presigned GET URL for avatar via StorageService
   * - Returns ConversationDetailResponseDto

   * Requirements: 6.6

  async getGroupDetail(userId: string, groupId: string): Promise<ConversationDetailResponseDto> {

        conversationId: groupId,

  /**
   * Update group information (name and/or avatar).
   *
   * - Validates the acting user has owner/admin role
   * - Trims and validates name (1-100 chars, non-blank after trim)
   * - Validates avatarFileKey exists in uploaded_file records
   * - Persists changes
   * - Commits avatar file key via StorageService
   * - Generates system messages for each change
   * - Emits GroupUpdatedEvent
   *
   * Requirements: 6.1, 6.2, 6.3, 6.4, 6.5
   */
  async updateGroup(
    userId: string,
    groupId: string,
    dto: UpdateGroupDto,
  ): Promise<ConversationResponseDto> {
    // 1. Validate the group exists
    const group = await this.groupConversationRepo.findOne({
      where: { id: groupId, deletedAt: IsNull() },
    });
    if (!group) {
      throw new NotFoundException('Group not found');
    }

    // 2. Validate the acting user is owner or admin
    const actorParticipant = await this.participantRepo.findOne({
      where: { conversationId: groupId, userId, deletedAt: IsNull() },
    });
    if (!actorParticipant) {
      throw new ForbiddenException('You are not a participant of this group');
    }
    if (
      actorParticipant.role !== ParticipantRole.OWNER &&
      actorParticipant.role !== ParticipantRole.ADMIN
    ) {
      throw new ForbiddenException('Only owner or admin can update group information');
    }

    // 3. Validate and process name if provided
    let trimmedName: string | undefined;
    if (dto.name !== undefined) {
      trimmedName = dto.name.trim();
      if (!trimmedName || trimmedName.length > 100) {
        throw new BadRequestException(
          'Group name must be between 1 and 100 characters after trimming',
        );
      }
    }

    // 4. Validate avatarFileKey exists if provided
    if (dto.avatarFileKey !== undefined) {
      const uploadedFile = await this.dataSource.getRepository(UploadedFile).findOne({
        where: { fileKey: dto.avatarFileKey, deletedAt: IsNull() },
      });
      if (!uploadedFile) {
        throw new BadRequestException('Invalid file reference: avatar file key does not exist');
      }
    }

    // 5. Get actor user info for system messages
    const actorUser = await this.userRepo.findOne({ where: { id: userId } });
    const actorName = actorUser?.fullNameKanji || `User ${userId}`;

    // 6. Persist changes and generate system messages in a transaction
    await this.dataSource.transaction(async (manager) => {
      // Update name if provided
      if (trimmedName !== undefined) {
        const oldName = group.name;
        group.name = trimmedName;

        // Generate system message for name change
        const nameChangeMessage = manager.create(Message, {
          conversationId: groupId,
          senderId: userId,
          clientMessageId: generateUUID(),
          content: `${actorName} changed the group name from "${oldName}" to "${trimmedName}"`,
          messageType: MessageType.SYSTEM,
        });
        await manager.save(Message, nameChangeMessage);
      }

      // Update avatar if provided
      if (dto.avatarFileKey !== undefined) {
        group.avatarFileKey = dto.avatarFileKey;

        // Generate system message for avatar change
        const avatarChangeMessage = manager.create(Message, {
          conversationId: groupId,
          senderId: userId,
          clientMessageId: generateUUID(),
          content: `${actorName} changed the group avatar`,
          messageType: MessageType.SYSTEM,
        });
        await manager.save(Message, avatarChangeMessage);
      }

      // Save the updated group
      await manager.save(Conversation, group);
    });

    // 7. Commit avatar file key via StorageService if provided
    if (dto.avatarFileKey !== undefined) {
      await this.storageService.commitFiles([dto.avatarFileKey]);
    }

    // 8. Emit GroupUpdatedEvent
    this.eventEmitter.emit(
      GroupUpdatedEvent.eventName,
      new GroupUpdatedEvent(groupId, trimmedName, dto.avatarFileKey),
    );

    // 9. Invalidate group info cache
    await this.invalidateGroupInfoCache(groupId);

    this.logger.log(`Group ${groupId} updated by user ${userId}`);

    // 10. Return response DTO
    let avatarUrl: string | null = null;
    if (group.avatarFileKey) {
      try {
        const urls = await this.storageService.getPresignedGetUrl([group.avatarFileKey]);
        avatarUrl = urls[group.avatarFileKey] || null;
      } catch {
        avatarUrl = null;
      }
    }

    const response = new ConversationResponseDto();
    response.id = group.id;
    response.name = group.name;
    response.avatarUrl = avatarUrl;
    response.lastMessagePreview = null;
    response.unreadCount = 0;
    response.lastMessageAt = group.lastMessageAt;

    return response;
  }

  /**
   * Dissolve a group conversation (owner only).

   * - Validates the acting user is the owner
   * - Persists system message indicating dissolution
   * - Soft-deletes all participants and the conversation
   * - Messages are preserved (not soft-deleted)
   * - Emits GroupDissolvedEvent
   *
   * Requirements: 7.4, 7.5, 7.7, 7.8
   */
  @DistributedLock({
    key: (args: any[]) => `conversation:dissolve:${args[1]}`,
    ttlMs: 10000,
    onFail: 'throw',
  })
  async dissolveGroup(userId: string, groupId: string): Promise<void> {
    // 1. Validate the group exists
    const group = await this.groupConversationRepo.findOne({
      where: { id: groupId, deletedAt: IsNull() },
    });
    if (!group) {
      throw new NotFoundException('Group not found');
    }

    // 2. Validate the acting user is the owner
    const actorParticipant = await this.participantRepo.findOne({
      where: { conversationId: groupId, userId, deletedAt: IsNull() },
    });
    if (!actorParticipant) {
      throw new ForbiddenException('You are not a participant of this group');
    }
    if (actorParticipant.role !== ParticipantRole.OWNER) {
      throw new ForbiddenException('Only the owner can dissolve the group');
    }

    // 3. Get all participant IDs before dissolution (for event emission)
    const allParticipants = await this.participantRepo.find({
      where: { conversationId: groupId, deletedAt: IsNull() },
      select: ['userId'],
    });
    const participantIds = allParticipants.map((p) => p.userId);

    // 4. Dissolve the group using the internal helper
    await this.dataSource.transaction(async (manager) => {
      await this.dissolveGroupInternal(manager, group, groupId);
    });

    // 5. Emit GroupDissolvedEvent
    this.eventEmitter.emit(
      GroupDissolvedEvent.eventName,
      new GroupDissolvedEvent(groupId, participantIds),
    );

    // 6. Invalidate all related caches on dissolution
    await this.invalidateParticipantsCache(groupId);
    await this.invalidateGroupInfoCache(groupId);
    for (const pId of participantIds) {
      await this.invalidateUserGroupsCache(pId);
    }

    this.logger.log(`Group ${groupId} dissolved by owner ${userId}`);
  }

  /**
   * Internal helper: Transfer ownership to the next eligible participant.
   * Priority: earliest admin by joined_at, then earliest member by joined_at.
   */
  private async transferOwnership(
    manager: any,
    groupId: string,
    group: Conversation,
  ): Promise<void> {
    // Find earliest admin
    const earliestAdmin = await manager.findOne(Participant, {
      where: {
        conversationId: groupId,
        role: ParticipantRole.ADMIN,
        deletedAt: IsNull(),
      },
      order: { joinedAt: 'ASC' },
    });

    if (earliestAdmin) {
      earliestAdmin.role = ParticipantRole.OWNER;
      await manager.save(Participant, earliestAdmin);
      group.ownerId = earliestAdmin.userId;
      await manager.save(Conversation, group);
      return;
    }

    // Find earliest member
    const earliestMember = await manager.findOne(Participant, {
      where: {
        conversationId: groupId,
        role: ParticipantRole.MEMBER,
        deletedAt: IsNull(),
      },
      order: { joinedAt: 'ASC' },
    });

    if (earliestMember) {
      earliestMember.role = ParticipantRole.OWNER;
      await manager.save(Participant, earliestMember);
      group.ownerId = earliestMember.userId;
      await manager.save(Conversation, group);
    }
  }

  /**
   * Internal helper: Dissolve a group (soft-delete conversation and all participants).
   * Messages are preserved (not soft-deleted).
   */
  private async dissolveGroupInternal(
    manager: any,
    group: Conversation,
    groupId: string,
  ): Promise<void> {
    // Generate dissolution system message
    const dissolutionMessage = manager.create(Message, {
      conversationId: groupId,
      senderId: group.ownerId,
      clientMessageId: generateUUID(),
      content: 'Group has been dissolved',
      messageType: MessageType.SYSTEM,
    });
    await manager.save(Message, dissolutionMessage);

    // Soft-delete all remaining active participants
    const remainingParticipants = await manager.find(Participant, {
      where: { conversationId: groupId, deletedAt: IsNull() },
    });
    for (const p of remainingParticipants) {
      await manager.softRemove(Participant, p);
    }

    // Soft-delete the group conversation
    await manager.softRemove(Conversation, group);
  }

  /**
   * Get the list of group conversations for a user.
   *
   * - Queries active Conversations where user is an active participant
   * - Joins last message for preview (truncated to 100 chars)
   * - Includes unread count from participant record (with Redis fallback)
   * - Sorts by COALESCE(last_message_at, created_at) DESC
   * - Cursor-based pagination using the group's Snowflake ID
   *
   * Requirements: 9.1, 9.2, 9.3, 9.4, 9.5
   */
  async getUserGroups(
    userId: string,
    dto: GetGroupsDto,
  ): Promise<PaginatedConversationsResponseDto> {
    const limit = dto.limit ?? 20;

    // Build query: join Conversation with Participant where user is active participant
    const queryBuilder = this.groupConversationRepo
      .createQueryBuilder('gc')
      .innerJoin(
        Participant,
        'gp',
        'gp.conversation_id = gc.id AND gp.user_id = :userId AND gp.deleted_at IS NULL',
        { userId },
      )
      .leftJoin(Message, 'lm', 'lm.id = gc.last_message_id')
      .addSelect('gp.unread_count', 'unreadCount')
      .addSelect('lm.content', 'lastMessageContent')
      .where('gc.deleted_at IS NULL');

    // Apply cursor-based pagination
    if (dto.cursor) {
      // Get the sort value of the cursor group to compare
      const cursorGroup = await this.groupConversationRepo.findOne({
        where: { id: dto.cursor },
        select: ['id', 'lastMessageAt', 'createdAt'],
      });

      if (cursorGroup) {
        const cursorSortDate = cursorGroup.lastMessageAt || cursorGroup.createdAt;
        queryBuilder.andWhere(
          '(COALESCE(gc.last_message_at, gc.created_at) < :cursorDate OR (COALESCE(gc.last_message_at, gc.created_at) = :cursorDate AND gc.id < :cursorId))',
          { cursorDate: cursorSortDate, cursorId: dto.cursor },
        );
      }
    }

    // Sort by COALESCE(last_message_at, created_at) DESC, then by id DESC for tie-breaking
    queryBuilder
      .orderBy('COALESCE(gc.last_message_at, gc.created_at)', 'DESC')
      .addOrderBy('gc.id', 'DESC')
      .limit(limit + 1); // Fetch one extra to determine hasMore

    // Execute raw query to get the joined data
    const rawResults = await queryBuilder
      .select([
        'gc.id AS id',
        'gc.name AS name',
        'gc.type AS type',
        'gc.avatar_file_key AS "avatarFileKey"',
        'gc.last_message_at AS "lastMessageAt"',
        'gc.created_at AS "createdAt"',
        'gp.unread_count AS "unreadCount"',
        'lm.content AS "lastMessageContent"',
      ])
      .getRawMany();

    // Determine hasMore
    const hasMore = rawResults.length > limit;
    const results = hasMore ? rawResults.slice(0, limit) : rawResults;

    // Get avatar presigned URLs for groups that have avatars
    const avatarFileKeys = results
      .map((r) => r.avatarFileKey)
      .filter((key): key is string => key !== null && key !== undefined);

    let avatarUrls: Record<string, string> = {};
    if (avatarFileKeys.length > 0) {
      try {
        avatarUrls = await this.storageService.getPresignedGetUrl(avatarFileKeys);
      } catch {
        // If presigned URL generation fails, continue with null avatars
      }
    }

    // Batch fetch unread counts from Redis in one MGET call
    let redisUnreadMap: Record<string, number> = {};
    try {
      const unreadKeys = results.map((row) => getUnreadCountCacheKey(row.id, userId));
      const redisValues = await this.redisCacheService.mget(unreadKeys);
      for (let i = 0; i < unreadKeys.length; i++) {
        if (redisValues[i] !== null) {
          redisUnreadMap[results[i].id] = parseInt(redisValues[i]!, 10) || 0;
        }
      }
    } catch {
      // Redis unavailable, use DB values
    }

    // Try to get unread counts from Redis (faster), fall back to DB value
    const groupData: ConversationResponseDto[] = results.map((row) => {
      let unreadCount = parseInt(row.unreadCount || '0', 10) || 0;

      // Use Redis value if available (already batch-fetched above)
      if (redisUnreadMap[row.id] !== undefined) {
        unreadCount = redisUnreadMap[row.id];
      }

      // Truncate last message preview to 100 chars
      let lastMessagePreview: string | null = null;
      if (row.lastMessageContent) {
        lastMessagePreview =
          row.lastMessageContent.length > 100
            ? row.lastMessageContent.substring(0, 100)
            : row.lastMessageContent;
      }

      const response = new ConversationResponseDto();
      response.id = row.id;
      response.name = row.name;
      response.type = row.type || ConversationType.GROUP;
      response.avatarUrl = row.avatarFileKey ? avatarUrls[row.avatarFileKey] || null : null;
      response.lastMessagePreview = lastMessagePreview;
      response.unreadCount = unreadCount;
      response.lastMessageAt = row.lastMessageAt || null;

      return response;
    });

    // Compute nextCursor (Snowflake ID of the last group in the current page)
    const nextCursor = results.length > 0 ? results[results.length - 1].id : null;

    const paginatedResponse = new PaginatedConversationsResponseDto();
    paginatedResponse.data = groupData;
    paginatedResponse.nextCursor = hasMore ? nextCursor : null;
    paginatedResponse.hasMore = hasMore;

    return paginatedResponse;
  }
