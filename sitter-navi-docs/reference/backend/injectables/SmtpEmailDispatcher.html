import { SESv2Client, SendEmailCommand } from '@aws-sdk/client-sesv2';
import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import nodemailer, { Transporter } from 'nodemailer';
import { IEmailDispatcher, EmailJob } from './email-dispatcher.interface';
import { EmailTemplateService } from './email-template.service';
import { EmailRendererService } from './email-renderer.service';
import { TokenBucket } from '@infrastructure/utils/token-bucket';

/** SMTP error codes that are permanent (should NOT be retried) */
const PERMANENT_SMTP_CODES = new Set([501, 502, 503, 504, 550, 551, 552, 553, 554, 555]);

/**
 * SMTP / AWS SES email dispatcher.
 *
 * - Handles both 'template' and 'custom' EmailJob types.
 * - For 'template': looks up DB template → renders EJS → sends.
 * - For 'custom': sends the already-provided subject/html directly.
 * - Uses in-memory TokenBucket to enforce rate limits (14 emails/sec for SES).
 * - Never throws — returns DispatchResult with success/error.
 */
@Injectable()
export class SmtpEmailDispatcher implements IEmailDispatcher {
  private readonly transporter: Transporter;
  private readonly logger = new Logger(SmtpEmailDispatcher.name);
  private readonly tokenBucket: TokenBucket;

  constructor(
    private readonly configService: ConfigService,
    private readonly templateService: EmailTemplateService,
    private readonly renderer: EmailRendererService,
  ) {
    const rate = this.configService.get<number>('email.ratePerSecond') ?? 14;
    this.tokenBucket = new TokenBucket({ rate, burst: rate });

    if (configService.get<boolean>('email.enable_aws_ses')) {
      const sesClient = new SESv2Client({
        region: configService.get<string>('aws.region') || 'ap-northeast-1',
      });

      this.transporter = nodemailer.createTransport({
        SES: { sesClient, SendEmailCommand },
        sendingRate: rate,
      });

      this.logger.log(`Email dispatcher initialized: AWS SES (${rate} req/s)`);
    } else {
      this.transporter = nodemailer.createTransport(configService.get('email.server'));

      this.logger.log('Email dispatcher initialized: SMTP');
    }
  }

  /**
   * Dispatch an email job.
   *
   * - `type: 'template'` → look up template → render EJS → send.
   * - `type: 'custom'`   → send the provided subject/html directly.
   */
  async dispatch(job: EmailJob) {
    let to: string;
    let subject: string;
    let html: string;
    // For template jobs, the subject/body are rendered here and must be
    // written back to the outbox row (which was created empty).
    let renderedPersist: Record<string, unknown> | undefined;

    if (job.type === 'template') {
      // Render on-demand (CPU-bound — ok for single emails, but for bulk,
      // BullMQ workers handle this in a separate process)
      const template = await this.templateService.findByTemplateName(job.templateName);
      if (!template || !template.isActive) {
        this.logger.error(`Template "${job.templateName}" not found or inactive`);
        return {
          success: false as const,
          error: 'Template not found',
          permanent: true,
        };
      }

      try {
        const rendered = await this.renderer.render(template, job.data);
        subject = rendered.subject;
        html = rendered.html;
        renderedPersist = { subject, body: html };
      } catch (err) {
        this.logger.error(`Render failed for template "${job.templateName}"`, (err as Error).stack);
        return {
          success: false as const,
          error: 'Render failed',
          permanent: true,
        };
      }

      to = job.to;
    } else {
      // Custom — already rendered by caller (sendCustomEmail)
      to = job.to;
      subject = job.subject;
      html = job.html;
    }

    try {
      await this.tokenBucket.waitForToken();

      await this.transporter.sendMail({
        from: this.configService.get<string>('email.from_email'),
        to,
        subject,
        html,
      });

      this.logger.debug(`Email sent to=${to} outboxId=${job.outboxId}`);
      return { success: true as const, persist: renderedPersist };
    } catch (error) {
      const message = error instanceof Error ? error.message : 'Unknown error';
      const smtpCode = (error as any)?.responseCode ?? (error as any)?.code;
      const permanent = typeof smtpCode === 'number' && PERMANENT_SMTP_CODES.has(smtpCode);

      this.logger.error(
        `Email failed to=${to} outboxId=${job.outboxId}${permanent ? ' (PERMANENT)' : ''}: ${message}`,
      );

      // Send failed, but if the template was already rendered, persist the
      // subject/body anyway so the outbox row reflects what was built.
      return {
        success: false as const,
        error: message,
        permanent,
        persist: renderedPersist,
      };
    }
  }
}
