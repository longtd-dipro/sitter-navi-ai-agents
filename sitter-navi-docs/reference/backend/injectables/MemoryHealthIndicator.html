import { Injectable, Optional } from '@nestjs/common';
import { HealthStatus } from '../dto/health-response.dto';
import { IHealthIndicator } from './health-indicator.interface';
import { MemoryMonitorService } from '../memory-monitor.service';

@Injectable()
export class MemoryHealthIndicator implements IHealthIndicator {
  readonly name = 'memory';

  /**
   * Heap usage ratio threshold (0-1).
   * If heapUsed / heapTotal exceeds this, status becomes degraded.
   * Default: 0.9 (90%) — indicates memory leak, GC under pressure.
   */
  private readonly heapThreshold = 0.9;

  /**
   * RSS threshold in bytes.
   * If RSS exceeds this, status becomes degraded.
   * Default: 450MB — typical for NestJS app. Adjust per env.
   */
  private readonly rssThreshold = 450 * 1024 * 1024;

  constructor(
    @Optional()
    private readonly memoryMonitor?: MemoryMonitorService,
  ) {}

  async check() {
    const mem = process.memoryUsage();

    const heapTotal = this.formatBytes(mem.heapTotal);
    const heapUsed = this.formatBytes(mem.heapUsed);
    const rss = this.formatBytes(mem.rss);
    const external = this.formatBytes(mem.external);

    const heapRatio = mem.heapUsed / mem.heapTotal;
    const rssExceeded = mem.rss > this.rssThreshold;
    const heapExceeded = heapRatio > this.heapThreshold;

    const status = rssExceeded || heapExceeded ? HealthStatus.DEGRADED : HealthStatus.OK;

    const warnings: string[] = [];
    if (rssExceeded) {
      warnings.push(`RSS (${rss}) exceeds threshold (${this.formatBytes(this.rssThreshold)})`);
    }
    if (heapExceeded) {
      warnings.push(
        `Heap usage ${(heapRatio * 100).toFixed(1)}% exceeds threshold (${(this.heapThreshold * 100).toFixed(0)}%) — possible memory leak`,
      );
    }

    const trend = this.computeTrend();

    return {
      name: this.name,
      status,
      metadata: {
        heapTotal,
        heapUsed,
        heapUsagePercent: `${(heapRatio * 100).toFixed(1)}%`,
        rss,
        external,
        rssThreshold: this.formatBytes(this.rssThreshold),
        heapThreshold: `${(this.heapThreshold * 100).toFixed(0)}%`,
        ...(trend && { trend }),
        ...(warnings.length > 0 && { warnings }),
      },
    };
  }

  /**
   * Compute memory trend from the monitor's sliding window.
   * Answers: has memory been growing steadily, or was that just a spike?
   */
  private computeTrend() {
    if (!this.memoryMonitor) return undefined;

    const recent = this.memoryMonitor.getRecentHistory();
    if (recent.length < 2) return undefined;

    const oldest = recent[0];
    const newest = recent[recent.length - 1];
    const growthPercent = ((newest.rss - oldest.rss) / oldest.rss) * 100;
    const consecutiveGrowth = this.memoryMonitor.getConsecutiveGrowth();

    return {
      direction: growthPercent > 5 ? 'up' : growthPercent < -5 ? 'down' : 'stable',
      rssGrowthPercent: `${growthPercent > 0 ? '+' : ''}${growthPercent.toFixed(1)}%`,
      consecutiveGrowth,
      samplesAnalyzed: recent.length,
      periodSeconds: (newest.timestamp - oldest.timestamp) / 1000,
      ...(consecutiveGrowth >= 3 && {
        warning: `RSS has been growing for ${consecutiveGrowth} consecutive samples — possible memory leak`,
      }),
    };
  }

  private formatBytes(bytes: number): string {
    const units = ['B', 'KB', 'MB', 'GB'];
    let value = bytes;
    let unitIndex = 0;

    while (value >= 1024 && unitIndex < units.length - 1) {
      value /= 1024;
      unitIndex++;
    }

    return `${value.toFixed(1)} ${units[unitIndex]}`;
  }
}
