import { UserOtpSentEvent } from '@events/user';
import { addMinutes, generateUUID, isAfter, numericId } from '@foundation';
import { BadRequestException, Injectable, UnauthorizedException } from '@nestjs/common';
import { EventEmitter2 } from '@nestjs/event-emitter';
import { JwtService } from '@nestjs/jwt';
import { UserEntity } from '@modules/user/user.entity';
import { UserRepository } from '@modules/user/user.repository';
import { AuthRepository } from '../auth.repository';
import { OtpSentResponseDto, VerifyOtpDto, VerifyOtpResponseDto } from '../dto/otp';
import { OTPEntity, OtpPurpose } from '../otp.entity';
import { RefreshTokenService } from './refresh-token.service';

const DEFAULT_OTP_RETRY = 5;

/**
 * Lifetime of the access token issued once an OTP is verified. Kept
 * deliberately short: the token is an immediate, single-purpose exchange
 * credential (e.g. the next step of the password-reset flow), not a
 * long-lived session token. Overrides the module-wide `jwt.expireTime`.
 */
const OTP_ACCESS_TOKEN_EXPIRES_IN = '10m';

/**
 * OtpService — owns the OTP lifecycle for the Auth module.
 *
 * Extracted from {@link AuthService}. Only the OTP "identifier" concept
 * and the persistence handle ({@link AuthRepository}) are kept here for
 * now; the verification / resend business logic is intentionally left
 * unimplemented and will be rewritten in this service.
 */
@Injectable()
export class OtpService {
  private MaxOtpRetry: number;
  private OtpExpirationMinutes: number;

  constructor(
    private readonly eventEmitter: EventEmitter2,
    private readonly authRepository: AuthRepository,
    private readonly jwtService: JwtService,
    private readonly userRepository: UserRepository,
    private readonly refreshTokenService: RefreshTokenService,
  ) {
    this.MaxOtpRetry = DEFAULT_OTP_RETRY;
    this.OtpExpirationMinutes = 5;
  }

  /**
   * Issue + send a login OTP for an already-authenticated (email+password
   * verified) user. The OTP is keyed on `userId` (unique → one live OTP per
   * user) with purpose {@link OtpPurpose.LOGIN_VERIFY}; the verify step loads
   * the user back from `userId` to mint the session, so no PII is stored here.
   */
  async sendLoginOtp(user: UserEntity): Promise<OtpSentResponseDto> {
    const { identifier, otp, expiredAt, expires } = await this.generateOTP();

    await this.authRepository.upsertOtp({
      userId: user.id,
      otp,
      expiredAt,
      identifier,
      purpose: OtpPurpose.LOGIN_VERIFY,
    });

    const eventPayload = new UserOtpSentEvent(user.email, otp, expires);
    this.eventEmitter.emit(UserOtpSentEvent.eventName, eventPayload);

    return { identifier, expiredAt };
  }

  async verifyOtp(dto: VerifyOtpDto): Promise<VerifyOtpResponseDto> {
    const otpItem = await this.consumeOtp(dto);

    // Login OTP: complete the second factor by minting a full session.
    if (otpItem.purpose === OtpPurpose.LOGIN_VERIFY) {
      return this.issueLoginSession(otpItem.userId);
    }

    // Password-reset / email-verify: short-lived, single-purpose exchange token.
    return {
      accessToken: this.jwtService.sign(
        {
          identifier: dto.identifier,
          email: otpItem?.meta?.email || null,
        },
        { expiresIn: OTP_ACCESS_TOKEN_EXPIRES_IN },
      ),
    };
  }

  /**
   * Validate an OTP attempt (existence, expiry, retry cap, code match) and
   * consume it (hard-delete) on success, returning the matched row. Throws a
   * {@link BadRequestException} on any failure — shared by every verify flow.
   */
  private async consumeOtp(dto: VerifyOtpDto): Promise<OTPEntity> {
    const { identifier, code } = dto;
    const otpItem = await this.authRepository.findByIdentifier(identifier);
    if (!otpItem) {
      throw new BadRequestException('auth.wrong_otp');
    }
    if (isAfter(new Date(), otpItem.expiredAt)) {
      await this.authRepository.softDelete(otpItem.id);
      throw new BadRequestException('auth.otp_expired');
    }

    if (otpItem.otp !== code) {
      const maxRetry = this.MaxOtpRetry > 0 ? this.MaxOtpRetry : DEFAULT_OTP_RETRY;
      if (otpItem.retry + 1 >= maxRetry) {
        await this.authRepository.hardDelete(otpItem.id);
      } else {
        await this.authRepository.incrementRetry(otpItem.id, otpItem.retry + 1);
      }
      throw new BadRequestException('auth.wrong_otp');
    }

    await this.authRepository.hardDelete(otpItem.id);

    return otpItem;
  }

  /**
   * Mint a full login session (access + refresh) for the user behind a
   * verified login OTP. Re-checks account state in case it changed between
   * the password step and OTP verification.
   */
  private async issueLoginSession(userId: string): Promise<VerifyOtpResponseDto> {
    const user = await this.userRepository.findById(userId);
    if (!user) {
      throw new BadRequestException('auth.user_not_valid');
    }
    if (user.isLocked) {
      throw new UnauthorizedException('auth.account_locked');
    }

    const { id, email, role } = user;
    const accessToken = this.jwtService.sign({ id, email, role });
    const { refreshToken, refreshTokenExpiresAt } = await this.refreshTokenService.issueForUser(
      user.id,
    );

    return { accessToken, refreshToken, refreshTokenExpiresAt, data: user };
  }

  async resendOtp(identifier: string): Promise<OtpSentResponseDto> {
    const otpItem = await this.authRepository.findByIdentifierWithDeleted(identifier);
    if (!otpItem) {
      throw new BadRequestException('auth.wrong_otp');
    }

    const { identifier: newIdentifier, otp, expiredAt, expires } = await this.generateOTP();

    otpItem.identifier = newIdentifier;
    otpItem.otp = otp;
    otpItem.expiredAt = expiredAt;
    otpItem.retry = 0;
    otpItem.isUsed = false;
    otpItem.deletedAt = null;
    await this.authRepository.save(otpItem);

    const email = otpItem?.meta?.email || null;
    if (email) {
      const eventName = UserOtpSentEvent.eventName;
      const eventPayload = new UserOtpSentEvent(email, otp, expires);
      this.eventEmitter.emit(eventName, eventPayload);
    }

    return { identifier: newIdentifier, expiredAt };
  }

  async sendOtp(email: string): Promise<OtpSentResponseDto> {
    const { identifier, expiredAt, otp, expires } = await this.generateOTP();

    await this.authRepository.insertOtp(
      {
        otp,
        expiredAt,
        identifier,
        purpose: OtpPurpose.EMAIL_VERIFY,
      },
      { email },
    );

    const eventName = UserOtpSentEvent.eventName;
    const eventPayload = new UserOtpSentEvent(email, otp, expires);

    this.eventEmitter.emit(eventName, eventPayload);

    return { identifier, expiredAt };
  }

  async generateOTP(
    otpSize: number = 6,
  ): Promise<{ identifier: string; expiredAt: Date; otp: string; expires: number }> {
    const otp = numericId(otpSize);
    const identifier = generateUUID();
    const expiredAt = addMinutes(new Date(), this.OtpExpirationMinutes);
    const expires = this.OtpExpirationMinutes;
    return { identifier, expiredAt, otp, expires };
  }
}
