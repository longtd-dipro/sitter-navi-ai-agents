Description

Central mail service — orchestrates template validation, outbox persistence, and dispatch. EJS rendering is delegated to the dispatcher (SmtpEmailDispatcher or BullMqEmailDispatcher), keeping the HTTP server lightweight.

Public API:

  • sendByTemplate(templateName, to, data) — validate → outbox → dispatch raw
  • sendCustomEmail(to, subject, body) — outbox → dispatch pre-rendered

Event handlers:

  • Dynamically registered via EventEmitter2.on() for each MAIL_EVENT_MAP entry.
  • Fire-and-forget: errors are logged but never re-thrown.

import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { EventEmitter2 } from '@nestjs/event-emitter';
import { sanitizeEventPayload } from '@utils';
import { EmailTemplateService } from './email-template.service';
import { OutboxEmailService } from './outbox-email.service';
import { IEmailDispatcher, EMAIL_DISPATCHER, EmailJob } from './email-dispatcher.interface';
import { MAIL_EVENT_MAP, MailEventName } from './mail-event.registry';
import { Inject } from '@nestjs/common';

/**
 * Central mail service — orchestrates template validation, outbox persistence,
 * and dispatch. EJS rendering is delegated to the dispatcher (SmtpEmailDispatcher
 * or BullMqEmailDispatcher), keeping the HTTP server lightweight.
 *
 * Public API:
 *   - sendByTemplate(templateName, to, data)   — validate → outbox → dispatch raw
 *   - sendCustomEmail(to, subject, body)        — outbox → dispatch pre-rendered
 *
 * Event handlers:
 *   - Dynamically registered via EventEmitter2.on() for each MAIL_EVENT_MAP entry.
 *   - Fire-and-forget: errors are logged but never re-thrown.
 */
@Injectable()
export class MailService implements OnModuleInit {
  private readonly logger = new Logger(MailService.name);

  constructor(
    private readonly templateService: EmailTemplateService,
    private readonly outboxService: OutboxEmailService,
    @Inject(EMAIL_DISPATCHER)
    private readonly dispatcher: IEmailDispatcher,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  /**
   * Register dynamic @OnEvent handlers for each event in the registry.
   * Uses EventEmitter2.on() instead of 8 separate @OnEvent decorators.
   */
  onModuleInit() {
    for (const eventName of Object.keys(MAIL_EVENT_MAP)) {
      this.eventEmitter.on(eventName, async (event: any) => {
        await this.handleMailEvent(event);
      });
    }
    this.logger.log(`Registered ${Object.keys(MAIL_EVENT_MAP).length} mail event handlers`);
  }

  // ──────────────────────────────────────────────────────────────────
  // Public API
  // ──────────────────────────────────────────────────────────────────

  /**
   * Send an email using a DB-stored template.
   *
   * Flow: validate template → outbox(PENDING) → dispatch raw EmailJob.
   * EJS rendering happens in the dispatcher (SmtpEmailDispatcher or Worker).
   *
   * @param templateName - The unique template name (e.g. 'UM001').
   * @param to            - Recipient email address.
   * @param data          - Context data for EJS rendering (deferred to dispatcher).
   */
  async sendByTemplate(
    templateName: string,
    to: string,
    data: Record<string, unknown> = {},
  ): Promise<void> {
    // 1. Validate template exists and is active (cache hit, no CPU work)
    const template = await this.templateService.findByTemplateName(templateName);
    if (!template) {
      this.logger.error(`Template not found: "${templateName}"`);
      return;
    }

    if (!template.isActive) {
      this.logger.warn(`Template "${templateName}" is inactive, skipping`);
      return;
    }

    // 2. Persist to outbox (no render yet — dispatcher will render)
    const outboxRecord = await this.outboxService.createPending({
      toAddress: to,
      subject: '', // Will be filled by dispatcher/worker on render
      body: '', // Will be filled by dispatcher/worker on render
      templateName,
    } as any);

    // 3. Dispatch raw job — dispatcher handles rendering
    const job: EmailJob = {
      type: 'template',
      outboxId: outboxRecord.id,
      templateName,
      to,
      data,
    };

    const result = await this.dispatcher.dispatch(job);

    if (result.success) {
      // Persist the subject/body rendered inside the dispatcher back to the
      // outbox row (it was created empty for the deferred-render flow).
      await this.outboxService.markSent(outboxRecord.id, result.persist as any);
    } else {
      await this.outboxService.markFailed(
        outboxRecord.id,
        result.error ?? 'Unknown dispatch error',
        result.permanent ?? false,
        result.persist as any,
      );
    }
  }

  /**
   * Send a raw email without a template.
   * Still goes through outbox for durability.
   */
  async sendCustomEmail(to: string, subject: string, body: string): Promise<void> {
    const outboxRecord = await this.outboxService.createPending({
      toAddress: to,
      subject,
      body,
    } as any);

    const result = await this.dispatcher.dispatch({
      type: 'custom',
      to,
      subject,
      html: body,
      outboxId: outboxRecord.id,
    });

    if (result.success) {
      await this.outboxService.markSent(outboxRecord.id);
    } else {
      await this.outboxService.markFailed(
        outboxRecord.id,
        result.error ?? 'Unknown dispatch error',
        result.permanent ?? false,
      );
    }
  }

  // ──────────────────────────────────────────────────────────────────
  // Private
  // ──────────────────────────────────────────────────────────────────

  /**
   * Common handler for all mail events.
   *
   * - Looks up template name from MAIL_EVENT_MAP (single source of truth).
   * - Delegates to sendByTemplate.
   * - Fire-and-forget: errors are logged, never re-thrown.
   */
  private async handleMailEvent(event: any): Promise<void> {
    const eventName: string =
      event?.constructor?.eventName ?? event?.constructor?.name ?? 'unknown';

    try {
      const templateName = MAIL_EVENT_MAP[eventName as MailEventName];
      if (!templateName) {
        // Event not mapped — skip silently (not an error)
        return;
      }

      await this.sendByTemplate(templateName, event.sendTo, event);
    } catch (error) {
      const sanitized = sanitizeEventPayload(event);
      this.logger.error(
        `Error handling mail event "${eventName}". Payload: ${JSON.stringify(sanitized)}`,
        (error as Error).stack,
      );
      // Fire-and-forget: never re-throw to main business flow.
    }
  }
}
