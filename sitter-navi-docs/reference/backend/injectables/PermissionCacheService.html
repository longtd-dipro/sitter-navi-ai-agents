import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, In } from 'typeorm';

import { RedisCacheService, CacheTtlSeconds } from '@infrastructure/cache';
import { RolePermission } from '../entities/role-permission.entity';
import { UserPermission, UserPermissionType } from '../entities/user-permission.entity';
import { UserPermissionGroup } from '../entities/user-permission-group.entity';

const CACHE_PREFIX = 'auth_role_permission_';
const CACHE_TTL = CacheTtlSeconds.ONE_DAY;

@Injectable()
export class PermissionCacheService implements OnModuleInit {
  private readonly logger = new Logger(PermissionCacheService.name);

  constructor(
    private readonly cacheService: RedisCacheService,
    @InjectRepository(RolePermission)
    private readonly rolePermRepo: Repository<RolePermission>,
    @InjectRepository(UserPermission)
    private readonly userPermRepo: Repository<UserPermission>,
    @InjectRepository(UserPermissionGroup)
    private readonly userGroupRepo: Repository<UserPermissionGroup>,
  ) {}

  onModuleInit() {
    this.logger.log('PermissionCacheService initialized');
  }

  /**
   * Get effective permissions for a user. Checks cache first, builds if miss.
   */
  async getUserPermissions(userId: string, role: string): Promise<Set<string>> {
    const cacheKey = `${CACHE_PREFIX}${userId}`;

    try {
      const cached = await this.cacheService.getValue(cacheKey);
      if (cached) {
        return new Set(JSON.parse(cached));
      }
    } catch {
      // Redis down — compute from DB directly
    }

    // Cache miss → build from DB
    const permissions = await this.buildEffectivePermissions(userId, role);

    // Store in cache
    try {
      await this.cacheService.setValue(cacheKey, JSON.stringify([...permissions]));
    } catch {
      // Redis down — still return computed permissions
    }

    return permissions;
  }

  /**
   * Invalidate cache for a specific user. Call after permission changes.
   */
  async invalidateUser(userId: string): Promise<void> {
    const cacheKey = `${CACHE_PREFIX}${userId}`;
    try {
      await this.cacheService.clearCacheByKey(cacheKey);
    } catch {
      this.logger.warn(`Failed to invalidate permission cache for user ${userId}`);
    }
  }

  /**
   * Invalidate cache for all users with a specific role.
   * Call after role permission changes.
   */
  async invalidateRole(role: string): Promise<void> {
    // We need to clear all users with this role — use pattern
    try {
      await this.cacheService.clearCacheByPattern(`${CACHE_PREFIX}*`);
    } catch {
      this.logger.warn(`Failed to invalidate permission cache for role ${role}`);
    }
  }

  /**
   * Build effective permissions from DB:
   * role_permissions + user_groups.permissions + user_permissions.granted - user_permissions.revoked
   */
  private async buildEffectivePermissions(userId: string, role: string): Promise<Set<string>> {
    const permissions = new Set<string>();

    // 1. Role permissions
    const rolePerms = await this.rolePermRepo.find({
      where: { role },
      relations: ['permission'],
    });
    for (const rp of rolePerms) {
      if (rp.permission) {
        permissions.add(rp.permission.key);
      }
    }

    // 2. User permission groups → expand to individual permissions
    const userGroups = await this.userGroupRepo.find({
      where: { userId },
      relations: ['group', 'group.items', 'group.items.permission'],
    });
    for (const ug of userGroups) {
      if (ug.group?.items) {
        for (const item of ug.group.items) {
          if (item.permission) {
            permissions.add(item.permission.key);
          }
        }
      }
    }

    // 3. User individual permissions (granted/revoked)
    const userPerms = await this.userPermRepo.find({
      where: { userId },
      relations: ['permission'],
    });
    for (const up of userPerms) {
      if (!up.permission) continue;
      if (up.type === UserPermissionType.GRANTED) {
        permissions.add(up.permission.key);
      } else if (up.type === UserPermissionType.REVOKED) {
        permissions.delete(up.permission.key);
      }
    }

    return permissions;
  }
}
