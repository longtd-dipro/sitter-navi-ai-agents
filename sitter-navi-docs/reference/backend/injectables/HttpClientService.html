import { Injectable, Optional, Inject } from '@nestjs/common';
import { Readable } from 'stream';
import { LoggingService } from '../logging/logging.service';
import { HttpRequestError, HttpTimeoutError } from './http-client.errors';
import {
  HttpClientConfig,
  HttpResponse,
  HttpStreamResponse,
  RequestInterceptor,
  RequestOptions,
  ResponseInterceptor,
} from './http-client.interfaces';

export const HTTP_CLIENT_CONFIG = Symbol('HTTP_CLIENT_CONFIG');

const DEFAULT_RETRY_STATUS = [408, 429, 500, 502, 503, 504];

const CONTENT_TYPE_JSON = 'application/json';
const CONTENT_TYPE_FORM = 'application/x-www-form-urlencoded';

@Injectable()
export class HttpClientService {
  private readonly config: Required<
    Omit<HttpClientConfig, 'retry'> & { retry: Required<NonNullable<HttpClientConfig['retry']>> }
  >;
  private readonly requestInterceptors: RequestInterceptor[] = [];
  private readonly responseInterceptors: ResponseInterceptor[] = [];
  private readonly logger: LoggingService;

  constructor(
    @Optional() @Inject(HTTP_CLIENT_CONFIG) config?: HttpClientConfig,
    @Optional() loggingService?: LoggingService,
  ) {
    this.config = {
      baseURL: config?.baseURL ?? '',
      timeout: config?.timeout ?? 30_000,
      headers: config?.headers ?? { 'Content-Type': CONTENT_TYPE_JSON },
      throwOnError: config?.throwOnError ?? true,
      retry: {
        attempts: config?.retry?.attempts ?? 0,
        delayMs: config?.retry?.delayMs ?? 1000,
        exponential: config?.retry?.exponential ?? true,
        retryOnStatus: config?.retry?.retryOnStatus ?? DEFAULT_RETRY_STATUS,
      },
    };
    this.logger = loggingService
      ? loggingService.child('HttpClient')
      : new Proxy({} as LoggingService, {
          get: () => () => undefined,
        });
  }

  // ---------------------------------------------------------------------------
  // Interceptors
  // ---------------------------------------------------------------------------

  addRequestInterceptor(interceptor: RequestInterceptor): void {
    this.requestInterceptors.push(interceptor);
  }

  addResponseInterceptor(interceptor: ResponseInterceptor): void {
    this.responseInterceptors.push(interceptor);
  }

  // ---------------------------------------------------------------------------
  // Public HTTP methods (Axios-like API)
  // ---------------------------------------------------------------------------

  async get<T = unknown>(
    url: string,
    options?: RequestOptions,
  ): Promise<HttpResponse<T>> {
    return this.request<T>('GET', url, undefined, options);
  }

  async post<T = unknown>(
    url: string,
    body?: unknown,
    options?: RequestOptions,
  ): Promise<HttpResponse<T>> {
    return this.request<T>('POST', url, body, options);
  }

  async put<T = unknown>(
    url: string,
    body?: unknown,
    options?: RequestOptions,
  ): Promise<HttpResponse<T>> {
    return this.request<T>('PUT', url, body, options);
  }

  async patch<T = unknown>(
    url: string,
    body?: unknown,
    options?: RequestOptions,
  ): Promise<HttpResponse<T>> {
    return this.request<T>('PATCH', url, body, options);
  }

  async delete<T = unknown>(
    url: string,
    options?: RequestOptions,
  ): Promise<HttpResponse<T>> {
    return this.request<T>('DELETE', url, undefined, options);
  }

  // ---------------------------------------------------------------------------
  // Stream methods — returns raw stream for piping/forwarding
  // ---------------------------------------------------------------------------

  /**
   * GET request that returns a readable stream.
   * Useful for proxying file downloads, SSE, or large responses.
   *
   * @example
   * // Pipe to Express response (file download proxy)
   * const { stream, headers } = await this.http.getStream('/files/report.pdf');
   * res.set('Content-Type', headers['content-type']);
   * Readable.fromWeb(stream).pipe(res);
   *
   * @example
   * // SSE forwarding
   * const { stream } = await this.http.getStream('/events', {
   *   headers: { Accept: 'text/event-stream' },
   *   timeout: 0, // no timeout for SSE
   * });
   */
  async getStream(
    url: string,
    options?: RequestOptions,
  ): Promise<HttpStreamResponse> {
    return this.requestStream('GET', url, undefined, options);
  }

  /**
   * POST request that returns a readable stream.
   * Useful for AI/LLM streaming responses, chunked uploads, etc.
   */
  async postStream(
    url: string,
    body?: unknown,
    options?: RequestOptions,
  ): Promise<HttpStreamResponse> {
    return this.requestStream('POST', url, body, options);
  }

  /**
   * Convert a Web ReadableStream to a Node.js Readable stream.
   * Convenience helper for piping to Express/NestJS responses.
   *
   * @example
   * const { stream } = await this.http.getStream('/file.zip');
   * const nodeStream = this.http.toNodeStream(stream);
   * nodeStream.pipe(res);
   */
  toNodeStream(webStream: ReadableStream<Uint8Array>): Readable {
    return Readable.fromWeb(webStream as any);
  }

  // ---------------------------------------------------------------------------
  // Core request method
  // ---------------------------------------------------------------------------

  private async request<T>(
    method: string,
    url: string,
    body?: unknown,
    options?: RequestOptions,
  ): Promise<HttpResponse<T>> {
    const fullUrl = this.buildUrl(url, options?.params);
    const timeoutMs = options?.timeout ?? this.config.timeout;
    const shouldThrow = options?.throwOnError ?? this.config.throwOnError;

    const maxAttempts = this.config.retry.attempts + 1; // 1 initial + retries
    let lastError: Error | undefined;

    for (let attempt = 1; attempt <= maxAttempts; attempt++) {
      try {
        const result = await this.executeRequest<T>(
          method,
          fullUrl,
          body,
          options,
          timeoutMs,
          shouldThrow,
        );
        return result;
      } catch (error) {
        lastError = error as Error;

        const shouldRetry =
          attempt < maxAttempts && this.isRetryable(error as Error);

        if (shouldRetry) {
          const delay = this.config.retry.exponential
            ? this.config.retry.delayMs * Math.pow(2, attempt - 1)
            : this.config.retry.delayMs;

          this.logger.warn(`Request failed, retrying (${attempt}/${this.config.retry.attempts})`, {
            method,
            url: fullUrl,
            attempt,
            nextRetryMs: delay,
          });

          await this.sleep(delay);
        }
      }
    }

    throw lastError;
  }

  private async executeRequest<T>(
    method: string,
    fullUrl: string,
    body: unknown,
    options: RequestOptions | undefined,
    timeoutMs: number,
    shouldThrow: boolean,
  ): Promise<HttpResponse<T>> {
    // Build RequestInit
    let init: RequestInit = {
      method,
      headers: this.buildHeaders(body, options?.headers),
      body: this.serializeBody(body).body,
    };

    // Apply request interceptors
    for (const interceptor of this.requestInterceptors) {
      init = await interceptor(fullUrl, init);
    }

    // Timeout via AbortController
    const controller = new AbortController();
    const externalSignal = options?.signal;
    let timeoutId: ReturnType<typeof setTimeout> | undefined;

    if (timeoutMs > 0) {
      timeoutId = setTimeout(() => controller.abort(), timeoutMs);
    }

    // Link external signal
    if (externalSignal) {
      externalSignal.addEventListener('abort', () => controller.abort());
    }

    init.signal = controller.signal;

    const startTime = Date.now();

    try {
      const response = await fetch(fullUrl, init);
      const elapsed = Date.now() - startTime;

      // Parse response headers
      const responseHeaders: Record<string, string> = {};
      response.headers.forEach((value, key) => {
        responseHeaders[key] = value;
      });

      // Parse body
      const contentType = response.headers.get('content-type') ?? '';
      let data: T;
      if (contentType.includes('application/json')) {
        data = (await response.json()) as T;
      } else {
        data = (await response.text()) as unknown as T;
      }

      this.logger.debug(`${method} ${fullUrl} → ${response.status} (${elapsed}ms)`, {
        method,
        url: fullUrl,
        status: response.status,
        elapsed,
      });

      // Throw on error status (Axios behavior)
      if (shouldThrow && !response.ok) {
        throw new HttpRequestError(
          `Request failed with status ${response.status}: ${response.statusText}`,
          response.status,
          response.statusText,
          fullUrl,
          method,
          data,
          responseHeaders,
        );
      }

      let result: HttpResponse<T> = {
        data,
        status: response.status,
        statusText: response.statusText,
        headers: responseHeaders,
        ok: response.ok,
      };

      // Apply response interceptors
      for (const interceptor of this.responseInterceptors) {
        result = (await interceptor(result)) as HttpResponse<T>;
      }

      return result;
    } catch (error) {
      if (error instanceof HttpRequestError) {
        throw error;
      }

      if ((error as Error).name === 'AbortError') {
        throw new HttpTimeoutError(fullUrl, method, timeoutMs);
      }

      throw error;
    } finally {
      if (timeoutId) clearTimeout(timeoutId);
    }
  }

  // ---------------------------------------------------------------------------
  // Stream request (no body parsing, returns raw stream)
  // ---------------------------------------------------------------------------

  private async requestStream(
    method: string,
    url: string,
    body?: unknown,
    options?: RequestOptions,
  ): Promise<HttpStreamResponse> {
    const fullUrl = this.buildUrl(url, options?.params);
    const timeoutMs = options?.timeout ?? this.config.timeout;
    const shouldThrow = options?.throwOnError ?? this.config.throwOnError;

    // Build RequestInit
    let init: RequestInit = {
      method,
      headers: this.buildHeaders(body, options?.headers),
      body: this.serializeBody(body).body,
    };

    // Apply request interceptors
    for (const interceptor of this.requestInterceptors) {
      init = await interceptor(fullUrl, init);
    }

    // Timeout via AbortController
    const controller = new AbortController();
    const externalSignal = options?.signal;
    let timeoutId: ReturnType<typeof setTimeout> | undefined;

    if (timeoutMs > 0) {
      timeoutId = setTimeout(() => controller.abort(), timeoutMs);
    }

    if (externalSignal) {
      externalSignal.addEventListener('abort', () => controller.abort());
    }

    init.signal = controller.signal;

    try {
      const response = await fetch(fullUrl, init);

      // Clear timeout once we get headers — stream may take longer
      if (timeoutId) {
        clearTimeout(timeoutId);
        timeoutId = undefined;
      }

      // Parse response headers
      const responseHeaders: Record<string, string> = {};
      response.headers.forEach((value, key) => {
        responseHeaders[key] = value;
      });

      // Throw on error status
      if (shouldThrow && !response.ok) {
        // For error responses, read body for error details
        const errorBody = await response.text().catch(() => '');
        throw new HttpRequestError(
          `Stream request failed with status ${response.status}: ${response.statusText}`,
          response.status,
          response.statusText,
          fullUrl,
          method,
          errorBody,
          responseHeaders,
        );
      }

      if (!response.body) {
        throw new Error(`No response body for stream request: ${method} ${fullUrl}`);
      }

      this.logger.debug(`${method} ${fullUrl} → ${response.status} (stream)`, {
        method,
        url: fullUrl,
        status: response.status,
        stream: true,
      });

      return {
        stream: response.body,
        status: response.status,
        statusText: response.statusText,
        headers: responseHeaders,
        ok: response.ok,
      };
    } catch (error) {
      if (error instanceof HttpRequestError) {
        throw error;
      }

      if ((error as Error).name === 'AbortError') {
        throw new HttpTimeoutError(fullUrl, method, timeoutMs);
      }

      throw error;
    } finally {
      if (timeoutId) clearTimeout(timeoutId);
    }
  }

  // ---------------------------------------------------------------------------
  // Helpers
  // ---------------------------------------------------------------------------

  /**
   * Serialize a request body into a fetch-compatible `BodyInit`.
   *
   * Plain objects are JSON-encoded (default). Already-encoded bodies —
   * `URLSearchParams` (form-urlencoded), strings, Buffers/typed arrays, Blobs
   * and streams — are passed through untouched so callers can send
   * form-urlencoded or binary payloads. The returned `contentType` is applied
   * by {@link buildHeaders} unless the caller sets its own `Content-Type`.
   */
  private serializeBody(body: unknown): { body: BodyInit | undefined; contentType?: string } {
    if (body === undefined || body === null) {
      return { body: undefined };
    }
    if (body instanceof URLSearchParams) {
      return { body, contentType: CONTENT_TYPE_FORM };
    }
    if (
      typeof body === 'string' ||
      body instanceof Buffer ||
      body instanceof ArrayBuffer ||
      ArrayBuffer.isView(body) ||
      body instanceof Blob ||
      body instanceof ReadableStream
    ) {
      return { body: body as BodyInit };
    }
    return { body: JSON.stringify(body), contentType: CONTENT_TYPE_JSON };
  }

  /**
   * Merge default, body-derived and per-request headers.
   * Precedence (low → high): config defaults → body content-type → caller headers.
   */
  private buildHeaders(
    body: unknown,
    overrides?: Record<string, string>,
  ): Record<string, string> {
    const { contentType } = this.serializeBody(body);
    return {
      ...this.config.headers,
      ...(contentType ? { 'Content-Type': contentType } : {}),
      ...overrides,
    };
  }

  private buildUrl(
    url: string,
    params?: Record<string, string | number | boolean | undefined>,
  ): string {
    const base = url.startsWith('http') ? url : `${this.config.baseURL}${url}`;

    if (!params) return base;

    const searchParams = new URLSearchParams();
    for (const [key, value] of Object.entries(params)) {
      if (value !== undefined) {
        searchParams.append(key, String(value));
      }
    }

    const separator = base.includes('?') ? '&' : '?';
    return `${base}${separator}${searchParams.toString()}`;
  }

  private isRetryable(error: Error): boolean {
    if (error instanceof HttpTimeoutError) return true;
    if (error instanceof HttpRequestError) {
      return this.config.retry.retryOnStatus.includes(error.status);
    }
    // Network errors are retryable
    return error.name === 'TypeError' || error.message.includes('fetch');
  }

  private sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}
