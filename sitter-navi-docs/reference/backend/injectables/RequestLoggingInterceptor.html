import {
  CallHandler,
  ExecutionContext,
  Injectable,
  NestInterceptor,
} from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { Observable, throwError } from 'rxjs';
import { catchError, tap } from 'rxjs/operators';
import { LoggingService } from '../logging/logging.service';
import { LoggingConfigType } from '../logging/logging.config';

@Injectable()
export class RequestLoggingInterceptor implements NestInterceptor {
  private readonly ignorePaths: string[];

  constructor(
    private readonly logger: LoggingService,
    private readonly configService: ConfigService,
  ) {
    this.logger.setContext('HTTP');
    const config = this.configService.get<LoggingConfigType>('logging');
    this.ignorePaths = config?.httpIgnorePaths ?? ['/api/health', '/swagger'];
  }

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const req = context.switchToHttp().getRequest();
    const url: string = req.url as string;

    // Check if URL starts with any ignored path (case-insensitive prefix match)
    const urlLower = url.toLowerCase();
    const shouldIgnore = this.ignorePaths.some((ignorePath) =>
      urlLower.startsWith(ignorePath.toLowerCase()),
    );

    if (shouldIgnore) {
      return next.handle();
    }

    const method: string = req.method as string;
    const correlationId =
      LoggingService.getAsyncLocalStorage().getStore()?.correlationId;
    const userId = (req.user as { id?: unknown } | undefined)?.id;
    const userAgent = req.headers['user-agent'] as string | undefined;
    const ip: string =
      (req.ip as string | undefined) ??
      (req.socket?.remoteAddress as string | undefined) ??
      '';

    // Log incoming request
    this.logger.http('Incoming request', {
      method,
      url,
      ...(correlationId !== undefined ? { correlationId } : {}),
      ...(userId !== undefined ? { userId } : {}),
      ...(userAgent !== undefined ? { userAgent } : {}),
      ip,
    });

    const startTime = Date.now();

    return next.handle().pipe(
      tap((/* _data */) => {
        const res = context.switchToHttp().getResponse();
        const statusCode: number = res.statusCode as number;
        const durationMs = Math.round(Date.now() - startTime);
        const correlationIdAfter =
          LoggingService.getAsyncLocalStorage().getStore()?.correlationId;

        this.logger.http('Response sent', {
          method,
          url,
          statusCode,
          ...(correlationIdAfter !== undefined
            ? { correlationId: correlationIdAfter }
            : {}),
          durationMs,
        });
      }),
      catchError((err: unknown) => {
        const durationMs = Math.round(Date.now() - startTime);
        const correlationIdAfter =
          LoggingService.getAsyncLocalStorage().getStore()?.correlationId;

        // Determine status code from the error (HttpException exposes getStatus())
        const statusCode: number =
          typeof (err as { getStatus?: () => number }).getStatus === 'function'
            ? (err as { getStatus: () => number }).getStatus()
            : 500;

        this.logger.http('Response sent (error)', {
          method,
          url,
          statusCode,
          ...(correlationIdAfter !== undefined
            ? { correlationId: correlationIdAfter }
            : {}),
          durationMs,
        });

        return throwError(() => err);
      }),
    );
  }
}
