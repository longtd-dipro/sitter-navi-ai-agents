import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import * as ejs from 'ejs';
import { IMessageRenderer } from '@messaging';
import { EmailTemplate } from './email-template.entity';

export interface RenderedEmail {
  subject: string;
  html: string;
}

/**
 * Pure service: renders EmailTemplate + data → subject + HTML.
 *
 * No DB/Redis access, no side effects.
 * Uses EJS directly — no dependency on the "email-templates" npm package.
 */
@Injectable()
export class EmailRendererService implements IMessageRenderer<
  Record<string, unknown>,
  RenderedEmail
> {
  private readonly logger = new Logger(EmailRendererService.name);

  constructor(private readonly configService: ConfigService) {}

  /**
   * Common info injected into every email template.
   */
  private get commonInfo(): Record<string, unknown> {
    return {
      supportEmail: this.configService.get<string>('email.supportEmail') ?? '',
      companyName: this.configService.get<string>('email.companyName') ?? '',
      userDomain: this.configService.get<string>('app.userDomain') ?? '',
    };
  }

  /**
   * Render an email template with the provided data.
   *
   * @param template - The EmailTemplate entity from DB.
   * @param data     - Context data for the EJS template.
   * @returns Rendered subject and HTML body.
   */
  async render(template: unknown, data: Record<string, unknown>): Promise<RenderedEmail> {
    const tpl = template as EmailTemplate;

    const mergedData = {
      ...this.commonInfo,
      ...data,
    };

    try {
      const subject = ejs.render(tpl.subject, { data: mergedData });
      const html = ejs.render(tpl.body, { data: mergedData });

      return { subject, html };
    } catch (error) {
      this.logger.error(`Failed to render template "${tpl.name}"`, (error as Error).stack);
      throw error;
    }
  }
}
