Description

Repository_Layer for the refresh-token aggregate inside the Auth module.

Responsibilities:

  • Encapsulate every read/write against the refresh_token table.
  • Inherit the standard BaseRepository contract (findById, findByIdOrFail, save, softDelete, count).
  • Expose the lookup/revoke commands the import('./service/refresh-token.service').RefreshTokenService needs for the issue / rotate / logout flows.

Constraints (mirrors import('./auth.repository').AuthRepository):

  • MUST inject Repository<RefreshTokenEntity> via @InjectRepository.
  • MUST NOT inject services from other modules — cross-module access (looking up the User) is handled at the service layer.
  • MUST NOT enforce business validation (expiry / reuse decisions stay in the service layer). "Active" here means only "not yet revoked".

import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { IsNull, Repository } from 'typeorm';

import { BaseRepository } from '@common/repositories';
import { RefreshTokenEntity } from './refresh-token.entity';

/**
 * Repository_Layer for the refresh-token aggregate inside the Auth module.
 *
 * Responsibilities:
 *  - Encapsulate every read/write against the `refresh_token` table.
 *  - Inherit the standard {@link BaseRepository} contract
 *    (`findById`, `findByIdOrFail`, `save`, `softDelete`, `count`).
 *  - Expose the lookup/revoke commands the {@link import('./service/refresh-token.service').RefreshTokenService}
 *    needs for the issue / rotate / logout flows.
 *
 * Constraints (mirrors {@link import('./auth.repository').AuthRepository}):
 *  - MUST inject `Repository<RefreshTokenEntity>` via `@InjectRepository`.
 *  - MUST NOT inject services from other modules — cross-module access
 *    (looking up the User) is handled at the service layer.
 *  - MUST NOT enforce business validation (expiry / reuse decisions stay
 *    in the service layer). "Active" here means only "not yet revoked".
 */
@Injectable()
export class RefreshTokenRepository extends BaseRepository<RefreshTokenEntity> {
  protected readonly entityName = 'refresh_token';

  constructor(
    @InjectRepository(RefreshTokenEntity)
    protected readonly repo: Repository<RefreshTokenEntity>,
  ) {
    super();
  }

  /**
   * Look up a not-yet-revoked token row by its SHA-256 hash. Returns
   * `null` when no row matches or the matching row has been revoked.
   *
   * Note: expiry is intentionally NOT filtered here — deciding what to do
   * with an expired-but-not-revoked row (reject + revoke) is a service
   * concern.
   */
  async findActiveByTokenHash(tokenHash: string): Promise<RefreshTokenEntity | null> {
    return this.repo.findOne({ where: { tokenHash, revokedAt: IsNull() } });
  }

  /** Mark a single token row revoked. Idempotent for an already-revoked row. */
  async revokeById(id: string, revokedAt: Date): Promise<void> {
    await this.repo.update(id, { revokedAt });
  }

  /**
   * Revoke every still-active token for a user (logout-everywhere). Only
   * rows with `revokedAt IS NULL` are touched so prior revocation
   * timestamps are preserved.
   */
  async revokeAllByUserId(userId: string, revokedAt: Date): Promise<void> {
    await this.repo.update({ userId, revokedAt: IsNull() }, { revokedAt });
  }
}
