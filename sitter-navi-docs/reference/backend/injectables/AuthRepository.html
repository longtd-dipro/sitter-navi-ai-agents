Description

Repository_Layer for the OTP aggregate inside the Auth module.

Responsibilities:

  • Encapsulate every read/write against the otp table.
  • Inherit the standard BaseRepository contract (findById, findByIdOrFail, save, softDelete, count).
  • Expose OTP-specific query/command methods that the existing AuthService performs directly on Repository<OTPEntity> today.

Constraints (per design / Requirements 3.1, 3.2, 12.1, 12.3, 12.4):

  • MUST inject Repository<OTPEntity> via @InjectRepository.
  • MUST NOT inject services from other modules — cross-module access (e.g. looking up the User) is handled in the Service layer.
  • MUST NOT inject EventEmitter2 — emitting domain events is the Service layer's responsibility.
  • MUST NOT enforce business validation (token expiry, retry caps, is-locked checks all stay in the Service layer).

The entityName ("otp") is used by findByIdOrFail to throw a import('@nestjs/common').NotFoundException with the i18n key 'otp.not_found'.

import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';

import { BaseRepository } from '@common/repositories';
import { OTPEntity, OtpPurpose } from './otp.entity';
import { generateSnowflakeId } from '@utils/snowflake';

/**
 * Shape accepted by {@link AuthRepository.upsertOtp}. Mirrors the columns
 * the AuthService currently writes through TypeORM's `upsert(...)` call:
 * the row is inserted on first use and replaced (by `userId` conflict
 * path) on subsequent OTP requests for the same user.
 *
 * The repository accepts the data as a plain object — generating the
 * snowflake id, expiration and other defaults remains the Service
 * layer's responsibility.
 */
export interface UpsertOtpInput {
  /** Owning user id. The unique constraint on this column drives the upsert. */
  userId: string;
  /** Six-digit numeric OTP shown to the user. */
  otp: string;
  /** Wall-clock expiration timestamp. */
  expiredAt: Date;
  /** Opaque identifier returned to the client to correlate verify calls. */
  identifier: string;
  /** Purpose of the OTP (login vs password reset). */
  purpose: OtpPurpose;
  /** Whether the OTP has already been consumed (defaults to `false`). */
  isUsed?: boolean;
  /** Pass `null` to revive a soft-deleted row when reissuing an OTP. */
  deletedAt?: Date | null;
  /** Free-form context persisted alongside the OTP (e.g. `{ email }`). */
  // eslint-disable-next-line @typescript-eslint/no-explicit-any -- mirrors OTPEntity.meta (jsonb)
  meta?: Record<string, any>;
}

/**
 * Repository_Layer for the OTP aggregate inside the Auth module.
 *
 * Responsibilities:
 *  - Encapsulate every read/write against the `otp` table.
 *  - Inherit the standard {@link BaseRepository} contract
 *    (`findById`, `findByIdOrFail`, `save`, `softDelete`, `count`).
 *  - Expose OTP-specific query/command methods that the existing
 *    `AuthService` performs directly on `Repository<OTPEntity>` today.
 *
 * Constraints (per design / Requirements 3.1, 3.2, 12.1, 12.3, 12.4):
 *  - MUST inject `Repository<OTPEntity>` via `@InjectRepository`.
 *  - MUST NOT inject services from other modules — cross-module access
 *    (e.g. looking up the User) is handled in the Service layer.
 *  - MUST NOT inject `EventEmitter2` — emitting domain events is the
 *    Service layer's responsibility.
 *  - MUST NOT enforce business validation (token expiry, retry caps,
 *    is-locked checks all stay in the Service layer).
 *
 * The `entityName` ("otp") is used by `findByIdOrFail` to throw a
 * {@link import('@nestjs/common').NotFoundException} with the i18n key
 * `'otp.not_found'`.
 */
@Injectable()
export class AuthRepository extends BaseRepository<OTPEntity> {
  protected readonly entityName = 'otp';

  constructor(
    @InjectRepository(OTPEntity)
    protected readonly repo: Repository<OTPEntity>,
  ) {
    super();
  }

  /**
   * Look up an OTP row by its public identifier. When `purpose` is
   * provided the lookup is constrained to that purpose — this matches
   * the `findOne({ where: { identifier, purpose } })` calls in
   * `AuthService.verifyLogin` and `AuthService.adminResetPassword`.
   *
   * Returns `null` when no row matches.
   */
  async findByIdentifier(identifier: string, purpose?: OtpPurpose): Promise<OTPEntity | null> {
    if (purpose) {
      return this.repo.findOne({ where: { identifier, purpose } });
    }
    return this.repo.findOne({ where: { identifier } });
  }

  /**
   * Look up an OTP row by identifier, including soft-deleted rows.
   * Replicates the `findOne({ where: { identifier }, withDeleted: true })`
   * behaviour used by `AuthService.resendOtp` to find the previous OTP
   * even after a soft delete.
   */
  async findByIdentifierWithDeleted(identifier: string): Promise<OTPEntity | null> {
    return this.repo.findOne({ where: { identifier }, withDeleted: true });
  }

  /**
   * Find the OTP row currently associated with a user. The `userId`
   * column is unique, so at most one row is returned. Soft-deleted
   * rows are excluded by default.
   *
   * Note: this method intentionally does not filter on `isUsed` or
   * `expiredAt` — verifying that the OTP is still valid (expiry,
   * retry counter, isUsed flag) is the Service layer's job.
   */
  async findActiveByUserId(userId: string): Promise<OTPEntity | null> {
    return this.repo.findOne({ where: { userId } });
  }

  /**
   * Insert-or-replace the OTP row for a user. The unique constraint on
   * `userId` drives the conflict resolution — if the user already has
   * an OTP row it is overwritten (same behaviour as the existing
   * `AuthService.generateOTP` flow).
   */
  async upsertOtp(data: UpsertOtpInput): Promise<void> {
    await this.repo.upsert(
      {
        id: generateSnowflakeId(),
        userId: data.userId,
        otp: data.otp,
        expiredAt: data.expiredAt,
        identifier: data.identifier,
        purpose: data.purpose,
        isUsed: data.isUsed ?? false,
        deletedAt: data.deletedAt ?? null,
        meta: data.meta ?? {},
      },
      {
        conflictPaths: ['userId'],
      },
    );
  }

  async insertOtp(data: Omit<UpsertOtpInput, 'userId'>, meta?: Record<string, any>): Promise<void> {
    await this.repo.insert({
      id: generateSnowflakeId(),
      otp: data.otp,
      expiredAt: data.expiredAt,
      identifier: data.identifier,
      purpose: data.purpose,
      isUsed: data.isUsed ?? false,
      deletedAt: data.deletedAt ?? null,
      meta: meta ?? {},
    });
  }

  /**
   * Increment the retry counter on a specific OTP row. Callers compute
   * the next retry value (`current + 1`) so the operation is a direct
   * UPDATE — the repository is not concerned with retry caps.
   */
  async incrementRetry(id: string, retry: number): Promise<void> {
    await this.repo.update(id, { retry });
  }

  /**
   * Mark an OTP as consumed and extend its window so the caller can
   * complete the follow-up step (typically the password-reset flow's
   * "set new password" call). The repository performs the UPDATE only —
   * deciding when an OTP becomes "used" is a service concern.
   */
  async markUsed(id: string, expiredAt: Date): Promise<void> {
    await this.repo.update(id, { isUsed: true, expiredAt });
  }

  /**
   * Soft-delete an OTP row by its public identifier. Mirrors a
   * `softDelete({ identifier })` call so the Service layer can avoid
   * loading the row first when the identifier is the only known key.
   */
  async softDeleteByIdentifier(identifier: string): Promise<void> {
    await this.repo.softDelete({ identifier });
  }

  /**
   * Hard-delete an OTP row by id. Used by AuthService when an OTP is
   * fully consumed (successful verify) or invalidated (max retries
   * reached, account locked) — i.e. the row should not be recoverable.
   */
  async hardDelete(id: string): Promise<void> {
    await this.repo.delete(id);
  }
}
