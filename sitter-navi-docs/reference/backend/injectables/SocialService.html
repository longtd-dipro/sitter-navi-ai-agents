Description

SocialService — verifies third-party ID tokens (Apple, LINE) and resolves them to a stable SocialIdentity.

Apple tokens are RS256 JWTs verified locally against Apple's published JWKS using WebCrypto. LINE tokens are verified through LINE's /oauth2/v2.1/verify endpoint, which validates the signature, expiry and audience server-side.

import { Injectable, Logger, UnauthorizedException } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { HttpClientService } from '@infrastructure/http-client';
import { SocialProvider } from '@modules/user/social-account.entity';

const APPLE_ISSUER = 'https://appleid.apple.com';
const APPLE_JWKS_URL = 'https://appleid.apple.com/auth/keys';
const LINE_VERIFY_URL = 'https://api.line.me/oauth2/v2.1/verify';

/** i18n key surfaced to the client on any verification failure. */
const SOCIAL_LOGIN_FAILED = 'auth.social_login_failed';

/** Apple ID tokens are signed with RS256. */
const APPLE_SIGNING_ALG = 'RS256';

/** Cache JWKS for an hour so we are not refetching on every login. */
const JWKS_CACHE_TTL_MS = 60 * 60 * 1000;

const JWT_PARTS = 3;
const MS_PER_SECOND = 1000;

/** Normalized identity extracted from a verified social ID token. */
export interface SocialIdentity {
  /** Stable, provider-scoped user id (the JWT `sub`). */
  providerUserId: string;
  /** Email claim, when the provider exposes it. */
  email?: string;
}

interface JwtHeader {
  alg: string;
  kid?: string;
}

/** A JWK as published in a JWKS, carrying the key id we match on. */
type JwkWithKid = JsonWebKey & { kid?: string };

interface ApplePayload {
  iss: string;
  sub: string;
  aud: string | string[];
  exp: number;
  email?: string;
}

interface LineVerifyResponse {
  iss: string;
  sub: string;
  aud: string;
  exp: number;
  email?: string;
  error?: string;
  error_description?: string;
}

/**
 * SocialService — verifies third-party ID tokens (Apple, LINE) and
 * resolves them to a stable {@link SocialIdentity}.
 *
 * Apple tokens are RS256 JWTs verified locally against Apple's published
 * JWKS using WebCrypto. LINE tokens are verified through LINE's
 * `/oauth2/v2.1/verify` endpoint, which validates the signature, expiry
 * and audience server-side.
 */
@Injectable()
export class SocialService {
  private readonly logger = new Logger(SocialService.name);
  private appleJwksCache?: { keys: JwkWithKid[]; fetchedAt: number };

  constructor(
    private readonly config: ConfigService,
    private readonly http: HttpClientService,
  ) {}

  /** Verify an ID token for the given provider and return the identity. */
  async verifyIdToken(provider: SocialProvider, idToken: string): Promise<SocialIdentity> {
    switch (provider) {
      case SocialProvider.APPLE:
        return this.verifyApple(idToken);
      case SocialProvider.LINE:
        return this.verifyLine(idToken);
      // return this.getDumpSocialProfile(idToken);
      default:
        this.fail(`unsupported provider=${provider}`);
    }
  }

  async getDumpSocialProfile(jwtTokenDump: string) {
    const parts = jwtTokenDump?.split('.');
    if (parts.length !== JWT_PARTS) {
      this.fail(`malformed JWT: expected ${JWT_PARTS} parts, got ${parts.length}`);
    }
    const payload = JSON.parse(Buffer.from(parts[1], 'base64').toString('utf-8'));
    const providerUserId = payload.uid;
    return { providerUserId };
  }

  /**
   * Log the real reason (for ops/debugging) but never leak it to the client —
   * every failure surfaces the same generic {@link SOCIAL_LOGIN_FAILED} key.
   */
  private fail(reason: string): never {
    this.logger.warn(`Social token rejected: ${reason}`);
    throw new UnauthorizedException(SOCIAL_LOGIN_FAILED);
  }

  // --------------------------------------------------------------------------
  // Apple
  // --------------------------------------------------------------------------

  private async verifyApple(idToken: string): Promise<SocialIdentity> {
    const { header, payload, signedData, signature } = this.decodeJwt<ApplePayload>(idToken);

    if (header.alg !== APPLE_SIGNING_ALG || !header.kid) {
      this.fail(`unexpected Apple header alg=${header.alg} kid=${header.kid ?? 'none'}`);
    }

    const jwk = await this.getAppleJwk(header.kid);
    const isVerified = await this.verifyRs256(jwk, signedData, signature);
    if (!isVerified) {
      this.fail(`Apple signature verification failed kid=${header.kid}`);
    }

    this.assertAppleClaims(payload);

    return { providerUserId: payload.sub, email: payload.email };
  }

  /** Validate the issuer, audience and expiry claims of an Apple ID token. */
  private assertAppleClaims(payload: ApplePayload): void {
    if (payload.iss !== APPLE_ISSUER) {
      this.fail(`unexpected Apple issuer=${payload.iss}`);
    }

    const allowedAudiences = this.config.get<string[]>('social.apple.clientIds') ?? [];
    const audiences = Array.isArray(payload.aud) ? payload.aud : [payload.aud];
    const hasAudienceMatch = audiences.some((aud) => allowedAudiences.includes(aud));
    if (allowedAudiences.length > 0 && !hasAudienceMatch) {
      this.fail(`Apple audience mismatch aud=${audiences.join(',')}`);
    }

    if (this.isExpired(payload.exp)) {
      this.fail(`Apple token expired exp=${payload.exp}`);
    }
  }

  /** Resolve the Apple signing key matching `kid`, refreshing JWKS on miss. */
  private async getAppleJwk(kid: string): Promise<JwkWithKid> {
    const fromCache = this.appleJwksCache?.keys.find((key) => key.kid === kid);
    if (fromCache && !this.isJwksStale()) {
      return fromCache;
    }

    const keys = await this.fetchAppleJwks();
    const jwk = keys.find((key) => key.kid === kid);
    if (!jwk) {
      this.fail(`no Apple JWK matches kid=${kid}`);
    }
    return jwk;
  }

  private isJwksStale(): boolean {
    if (!this.appleJwksCache) {
      return true;
    }
    return Date.now() - this.appleJwksCache.fetchedAt > JWKS_CACHE_TTL_MS;
  }

  private async fetchAppleJwks(): Promise<JwkWithKid[]> {
    try {
      const { data } = await this.http.get<{ keys: JwkWithKid[] }>(APPLE_JWKS_URL);
      this.appleJwksCache = { keys: data.keys, fetchedAt: Date.now() };
      return data.keys;
    } catch (error) {
      const message = error instanceof Error ? error.message : 'Unknown';
      this.logger.error(`Failed to fetch Apple JWKS from ${APPLE_JWKS_URL}: ${message}`);
      this.fail('Apple JWKS unavailable');
    }
  }

  /** Verify an RS256 signature over `signedData` using a JWK public key. */
  private async verifyRs256(
    jwk: JsonWebKey,
    signedData: Uint8Array,
    signature: Uint8Array,
  ): Promise<boolean> {
    const key = await crypto.subtle.importKey(
      'jwk',
      jwk,
      { name: 'RSASSA-PKCS1-v1_5', hash: 'SHA-256' },
      false,
      ['verify'],
    );
    return crypto.subtle.verify('RSASSA-PKCS1-v1_5', key, signature, signedData);
  }

  // --------------------------------------------------------------------------
  // LINE
  // --------------------------------------------------------------------------

  private async verifyLine(idToken: string): Promise<SocialIdentity> {
    const channelId = this.config.get<string>('social.line.channelId');
    if (!channelId) {
      this.logger.error('LINE_CHANNEL_ID is not configured');
      this.fail('LINE channel id missing');
    }

    const payload = await this.requestLineVerify(idToken, channelId);
    return { providerUserId: payload.sub, email: payload.email };
  }

  /** Call LINE's verify endpoint and return the validated payload. */
  private async requestLineVerify(idToken: string, channelId: string): Promise<LineVerifyResponse> {
    let response: { data: LineVerifyResponse; ok: boolean };
    try {
      // throwOnError: false so we can inspect LINE's error payload on a 4xx.
      response = await this.http.post<LineVerifyResponse>(
        LINE_VERIFY_URL,
        new URLSearchParams({ id_token: idToken, client_id: channelId }),
        { throwOnError: false },
      );
    } catch (error) {
      const message = error instanceof Error ? error.message : 'Unknown';
      this.logger.error(`Failed to reach LINE verify endpoint: ${message}`);
      this.fail('LINE verify unavailable');
    }

    const payload = response.data;
    if (!response.ok || payload.error) {
      this.fail(`LINE verify rejected token: ${payload.error_description ?? payload.error}`);
    }
    return payload;
  }

  // --------------------------------------------------------------------------
  // Shared helpers
  // --------------------------------------------------------------------------

  /** Split and decode a compact JWS, returning header, payload and raw bytes. */
  private decodeJwt<T>(token: string): {
    header: JwtHeader;
    payload: T;
    signedData: Uint8Array;
    signature: Uint8Array;
  } {
    const parts = token.split('.');
    if (parts.length !== JWT_PARTS) {
      this.fail(`malformed JWT: expected ${JWT_PARTS} parts, got ${parts.length}`);
    }
    const [encodedHeader, encodedPayload, encodedSignature] = parts;
    try {
      const header = JSON.parse(this.base64UrlDecode(encodedHeader).toString('utf8')) as JwtHeader;
      const payload = JSON.parse(this.base64UrlDecode(encodedPayload).toString('utf8')) as T;
      return {
        header,
        payload,
        signedData: new TextEncoder().encode(`${encodedHeader}.${encodedPayload}`),
        signature: new Uint8Array(this.base64UrlDecode(encodedSignature)),
      };
    } catch (error) {
      if (error instanceof UnauthorizedException) {
        throw error;
      }
      this.fail(`JWT decode failed: ${(error as Error).message}`);
    }
  }

  private base64UrlDecode(value: string): Buffer {
    return Buffer.from(value, 'base64url');
  }

  private isExpired(exp: number): boolean {
    return Number.isFinite(exp) ? Date.now() >= exp * MS_PER_SECOND : true;
  }
}
