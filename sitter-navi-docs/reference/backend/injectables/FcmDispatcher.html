import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import * as admin from 'firebase-admin';
import { IPushDispatcher, FcmPayload } from './push-dispatcher.interface';
import { TokenBucket } from '@infrastructure/utils/token-bucket';
import { UserDeviceService } from './user-device.service';

/** Max tokens per FCM multicast request */
const FCM_BATCH_SIZE = 500;

/** FCM error codes that mean the token is permanently invalid and should be removed */
const PERMANENT_FCM_ERRORS = new Set([
  'messaging/registration-token-not-registered',
  'messaging/invalid-argument',
  'messaging/invalid-registration-token',
  'messaging/mismatched-credential',
]);

/**
 * Firebase Cloud Messaging dispatcher.
 *
 * - Sends push notifications via firebase-admin SDK.
 * - Splits device tokens into batches of 500 (FCM multicast limit).
 * - Uses in-memory TokenBucket to enforce rate limits (default: 10 req/s).
 * - Never throws — returns DispatchResult with success/error.
 */
@Injectable()
export class FcmDispatcher implements IPushDispatcher, OnModuleInit {
  private readonly logger = new Logger(FcmDispatcher.name);
  private readonly tokenBucket: TokenBucket;
  private initialized = false;

  constructor(
    private readonly configService: ConfigService,
    private readonly userDeviceService: UserDeviceService,
  ) {
    const rate = this.configService.get<number>('fcm.ratePerSecond') ?? 10;
    this.tokenBucket = new TokenBucket({ rate, burst: rate });
  }

  async onModuleInit() {
    const credentialPath = this.configService.get<string>('fcm.credentialPath');

    if (!credentialPath) {
      this.logger.warn(
        'FCM_CREDENTIAL_PATH not configured — push notifications will be NO-OP. ' +
          'Set FCM_CREDENTIAL_PATH in .env to enable FCM.',
      );
      return;
    }

    try {
      // Only initialize if not already initialized (firebase-admin is a singleton)
      if (admin.apps.length === 0) {
        admin.initializeApp({
          credential: admin.credential.cert(credentialPath),
        });
      }

      this.initialized = true;
      this.logger.log(`FCM dispatcher initialized (${this.tokenBucket['rate']} req/s)`);
    } catch (error) {
      this.logger.error(`Failed to initialize Firebase: ${(error as Error).message}`);
    }
  }

  /**
   * Dispatch a push notification to one or more devices.
   *
   * - Splits tokens into batches ≤ 500 (FCM limit).
   * - Rate-limits via TokenBucket (one token per batch).
   * - If FCM is not configured, logs and returns success (no-op mode).
   */
  async dispatch(payload: FcmPayload) {
    if (!this.initialized) {
      this.logger.warn(
        `FCM not initialized, skipping push (${payload.deviceTokens.length} devices)`,
      );
      return { success: true as const };
    }

    if (payload.deviceTokens.length === 0) {
      return { success: true as const };
    }

    try {
      // Split into batches of 500
      const batches = this.chunkArray(payload.deviceTokens, FCM_BATCH_SIZE);

      const message: admin.messaging.MulticastMessage = {
        tokens: [], // Will be set per batch
        notification: {
          title: payload.title,
          body: payload.body,
          imageUrl: payload.imageUrl,
        },
        data: payload.dataPayload ?? {},
      };

      let totalSuccess = 0;
      let totalFailure = 0;

      for (const batch of batches) {
        // Rate limit: one token per batch
        await this.tokenBucket.waitForToken();

        const response = await admin.messaging().sendEachForMulticast({
          ...message,
          tokens: batch,
        });

        totalSuccess += response.successCount;
        totalFailure += response.failureCount;

        // Log individual failures and auto-cleanup permanently invalid tokens
        if (response.failureCount > 0) {
          const deadTokens: string[] = [];
          response.responses.forEach((resp, idx) => {
            if (!resp.success) {
              const code = resp.error?.code ?? '';
              const isPermanent = PERMANENT_FCM_ERRORS.has(code);

              this.logger.warn(
                `FCM send failed for token[${idx}]: ${resp.error?.message}${isPermanent ? ' (PERMANENT — will auto-unregister)' : ''}`,
              );

              if (isPermanent) {
                deadTokens.push(batch[idx]);
              }
            }
          });

          // Auto-unregister permanently invalid tokens
          if (deadTokens.length > 0) {
            await Promise.allSettled(
              deadTokens.map((token) =>
                this.userDeviceService.unregisterDevice(token).catch(() => {}),
              ),
            );
            this.logger.log(
              `Auto-unregistered ${deadTokens.length} permanently invalid FCM tokens`,
            );
          }
        }
      }

      this.logger.debug(
        `FCM sent: ${totalSuccess} ok, ${totalFailure} failed (${batches.length} batches)`,
      );

      if (totalFailure > 0 && totalSuccess === 0) {
        return {
          success: false as const,
          error: `All ${totalFailure} sends failed`,
        };
      }

      return { success: true as const };
    } catch (error) {
      const message = error instanceof Error ? error.message : 'Unknown error';
      this.logger.error(`FCM dispatch error: ${message}`);
      return { success: false as const, error: message };
    }
  }

  private chunkArray<T>(array: T[], size: number): T[][] {
    const chunks: T[][] = [];
    for (let i = 0; i < array.length; i += size) {
      chunks.push(array.slice(i, i + size));
    }
    return chunks;
  }
}
