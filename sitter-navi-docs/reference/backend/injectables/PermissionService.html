import { Injectable, Logger, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { In, Repository } from 'typeorm';

import { Permission } from '../entities/permission.entity';
import { PermissionGroup } from '../entities/permission-group.entity';
import { PermissionGroupItem } from '../entities/permission-group-item.entity';
import { RolePermission } from '../entities/role-permission.entity';
import { UserPermission, UserPermissionType } from '../entities/user-permission.entity';
import { UserPermissionGroup } from '../entities/user-permission-group.entity';
import { PermissionCacheService } from './permission-cache.service';
import { PermissionRegistryService } from './permission-registry.service';

@Injectable()
export class PermissionService {
  private readonly logger = new Logger(PermissionService.name);

  constructor(
    @InjectRepository(Permission)
    private readonly permissionRepo: Repository<Permission>,
    @InjectRepository(PermissionGroup)
    private readonly groupRepo: Repository<PermissionGroup>,
    @InjectRepository(PermissionGroupItem)
    private readonly groupItemRepo: Repository<PermissionGroupItem>,
    @InjectRepository(RolePermission)
    private readonly rolePermRepo: Repository<RolePermission>,
    @InjectRepository(UserPermission)
    private readonly userPermRepo: Repository<UserPermission>,
    @InjectRepository(UserPermissionGroup)
    private readonly userGroupRepo: Repository<UserPermissionGroup>,
    private readonly cacheService: PermissionCacheService,
    private readonly registryService: PermissionRegistryService,
  ) {}

  // ─── Permissions ────────────────────────────────────────────────────────────

  async getAllPermissions() {
    return this.permissionRepo.find({ order: { resource: 'ASC', action: 'ASC' } });
  }

  async getDiscoveredPermissions() {
    return this.registryService.getDiscoveredPermissions();
  }

  // ─── Permission Groups ──────────────────────────────────────────────────────

  async getAllGroups() {
    return this.groupRepo.find({
      relations: ['items', 'items.permission'],
      order: { name: 'ASC' },
    });
  }

  async createGroup(data: {
    name: string;
    slug: string;
    description?: string;
    permissionIds: string[];
  }) {
    const group = this.groupRepo.create({
      name: data.name,
      slug: data.slug,
      description: data.description || null,
    });
    await this.groupRepo.save(group);

    if (data.permissionIds.length > 0) {
      const items = data.permissionIds.map((permissionId) =>
        this.groupItemRepo.create({ groupId: group.id, permissionId }),
      );
      await this.groupItemRepo.save(items);
    }

    return this.groupRepo.findOne({
      where: { id: group.id },
      relations: ['items', 'items.permission'],
    });
  }

  async updateGroupPermissions(groupId: string, permissionIds: string[]) {
    const group = await this.groupRepo.findOneBy({ id: groupId });
    if (!group) throw new NotFoundException('Permission group not found');

    // Remove existing items
    await this.groupItemRepo.delete({ groupId });

    // Add new items
    if (permissionIds.length > 0) {
      const items = permissionIds.map((permissionId) =>
        this.groupItemRepo.create({ groupId, permissionId }),
      );
      await this.groupItemRepo.save(items);
    }

    // Invalidate all users who have this group
    const userGroups = await this.userGroupRepo.find({ where: { groupId } });
    for (const ug of userGroups) {
      await this.cacheService.invalidateUser(ug.userId);
    }

    return this.groupRepo.findOne({
      where: { id: groupId },
      relations: ['items', 'items.permission'],
    });
  }

  async deleteGroup(groupId: string) {
    // Invalidate affected users first
    const userGroups = await this.userGroupRepo.find({ where: { groupId } });
    for (const ug of userGroups) {
      await this.cacheService.invalidateUser(ug.userId);
    }

    await this.groupRepo.delete(groupId);
    return { message: 'Deleted' };
  }

  // ─── Role Permissions ───────────────────────────────────────────────────────

  async getRolePermissions(role: string) {
    return this.rolePermRepo.find({
      where: { role },
      relations: ['permission'],
    });
  }

  async setRolePermissions(role: string, permissionIds: string[]) {
    // Remove existing
    await this.rolePermRepo.delete({ role });

    // Add new
    if (permissionIds.length > 0) {
      const items = permissionIds.map((permissionId) =>
        this.rolePermRepo.create({ role, permissionId }),
      );
      await this.rolePermRepo.save(items);
    }

    // Invalidate all users with this role
    await this.cacheService.invalidateRole(role);

    return this.getRolePermissions(role);
  }

  // ─── User Permissions ───────────────────────────────────────────────────────

  async getUserPermissionDetails(userId: string) {
    const [individual, groups] = await Promise.all([
      this.userPermRepo.find({
        where: { userId },
        relations: ['permission'],
      }),
      this.userGroupRepo.find({
        where: { userId },
        relations: ['group', 'group.items', 'group.items.permission'],
      }),
    ]);

    return { individual, groups };
  }

  async assignPermissionsToUser(
    userId: string,
    permissionIds: string[],
    type: UserPermissionType = UserPermissionType.GRANTED,
  ) {
    // Upsert: remove existing entries for these permissions, then insert
    await this.userPermRepo.delete({
      userId,
      permissionId: In(permissionIds),
    });

    const items = permissionIds.map((permissionId) =>
      this.userPermRepo.create({ userId, permissionId, type }),
    );
    await this.userPermRepo.save(items);

    await this.cacheService.invalidateUser(userId);
    return { message: 'Updated' };
  }

  async revokePermissionsFromUser(userId: string, permissionIds: string[]) {
    return this.assignPermissionsToUser(userId, permissionIds, UserPermissionType.REVOKED);
  }

  async removeUserPermissions(userId: string, permissionIds: string[]) {
    await this.userPermRepo.delete({
      userId,
      permissionId: In(permissionIds),
    });
    await this.cacheService.invalidateUser(userId);
    return { message: 'Removed' };
  }

  // ─── User Permission Groups ─────────────────────────────────────────────────

  async assignGroupsToUser(userId: string, groupIds: string[]) {
    // Remove duplicates
    const existing = await this.userGroupRepo.find({
      where: { userId, groupId: In(groupIds) },
    });
    const existingGroupIds = new Set(existing.map((e) => e.groupId));
    const newGroupIds = groupIds.filter((id) => !existingGroupIds.has(id));

    if (newGroupIds.length > 0) {
      const items = newGroupIds.map((groupId) => this.userGroupRepo.create({ userId, groupId }));
      await this.userGroupRepo.save(items);
    }

    await this.cacheService.invalidateUser(userId);
    return { message: 'Updated' };
  }

  async removeGroupsFromUser(userId: string, groupIds: string[]) {
    await this.userGroupRepo.delete({
      userId,
      groupId: In(groupIds),
    });
    await this.cacheService.invalidateUser(userId);
    return { message: 'Removed' };
  }

  // ─── Effective Permissions (resolved) ───────────────────────────────────────

  async getEffectivePermissions(userId: string, role: string) {
    const permissions = await this.cacheService.getUserPermissions(userId, role);
    return [...permissions].sort();
  }
}
