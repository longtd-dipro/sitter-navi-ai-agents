import { Injectable, LoggerService } from '@nestjs/common';
import { LoggingService } from './logging.service';

/**
 * Adapter that bridges NestJS's built-in LoggerService interface to our
 * Winston-backed LoggingService.
 *
 * Usage in main.ts:
 *   const adapter = new NestLoggerAdapter();
 *   const app = await NestFactory.create(AppModule, { logger: adapter, bufferLogs: true });
 *   const loggingService = app.get(LoggingService);
 *   adapter.setLoggingService(loggingService);
 */
@Injectable()
export class NestLoggerAdapter implements LoggerService {
  private loggingService: LoggingService | null = null;

  /**
   * Called after the DI container is ready to wire in the real LoggingService.
   * Until this is called, all log calls fall back to process.stdout JSON lines.
   */
  setLoggingService(service: LoggingService): void {
    this.loggingService = service;
  }

  /**
   * Maps NestJS `log` → LoggingService.info
   */
  log(message: any, context?: string): void {
    const ctx = context ?? 'NestJS';
    const msg = this.stringify(message);

    if (this.loggingService) {
      const child = this.loggingService.child(ctx);
      child.info(msg);
    } else {
      this.writeStdout('info', msg, ctx);
    }
  }

  /**
   * Maps NestJS `error` → LoggingService.error
   * Includes stack trace in metadata when provided.
   */
  error(message: any, stack?: string, context?: string): void {
    const ctx = context ?? 'NestJS';
    const msg = this.stringify(message);

    if (this.loggingService) {
      const child = this.loggingService.child(ctx);
      child.error(msg, stack ? { stack } : undefined);
    } else {
      this.writeStdout('error', msg, ctx, stack ? { stack } : undefined);
    }
  }

  /**
   * Maps NestJS `warn` → LoggingService.warn
   */
  warn(message: any, context?: string): void {
    const ctx = context ?? 'NestJS';
    const msg = this.stringify(message);

    if (this.loggingService) {
      const child = this.loggingService.child(ctx);
      child.warn(msg);
    } else {
      this.writeStdout('warn', msg, ctx);
    }
  }

  /**
   * Maps NestJS `debug` → LoggingService.debug
   */
  debug(message: any, context?: string): void {
    const ctx = context ?? 'NestJS';
    const msg = this.stringify(message);

    if (this.loggingService) {
      const child = this.loggingService.child(ctx);
      child.debug(msg);
    } else {
      this.writeStdout('debug', msg, ctx);
    }
  }

  /**
   * Maps NestJS `verbose` → LoggingService.verbose
   */
  verbose(message: any, context?: string): void {
    const ctx = context ?? 'NestJS';
    const msg = this.stringify(message);

    if (this.loggingService) {
      const child = this.loggingService.child(ctx);
      child.verbose(msg);
    } else {
      this.writeStdout('verbose', msg, ctx);
    }
  }

  // ---------------------------------------------------------------------------
  // Private helpers
  // ---------------------------------------------------------------------------

  /**
   * Converts any message value to a string.
   * Objects are JSON-serialized; primitives are coerced via String().
   */
  private stringify(message: any): string {
    if (typeof message === 'string') {
      return message;
    }
    if (message instanceof Error) {
      return message.message;
    }
    try {
      return JSON.stringify(message);
    } catch {
      return String(message);
    }
  }

  /**
   * Fallback writer used before the DI container is ready.
   * Writes a single-line JSON record to process.stdout.
   */
  private writeStdout(
    level: string,
    message: string,
    context: string,
    extra?: Record<string, unknown>,
  ): void {
    const record: Record<string, unknown> = {
      timestamp: new Date().toISOString(),
      level,
      context,
      message,
      ...extra,
    };
    process.stdout.write(JSON.stringify(record) + '\n');
  }
}
