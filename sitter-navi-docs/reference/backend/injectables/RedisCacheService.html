/*
  This class maps env variables to a redis io config object
  */
@Injectable()
export class RedisCacheConfigurationMapper {
  public static map(): IoRedis.RedisOptions {
    return {
      host: process.env.REDIS_HOST,
      port: parseInt(process.env.REDIS_PORT || '6379', 10),
      password: process.env.REDIS_PASSWORD,
      connectTimeout: 10000,
      enableOfflineQueue: true,
      keyPrefix: getRedisKeyPrefix(),
      tls: process.env.REDIS_USETLS === 'true' ? {} : undefined,
    };
  }
}

/*
  This is where we setup the typecacheable store. We use Nest's OnModuleInit
  interface to have the setup run immediately.
  This allows us to stop application start if there is a problem
  configuring our redis instance.
  */
@Injectable()
export class RedisCacheService implements OnModuleInit {
  private redisInstance: IoRedis.Redis | undefined;
  private hasLoggedError = false;
  private inMemoryAdapter: InMemoryCacheAdapter | undefined;
  private readonly logger = new Logger('RedisCacheService');

  /** Whether Redis is currently the active cache backend */
  get isRedisActive(): boolean {
    return this.isConnectReady();
  }

  /** Which backend is currently serving cache */
  get activeBackend(): 'redis' | 'in-memory' {
    return this.isConnectReady() ? 'redis' : 'in-memory';
  }

  public async onModuleInit(): Promise<void> {
    // Always set up in-memory fallback first
    this.inMemoryAdapter = new InMemoryCacheAdapter({ maxEntries: 10000 });
    cacheManager.setFallbackClient(this.inMemoryAdapter);

    try {
      if (this.isAlreadyConfigured()) {
        return;
      }

      this.redisInstance = new IORedis({
        ...RedisCacheConfigurationMapper.map(),
        retryStrategy: (times) => {
          // Retry with backoff, max 10s between retries
          return Math.min(times * 500, 10000);
        },
      });

      // Suppress repeated error logs — only log the first occurrence
      this.redisInstance.on('error', (e: Error) => {
        if (!this.hasLoggedError) {
          this.hasLoggedError = true;
          this.logger.warn(
            `Redis connection error (using in-memory fallback, subsequent errors suppressed): ${e.message}`,
          );
        }
      });

      // Reset error flag when reconnected so next disconnect is logged
      this.redisInstance.on('ready', () => {
        if (this.hasLoggedError) {
          this.logger.log('Redis reconnected, resuming as primary cache');
          this.hasLoggedError = false;
        }
      });

      useAdapter(this.redisInstance);
    } catch (e) {
      this.logger.warn(
        `Failed to initialize Redis cache, using in-memory fallback: ${(e as Error).message}`,
      );
    }
  }

  private isAlreadyConfigured(): boolean {
    return this.redisInstance !== undefined;
  }

  private isConnectReady(): boolean {
    return 'ready' === this?.redisInstance?.status;
  }

  public clearCacheByPattern(pattern) {
    return new Promise((resolve) => {
      if (!this.redisInstance) {
        throw new Error('Redis instance is not initialized');
      }
      const stream = this.redisInstance.scanStream({
        match: pattern,
      });
      stream.on('data', (keys) => {
        if (keys.length) {
          const pipeline = this.redisInstance?.pipeline();
          if (!pipeline) {
            throw new Error('Redis instance is not initialized');
          }
          keys.forEach(function (key) {
            pipeline.del(key);
          });
          pipeline.exec();
        }
      });
      stream.on('end', function () {
        resolve(null);
      });
    });
  }

  public async clearCacheByTopic(key: string) {
    if (!this.redisInstance) {
      throw new Error('Redis instance is not initialized');
    }
    if (key === 'config') {
      return this.redisInstance?.del('config');
    }
    if (key === 'product') {
      return this.clearCacheByPattern('product-list:*');
    }
    if (key === 'cms') {
      return this.clearCacheByPattern('cms:*');
    }
    if (key === 'address') {
      return this.redisInstance?.del('address');
    }
    if (key === 'productDetail') {
      return this.clearCacheByPattern('product-detail:*');
    }
    if (key === 'userRole') {
      return this.clearCacheByPattern(getRolePermissionCacheKey('*'));
    }
    if (key === 'all') {
      try {
        if (this.redisInstance) {
          this.redisInstance.flushall();
        }
      } catch {
        this.clearCacheByPattern('*');
      }
      return true;
    } else {
      return this.redisInstance.del(key);
    }
  }

  @CheckRedisReady
  public async clearCacheByKey(key: string) {
    if (!this.redisInstance) {
      throw new Error('Redis instance is not initialized');
    }
    return this.redisInstance.del(key);
  }

  @CheckRedisReady
  public async getValue(key: string) {
    if (!this.redisInstance) {
      throw new Error('Redis instance is not initialized');
    }
    return this.redisInstance.get(key);
  }

  @CheckRedisReady
  public async setValue(key: string, value: string | number | Buffer) {
    if (!this.redisInstance) {
      throw new Error('Redis instance is not initialized');
    }
    return this.redisInstance.set(key, value);
  }

  @CheckRedisReady
  public async setValueWithTTL(key: string, value: string | number | Buffer, ttlSeconds: number) {
    if (!this.redisInstance) {
      throw new Error('Redis instance is not initialized');
    }
    return this.redisInstance.set(key, value, 'EX', ttlSeconds);
  }

  @CheckRedisReady
  public async incrBy(key: string, increment: number = 1): Promise<number> {
    if (!this.redisInstance) {
      throw new Error('Redis instance is not initialized');
    }
    return this.redisInstance.incrby(key, increment);
  }

  @CheckRedisReady
  public async expire(key: string, ttlSeconds: number): Promise<number> {
    if (!this.redisInstance) {
      throw new Error('Redis instance is not initialized');
    }
    return this.redisInstance.expire(key, ttlSeconds);
  }

  @CheckRedisReady
  public async mget(keys: string[]): Promise<(string | null)[]> {
    if (!this.redisInstance) {
      throw new Error('Redis instance is not initialized');
    }
    return this.redisInstance.mget(...keys);
  }

  /**
   * Create a Redis pipeline for batching multiple commands into one round-trip.
   * Returns an object with chainable command methods and an exec() method.
   *
   * Usage:
   *   const pipe = cacheService.pipeline();
   *   pipe.incrby('key1', 1);
   *   pipe.incrby('key2', 1);
   *   pipe.expire('key1', 3600);
   *   await pipe.exec();
   */
  public pipeline(): RedisPipeline {
    if (!this.redisInstance) {
      // Fallback: return a no-op pipeline that silently fails each command
      return new NoopPipeline();
    }
    return new IoredisPipeline(this.redisInstance.pipeline());
  }
}

/**
 * Interface for chainable Redis pipeline operations.
 */
export interface RedisPipeline {
  incrby(key: string, increment: number): this;
  expire(key: string, seconds: number): this;
  exec(): Promise<void>;
}

/**
 * Real ioredis pipeline wrapper.
 */
class IoredisPipeline implements RedisPipeline {
  constructor(private readonly pipe: any) {}

  incrby(key: string, increment: number): this {
    this.pipe.incrby(key, increment);
    return this;
  }

  expire(key: string, seconds: number): this {
    this.pipe.expire(key, seconds);
    return this;
  }

  async exec(): Promise<void> {
    await this.pipe.exec();
  }
}

/**
 * No-op pipeline for when Redis is unavailable.
 */
class NoopPipeline implements RedisPipeline {
  incrby(_key: string, _increment: number): this {
    return this;
  }
  expire(_key: string, _seconds: number): this {
    return this;
  }
  async exec(): Promise<void> {}
}

function CheckRedisReady(
  target,
  propertyKey: string,
  descriptor: PropertyDescriptor,
): PropertyDescriptor {
  const originalValue = descriptor.value;
  descriptor.value = function (...args: []) {
    // "this" here will refer to the class instance
    if (!this.isConnectReady()) {
      throw Error('Redis Not Ready');
    }
    return originalValue.apply(this, args);
  };
  return descriptor;
}
