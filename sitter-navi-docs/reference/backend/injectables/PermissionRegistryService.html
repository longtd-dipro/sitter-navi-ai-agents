import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { DiscoveryService, MetadataScanner, Reflector } from '@nestjs/core';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';

import { Permission } from '../entities/permission.entity';
import { RESOURCE_PERMISSION_KEY } from '../decorators/resource-permission.decorator';
import { REQUIRE_PERMISSIONS_KEY } from '../decorators/require-permissions.decorator';

export interface DiscoveredPermission {
  resource: string;
  action: string;
  key: string;
  source: string;
  httpMethod: string | null;
}

const HTTP_METHOD_MAP: Record<number, string> = {
  0: 'GET',
  1: 'POST',
  2: 'PUT',
  3: 'DELETE',
  4: 'PATCH',
  5: 'OPTIONS',
  6: 'HEAD',
  7: 'ALL',
};

const HTTP_METHOD_ACTION_MAP: Record<string, string> = {
  GET: 'read',
  POST: 'create',
  PATCH: 'update',
  PUT: 'update',
  DELETE: 'delete',
};

@Injectable()
export class PermissionRegistryService implements OnModuleInit {
  private readonly logger = new Logger(PermissionRegistryService.name);
  private discoveredPermissions: DiscoveredPermission[] = [];

  constructor(
    private readonly discoveryService: DiscoveryService,
    private readonly metadataScanner: MetadataScanner,
    private readonly reflector: Reflector,
    @InjectRepository(Permission)
    private readonly permissionRepo: Repository<Permission>,
  ) {}

  async onModuleInit(): Promise<void> {
    this.scanControllers();
    await this.syncToDatabase();
  }

  /**
   * Returns all permissions discovered from controller decorators.
   */
  getDiscoveredPermissions(): DiscoveredPermission[] {
    return [...this.discoveredPermissions];
  }

  /**
   * Scan all registered controllers for @ResourcePermission and @RequirePermissions.
   */
  private scanControllers(): void {
    const controllers = this.discoveryService.getControllers();

    for (const wrapper of controllers) {
      const { instance, metatype } = wrapper;
      if (!instance || !metatype) continue;

      const controllerName = metatype.name;

      // Get class-level resource
      const resource = this.reflector.get<string>(RESOURCE_PERMISSION_KEY, metatype);

      // Scan all methods
      const prototype = Object.getPrototypeOf(instance);
      const methodNames = this.metadataScanner.getAllMethodNames(prototype);

      for (const methodName of methodNames) {
        const handler = prototype[methodName];
        if (!handler) continue;

        // Check method-level @RequirePermissions
        const methodPermissions = this.reflector.get<string[]>(REQUIRE_PERMISSIONS_KEY, handler);

        if (methodPermissions && methodPermissions.length > 0) {
          for (const perm of methodPermissions) {
            const [res, action] = perm.split(':');
            this.addPermission({
              resource: res,
              action: action || 'custom',
              key: perm,
              source: `${controllerName}.${methodName}`,
              httpMethod: this.getHttpMethod(handler),
            });
          }
        } else if (resource) {
          // Auto-map from HTTP method
          const httpMethod = this.getHttpMethod(handler);
          if (httpMethod && HTTP_METHOD_ACTION_MAP[httpMethod]) {
            const action = HTTP_METHOD_ACTION_MAP[httpMethod];
            this.addPermission({
              resource,
              action,
              key: `${resource}:${action}`,
              source: `${controllerName}.${methodName}`,
              httpMethod,
            });
          }
        }
      }
    }

    this.logger.log(
      `Discovered ${this.discoveredPermissions.length} permissions from ${controllers.length} controllers`,
    );
  }

  private addPermission(perm: DiscoveredPermission): void {
    // Deduplicate by key
    if (!this.discoveredPermissions.find((p) => p.key === perm.key)) {
      this.discoveredPermissions.push(perm);
    }
  }

  private getHttpMethod(handler: Function): string | null {
    const method = Reflect.getMetadata('method', handler);
    if (method === undefined || method === null) return null;
    return HTTP_METHOD_MAP[method] || null;
  }

  /**
   * Sync discovered permissions to DB:
   * - Insert new permissions that don't exist yet
   * - Log warnings for DB permissions that are no longer in code
   */
  private async syncToDatabase(): Promise<void> {
    const existingPermissions = await this.permissionRepo.find();
    const existingKeys = new Set(existingPermissions.map((p) => p.key));
    const discoveredKeys = new Set(this.discoveredPermissions.map((p) => p.key));

    // Insert new permissions
    const newPermissions = this.discoveredPermissions.filter((p) => !existingKeys.has(p.key));

    if (newPermissions.length > 0) {
      const entities = newPermissions.map((p) =>
        this.permissionRepo.create({
          resource: p.resource,
          action: p.action,
          key: p.key,
          description: null,
          source: p.source,
          httpMethod: p.httpMethod,
        }),
      );
      await this.permissionRepo.save(entities);
      this.logger.log(`Seeded ${newPermissions.length} new permissions to DB`);
    }

    // Warn about orphan permissions (in DB but not in code)
    const orphanPermissions = existingPermissions.filter((p) => !discoveredKeys.has(p.key));
    if (orphanPermissions.length > 0) {
      this.logger.warn(
        `Found ${orphanPermissions.length} orphan permissions in DB (no longer in code): ${orphanPermissions.map((p) => p.key).join(', ')}`,
      );
    }
  }
}
