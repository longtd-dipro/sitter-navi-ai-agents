Description

Central push notification service — orchestrates template lookup, rendering, outbox persistence, and dispatch via FCM.

Public API:

  • sendToUser(templateName, userId, data) — find template → user's devices → send
  • sendToDevices(templateName, deviceTokens, data) — send directly to token list
  • sendCustom(title, body, deviceTokens) — raw push, no template

Event handlers:

  • Each event in PUSH_EVENT_MAP is registered as an @OnEvent handler.
  • Handlers are fire-and-forget: errors are logged but never re-thrown.

import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { Inject } from '@nestjs/common';
import { PushTemplateService } from './push-template.service';
import { PushRendererService } from './push-renderer.service';
import { OutboxPushService } from './outbox-push.service';
import { UserDeviceService } from './user-device.service';
import { IPushDispatcher, PUSH_DISPATCHER } from './push-dispatcher.interface';
import { PUSH_EVENT_MAP, PushEventName } from './push-event.registry';

/**
 * Central push notification service — orchestrates template lookup,
 * rendering, outbox persistence, and dispatch via FCM.
 *
 * Public API:
 *   - sendToUser(templateName, userId, data)    — find template → user's devices → send
 *   - sendToDevices(templateName, deviceTokens, data) — send directly to token list
 *   - sendCustom(title, body, deviceTokens)     — raw push, no template
 *
 * Event handlers:
 *   - Each event in PUSH_EVENT_MAP is registered as an @OnEvent handler.
 *   - Handlers are fire-and-forget: errors are logged but never re-thrown.
 */
@Injectable()
export class PushService implements OnModuleInit {
  private readonly logger = new Logger(PushService.name);

  constructor(
    private readonly templateService: PushTemplateService,
    private readonly renderer: PushRendererService,
    private readonly outboxService: OutboxPushService,
    private readonly userDeviceService: UserDeviceService,
    @Inject(PUSH_DISPATCHER)
    private readonly dispatcher: IPushDispatcher,
  ) {}

  onModuleInit() {
    this.logger.log(`Registered ${Object.keys(PUSH_EVENT_MAP).length} push event handlers`);
  }

  // ──────────────────────────────────────────────────────────────────
  // Public API
  // ──────────────────────────────────────────────────────────────────

  /**
   * Send a push notification to all active devices of a user.
   *
   * @param templateName - The unique template name (e.g. 'PN001').
   * @param userId        - The user ID to send to.
   * @param data          - Context data for EJS rendering.
   */
  async sendToUser(
    templateName: string,
    userId: string,
    data: Record<string, unknown> = {},
  ): Promise<void> {
    // 1. Find template
    const template = await this.templateService.findByTemplateName(templateName);
    if (!template) {
      this.logger.error(`Template not found: "${templateName}"`);
      return;
    }

    if (!template.isActive) {
      this.logger.warn(`Template "${templateName}" is inactive, skipping`);
      return;
    }

    // 2. Get user's device tokens
    const deviceTokens = await this.userDeviceService.getActiveDevices(userId);
    if (deviceTokens.length === 0) {
      this.logger.debug(`No active devices for user ${userId}, skipping push`);
      return;
    }

    // 3. Render
    let rendered;
    try {
      rendered = await this.renderer.render(template, data);
    } catch (err) {
      this.logger.error(`Render failed for template "${templateName}"`, (err as Error).stack);
      return;
    }

    // 4. Persist to outbox
    const outboxRecord = await this.outboxService.createPending({
      userId,
      deviceTokens,
      title: rendered.title,
      body: rendered.body,
      imageUrl: rendered.imageUrl ?? null,
      dataPayload: rendered.dataPayload ?? null,
      templateName,
    } as any);

    // 5. Dispatch
    const result = await this.dispatcher.dispatch({
      deviceTokens,
      title: rendered.title,
      body: rendered.body,
      imageUrl: rendered.imageUrl,
      dataPayload: rendered.dataPayload,
    });

    if (result.success) {
      await this.outboxService.markSent(outboxRecord.id);
    } else {
      await this.outboxService.markFailed(
        outboxRecord.id,
        result.error ?? 'Unknown dispatch error',
        false,
      );
    }
  }

  /**
   * Send a push notification directly to a list of device tokens.
   * Useful when you already know the tokens (e.g., from a batch query).
   */
  async sendToDevices(
    templateName: string,
    deviceTokens: string[],
    data: Record<string, unknown> = {},
  ): Promise<void> {
    if (deviceTokens.length === 0) return;

    const template = await this.templateService.findByTemplateName(templateName);
    if (!template || !template.isActive) return;

    let rendered;
    try {
      rendered = await this.renderer.render(template, data);
    } catch (err) {
      this.logger.error(`Render failed for template "${templateName}"`, (err as Error).stack);
      return;
    }

    const outboxRecord = await this.outboxService.createPending({
      deviceTokens,
      title: rendered.title,
      body: rendered.body,
      imageUrl: rendered.imageUrl ?? null,
      dataPayload: rendered.dataPayload ?? null,
      templateName,
    } as any);

    const result = await this.dispatcher.dispatch({
      deviceTokens,
      title: rendered.title,
      body: rendered.body,
      imageUrl: rendered.imageUrl,
      dataPayload: rendered.dataPayload,
    });

    if (result.success) {
      await this.outboxService.markSent(outboxRecord.id);
    } else {
      await this.outboxService.markFailed(
        outboxRecord.id,
        result.error ?? 'Unknown dispatch error',
        false,
      );
    }
  }

  /**
   * Send a raw push notification without a template.
   */
  async sendCustom(title: string, body: string, deviceTokens: string[]): Promise<void> {
    if (deviceTokens.length === 0) return;

    const outboxRecord = await this.outboxService.createPending({
      deviceTokens,
      title,
      body,
    } as any);

    const result = await this.dispatcher.dispatch({
      deviceTokens,
      title,
      body,
    });

    if (result.success) {
      await this.outboxService.markSent(outboxRecord.id);
    } else {
      await this.outboxService.markFailed(
        outboxRecord.id,
        result.error ?? 'Unknown dispatch error',
        false,
      );
    }
  }
}
