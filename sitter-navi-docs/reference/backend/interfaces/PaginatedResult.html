import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { In, Not, Repository } from 'typeorm';

import { RoleType } from '@common/enums';
import { BaseRepository } from '@common/repositories';
import { UserEntity } from '@entities';
import { SocialAccountEntity } from './social-account.entity';
import { RegisterDto } from '../auth/dto/register';
import { generateSnowflakeId } from '@utils/snowflake';

/**
 * Options used by {@link UserRepository.findWithPagination} to build the
 * paginated user listing query (admin/manager screens).
 */
export interface UserQueryOptions {
  /** Optional case-insensitive substring matched against fullName and email. */
  search?: string;
  /** Optional role filter (admin / manager / staff). */
  role?: RoleType;
  /** 1-based page number. */
  page: number;
  /** Page size. */
  limit: number;
  /** Number of rows to skip (typically `(page - 1) * limit`). */
  offset: number;
}

/**
 * Generic shape returned from {@link UserRepository.findWithPagination}.
 */
export interface PaginatedResult<T> {
  data: T[];
  total: number;
  page: number;
  pageCount: number;
}

/**
 * Repository_Layer for the User aggregate.
 *
 * Responsibilities:
 *  - Encapsulate every read/write against the `user` table.
 *  - Inherit the standard {@link BaseRepository} contract
 *    (`findById`, `findByIdOrFail`, `save`, `softDelete`, `count`).
 *  - Expose User-specific query methods used by `UserService` and
 *    other Complex_Business_Module services that need to look up
 *    users via the User module's public surface.
 *
 * Constraints (per design / Requirements 3.x and 12.x):
 *  - MUST inject `Repository<UserEntity>` via `@InjectRepository`.
 *  - MUST NOT inject services from other modules.
 *  - MUST NOT inject `EventEmitter2` — emitting domain events is the
 *    Service layer's responsibility.
 *  - MUST NOT enforce authorization or contain business validation.
 *
 * The `entityName` ("user") is used by `findByIdOrFail` to throw a
 * {@link NotFoundException} with the i18n key `'user.not_found'`.
 */
@Injectable()
export class UserRepository extends BaseRepository<UserEntity> {
  protected readonly entityName = 'user';

  constructor(
    @InjectRepository(UserEntity)
    protected readonly repo: Repository<UserEntity>,
    @InjectRepository(SocialAccountEntity)
    protected readonly socialRepo: Repository<SocialAccountEntity>,
  ) {
    super();
  }

  async createUserWithSocialAccount(
    userRegisterDto: RegisterDto,
    provider: string,
    providerUserId: string,
    providerEmail?: string,
  ): Promise<UserEntity> {
    const userId = generateSnowflakeId();

    // Insert the user first, then the social account, in a single
    // transaction: the social account FK references the user row, and we
    // must not end up with one persisted without the other.
    return this.repo.manager.transaction(async (manager) => {
      const userEntity = new UserEntity();
      Object.assign(userEntity, userRegisterDto);
      userEntity.id = userId;

      const savedUser = await manager.save(userEntity);

      const socialAccount = new SocialAccountEntity();
      socialAccount.userId = savedUser.id;
      socialAccount.provider = provider as any;
      socialAccount.providerUserId = providerUserId;
      socialAccount.providerEmail = providerEmail;

      await manager.save(socialAccount);

      return savedUser;
    });
  }

  async createUser(userRegisterDto: RegisterDto): Promise<UserEntity> {
    const userEntity = new UserEntity();
    Object.assign(userEntity, userRegisterDto);

    const user = await this.repo.save(userEntity);
    return user;
  }

  /**
   * Look up a user by email. Returns `null` when no row matches.
   */
  async findByEmail(email: string): Promise<UserEntity | null> {
    return this.repo.findOneBy({ email });
  }

  /**
   * Check whether a user with the given email exists. When `excludeId`
   * is provided the row with that id is ignored — useful for
   * "email unique on update" checks.
   */
  async existsByEmail(email: string, excludeId?: string): Promise<boolean> {
    if (excludeId) {
      return this.repo.existsBy({ email, id: Not(excludeId) });
    }
    return this.repo.existsBy({ email });
  }

  /**
   * Bulk lookup by ids. Returns an empty array when `ids` is empty
   * (avoids issuing a `WHERE id IN ()` query that some drivers reject).
   */
  async findManyByIds(ids: string[]): Promise<UserEntity[]> {
    if (!ids.length) return [];
    return this.repo.findBy({ id: In(ids) });
  }

  /**
   * Return all admin users.
   */
  async findAdminUsers(): Promise<UserEntity[]> {
    return this.repo.findBy({ role: RoleType.ADMIN });
  }

  /**
   * Paginated user listing with optional search, role and company
   * filters. Search matches case-insensitively against `fullName` and
   * `email`.
   */
  async findWithPagination(options: UserQueryOptions): Promise<PaginatedResult<UserEntity>> {
    const qb = this.repo.createQueryBuilder('user');

    if (options.role) {
      qb.andWhere('user.role = :role', { role: options.role });
    }

    if (options.search) {
      qb.andWhere('(LOWER(user.fullName) LIKE :search OR LOWER(user.email) LIKE :search)', {
        search: `%${options.search.toLowerCase()}%`,
      });
    }

    const [data, total] = await qb.skip(options.offset).take(options.limit).getManyAndCount();

    const pageCount = options.limit > 0 ? Math.ceil(total / options.limit) : 0;

    return {
      data,
      total,
      page: options.page,
      pageCount,
    };
  }

  /**
   * Persist a new password hash for the given user. Implemented as a
   * direct UPDATE so the operation does not load the entity first
   * (used by password reset / change-password flows).
   */
  async updatePassword(id: string, hashedPassword: string): Promise<void> {
    await this.repo
      .createQueryBuilder()
      .update(UserEntity)
      .set({ password: hashedPassword })
      .where('id = :id', { id })
      .execute();
  }

  /**
   * Load an active user for the standard login flow.
   * Returns `null` when the user does not exist or is not active.
   */
  async findActiveLoginCandidate(email: string): Promise<UserEntity | null> {
    return this.repo.findOne({
      where: { email, isActive: true, role: Not(RoleType.ADMIN) },
      select: ['password', 'email', 'id', 'role', 'fullNameKanji', 'fullNameKana', 'isLocked'],
    });
  }

  /**
   * Find an active admin by email — used by `AuthService.adminLogin`.
   * Returns the user with the auth-relevant columns selected
   * (including `password` for hash validation).
   */
  async findActiveAdminByEmail(email: string): Promise<UserEntity | null> {
    return this.repo.findOne({
      where: {
        email,
        isActive: true,
        role: RoleType.ADMIN,
      },
      select: ['password', 'email', 'id', 'role', 'fullNameKanji', 'fullNameKana', 'isLocked'],
    });
  }

  /**
   * Find an admin by email regardless of `isActive` flag — used by
   * `AuthService.adminRequestReset`. Returns `null` when no admin
   * matches (callers may treat `null` as "no such account").
   */
  async findAdminByEmail(email: string): Promise<UserEntity | null> {
    return this.repo.findOneBy({
      email,
      role: RoleType.ADMIN,
    });
  }

  /**
   * Load a user by ID. Used by `AuthService.getUserInfo`.
   */
  async findByIdWithCompany(id: string): Promise<UserEntity | null> {
    return this.repo.findOne({
      where: { id },
      select: ['id', 'email', 'role', 'fullNameKanji', 'fullNameKana', 'isActive'],
    });
  }

  /**
   * Load a user by ID. Used by `AuthService.getUserInfo`.
   */
  async findSocialProfile(provider: string, providerUserId: string): Promise<UserEntity | null> {
    return this.repo.findOne({
      where: {
        socialAccounts: {
          provider: provider as any,
          providerUserId,
        },
      },
      relations: ['socialAccounts'],
      select: ['id', 'email', 'role', 'fullNameKanji', 'fullNameKana', 'isActive'],
    });
  }
}
