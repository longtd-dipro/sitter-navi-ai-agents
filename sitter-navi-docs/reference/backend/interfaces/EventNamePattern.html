export const EVENT_NAME_REGEX = /^[a-z][a-z0-9-]*\.[a-z][a-z0-9-]*$/;

/**
 * Interface mô tả contract cho event name pattern validation.
 *
 * Cho phép consumer kiểm tra một string có tuân theo convention
 * `{aggregate}.{action}` hay không trước khi register/emit.
 */
export interface EventNamePattern {
  /** Regex pattern dùng để validate event name. */
  readonly pattern: RegExp;

  /**
   * Trả về `true` nếu `name` tuân theo convention `{aggregate}.{action}`.
   *
   * @param name Tên event cần kiểm tra (ví dụ: `user.created`).
   */
  isValid(name: string): boolean;
}

/**
 * Singleton implementation của `EventNamePattern`.
 *
 * Sử dụng để validate event name tại runtime hoặc trong test:
 * ```ts
 * EventNamePatternValidator.isValid('user.created'); // true
 * EventNamePatternValidator.isValid('User.Created'); // false
 * ```
 */
export const EventNamePatternValidator: EventNamePattern = {
  pattern: EVENT_NAME_REGEX,
  isValid(name: string): boolean {
    return EVENT_NAME_REGEX.test(name);
  },
};

/**
 * Abstract base class cho tất cả domain events.
 *
 * Mọi concrete event class:
 * - PHẢI declare `static readonly eventName: string` tuân theo `EVENT_NAME_REGEX`.
 * - PHẢI declare các payload field là `readonly` (immutability).
 * - PHẢI có constructor nhận đầy đủ các trường bắt buộc của payload.
 * - KHÔNG được chứa các field nhạy cảm: `password`, `passwordHash`,
 *   `accessToken`, `refreshToken`, `otpCode`.
 *
 * `occurredAt` được auto-populate tại thời điểm event được khởi tạo,
 * dùng cho audit trail và ordering.
 *
 * @example
 * ```ts
 * export class UserCreatedEvent extends DomainEvent {
 *   static readonly eventName = 'user.created';
 *
 *   constructor(
 *     public readonly userId: string,
 *     public readonly companyId: string,
 *   ) {
 *     super();
 *   }
 * }
 * ```
 */
export abstract class DomainEvent {
  /**
   * Canonical event name, override trong concrete class.
   *
   * TypeScript không hỗ trợ `abstract static`, nên field này được khai báo
   * như một contract bằng convention. Concrete class PHẢI override.
   */
  static readonly eventName: string;

  /** Thời điểm event được khởi tạo (UTC). */
  readonly occurredAt: Date;

  protected constructor() {
    this.occurredAt = new Date();
  }
}
