import { toJST } from '@foundation';

export interface BusinessHoursConfig {
  startHour: number; // 9
  endHour: number; // 18
  workDays: number[]; // [1, 2, 3, 4, 5] Monday to Friday
}

export const DEFAULT_BUSINESS_HOURS: BusinessHoursConfig = {
  startHour: 9,
  endHour: 18,
  workDays: [1, 2, 3, 4, 5],
};

export function isWithinBusinessHours(
  date: Date,
  config: BusinessHoursConfig = DEFAULT_BUSINESS_HOURS,
  alreadyZoned = false,
): boolean {
  const zonedDate = alreadyZoned ? date : toJST(date);
  const day = zonedDate.getDay();
  const hour = zonedDate.getHours();
  if (!config.workDays.includes(day)) return false;
  return hour >= config.startHour && hour < config.endHour;
}

export function getBusinessHoursAutoMessage(
  date: Date = new Date(),
  config: BusinessHoursConfig = DEFAULT_BUSINESS_HOURS,
): string {
  const inBusinessHours = isWithinBusinessHours(date, config);

  const start = `${config.startHour.toString().padStart(2, '0')}:00`;
  const end = `${config.endHour.toString().padStart(2, '0')}:00`;

  return inBusinessHours
    ? 'システムは現在、管理者と接続中です。少々お待ちください。'
    : `現在は営業時間外のため、担当者は翌営業日にご回答いたします。営業時間（平日${start}～${end}）`;
}

export function shouldShowAutoMessage(
  currentDate: Date,
  lastMessageDate?: Date,
  config: BusinessHoursConfig = DEFAULT_BUSINESS_HOURS,
): boolean {
  const zonedCurrent = toJST(currentDate);

  const zonedLast = lastMessageDate ? toJST(lastMessageDate) : undefined;

  if (!zonedLast) return true;

  if (
    zonedCurrent.getDate() !== zonedLast.getDate() ||
    zonedCurrent.getMonth() !== zonedLast.getMonth() ||
    zonedCurrent.getFullYear() !== zonedLast.getFullYear()
  ) {
    return true;
  }

  return (
    isWithinBusinessHours(zonedCurrent, config, true) !==
    isWithinBusinessHours(zonedLast, config, true)
  );
}
