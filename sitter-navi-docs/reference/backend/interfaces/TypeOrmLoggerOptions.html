import { Logger, QueryRunner } from 'typeorm';
import * as winston from 'winston';
import 'winston-daily-rotate-file';
import * as path from 'path';
import * as fs from 'fs';
import { LoggingService } from './logging.service';

/**
 * Configuration options for the TypeORM logger.
 */
export interface TypeOrmLoggerOptions {
  /** Whether to write SQL queries to a dedicated file. Default: true in dev, false in prod. */
  sqlFileEnabled: boolean;
  /** Directory for the SQL log file. Default: './logs'. */
  sqlFileDir: string;
  /** Max file size before rotation. Default: '20m'. */
  sqlFileMaxSize: string;
  /** Max days to keep rotated files. Default: '14d'. */
  sqlFileMaxDays: string;
}

const DEFAULT_OPTIONS: TypeOrmLoggerOptions = {
  sqlFileEnabled: false,
  sqlFileDir: './logs',
  sqlFileMaxSize: '20m',
  sqlFileMaxDays: '14d',
};

/**
 * Custom TypeORM Logger that integrates with the application's LoggingService.
 *
 * All logs are routed through LoggingService to benefit from:
 * - Structured JSON logging
 * - Correlation ID propagation
 * - Sensitive field redaction
 * - Console output
 *
 * Additionally, SQL queries are written to a separate daily-rotated file
 * (`logs/sql-YYYY-MM-DD.log`) for easy inspection.
 */
export class TypeOrmLoggerService implements Logger {
  private readonly sqlFileLogger: winston.Logger | null = null;

  constructor(
    private readonly loggingService: LoggingService,
    private readonly options: TypeOrmLoggerOptions = DEFAULT_OPTIONS,
  ) {
    if (options.sqlFileEnabled) {
      this.sqlFileLogger = this.createSqlFileLogger();
    }
  }

  // ---------------------------------------------------------------------------
  // TypeORM Logger interface
  // ---------------------------------------------------------------------------

  logQuery(
    query: string,
    parameters?: any[],
    _queryRunner?: QueryRunner,
  ): void {
    const child = this.loggingService.child('TypeORM');
    child.debug(query, { parameters, type: 'sql-query' });
    this.writeSqlFile('query', query, parameters);
  }

  logQueryError(
    error: string | Error,
    query: string,
    parameters?: any[],
    _queryRunner?: QueryRunner,
  ): void {
    const child = this.loggingService.child('TypeORM');
    const errorMessage = error instanceof Error ? error.message : error;
    const errorStack = error instanceof Error ? error.stack : undefined;
    child.error(errorMessage, {
      query,
      parameters,
      stack: errorStack,
      type: 'sql-error',
    });
    this.writeSqlFile('error', query, parameters, errorMessage);
  }

  logQuerySlow(
    time: number,
    query: string,
    parameters?: any[],
    _queryRunner?: QueryRunner,
  ): void {
    const child = this.loggingService.child('TypeORM');
    child.warn(`Slow query (${time}ms)`, {
      time,
      query,
      parameters,
      type: 'sql-slow',
    });
    this.writeSqlFile('slow', query, parameters, undefined, time);
  }

  logSchemaBuild(message: string, _queryRunner?: QueryRunner): void {
    const child = this.loggingService.child('TypeORM');
    child.info(message, { type: 'schema-build' });
    this.writeSqlFile('schema', message);
  }

  logMigration(message: string, _queryRunner?: QueryRunner): void {
    const child = this.loggingService.child('TypeORM');
    child.info(message, { type: 'migration' });
    this.writeSqlFile('migration', message);
  }

  log(
    level: 'log' | 'info' | 'warn',
    message: any,
    _queryRunner?: QueryRunner,
  ): void {
    const child = this.loggingService.child('TypeORM');
    const msg = typeof message === 'string' ? message : JSON.stringify(message);
    switch (level) {
      case 'warn':
        child.warn(msg);
        break;
      default:
        child.info(msg);
        break;
    }
  }

  // ---------------------------------------------------------------------------
  // Private helpers
  // ---------------------------------------------------------------------------

  /**
   * Writes a log entry to the dedicated SQL file logger (if enabled).
   */
  private writeSqlFile(
    type: string,
    message: string,
    parameters?: any[],
    error?: string,
    time?: number,
  ): void {
    if (!this.sqlFileLogger) return;

    const entry: Record<string, unknown> = {
      timestamp: new Date().toISOString(),
      type,
      message,
    };

    // Attach correlationId from the request-scoped AsyncLocalStorage so SQL
    // log lines can be correlated with the request that triggered them.
    const correlationId =
      LoggingService.getAsyncLocalStorage().getStore()?.correlationId;
    if (correlationId) {
      entry.correlationId = correlationId;
    }

    if (parameters !== undefined && parameters.length > 0) {
      entry.parameters = parameters;
    }
    if (error !== undefined) {
      entry.error = error;
    }
    if (time !== undefined) {
      entry.timeMs = time;
    }

    this.sqlFileLogger.info(entry);
  }

  /**
   * Creates a dedicated Winston logger that writes SQL queries to a
   * daily-rotated file (logs/sql-YYYY-MM-DD.log).
   */
  private createSqlFileLogger(): winston.Logger {
    const dirCreated = this.ensureLogDir(this.options.sqlFileDir);

    if (!dirCreated) {
      // Fallback: return a logger that does nothing
      return winston.createLogger({ silent: true });
    }

    const transport = new winston.transports.DailyRotateFile({
      dirname: this.options.sqlFileDir,
      filename: 'sql-%DATE%.log',
      datePattern: 'YYYY-MM-DD',
      maxSize: this.options.sqlFileMaxSize,
      maxFiles: this.options.sqlFileMaxDays,
      format: winston.format.combine(
        winston.format.timestamp({ format: () => new Date().toISOString() }),
        winston.format.json(),
      ),
      level: 'info',
    });

    transport.on('error', (err: Error) => {
      process.stderr.write(
        `[TypeOrmLogger] SQL file transport error: ${err.message}\n`,
      );
    });

    const logger = winston.createLogger({
      level: 'info',
      transports: [transport],
      exitOnError: false,
    });

    logger.on('error', (err: Error) => {
      process.stderr.write(
        `[TypeOrmLogger] SQL logger error: ${err.message}\n`,
      );
    });

    return logger;
  }

  /**
   * Ensures the log directory exists, creating it recursively if needed.
   */
  private ensureLogDir(dirPath: string): boolean {
    const resolvedDir = path.resolve(dirPath);
    try {
      if (!fs.existsSync(resolvedDir)) {
        fs.mkdirSync(resolvedDir, { recursive: true });
      }
      return true;
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      process.stderr.write(
        `[TypeOrmLogger] Failed to create SQL log directory "${resolvedDir}": ${message}. SQL file logging disabled.\n`,
      );
      return false;
    }
  }
}
