import { CachePayloadError } from './cache-payload.error';

export interface ConversationInfo {
  id: string; // Snowflake bigint as string
  name: string;
  type: string;
  avatarFileKey: string | null;
  ownerId: string; // Snowflake bigint as string
  lastMessageId: string | null;
  lastMessageAt: string | null; // ISO-8601 or null
  createdAt: string; // ISO-8601
}

function assertField(
  value: unknown,
  typeName: string,
  fieldPath: string,
  check: (v: unknown) => boolean,
): void {
  if (!check(value)) {
    throw new CachePayloadError(
      typeName,
      fieldPath,
      `Expected valid value at "${fieldPath}" in ${typeName}`,
    );
  }
}

export const ConversationInfoSerializer = {
  serialize(data: ConversationInfo): string {
    return JSON.stringify(data);
  },

  deserialize(raw: string): ConversationInfo {
    let parsed: unknown;
    try {
      parsed = JSON.parse(raw);
    } catch {
      throw new CachePayloadError('ConversationInfo', '$', 'Malformed JSON');
    }

    if (!parsed || typeof parsed !== 'object') {
      throw new CachePayloadError('ConversationInfo', '$', 'Not an object');
    }

    const obj = parsed as Record<string, unknown>;

    // Required string fields
    for (const field of ['id', 'name', 'type', 'ownerId', 'createdAt']) {
      assertField(
        obj[field],
        'ConversationInfo',
        field,
        (v) => typeof v === 'string' && v.length > 0,
      );
    }

    // Nullable string fields
    for (const field of ['avatarFileKey', 'lastMessageId', 'lastMessageAt']) {
      if (obj[field] !== null && obj[field] !== undefined) {
        assertField(obj[field], 'ConversationInfo', field, (v) => typeof v === 'string');
      }
    }

    return {
      id: obj.id as string,
      name: obj.name as string,
      type: obj.type as string,
      avatarFileKey: (obj.avatarFileKey as string) ?? null,
      ownerId: obj.ownerId as string,
      lastMessageId: (obj.lastMessageId as string) ?? null,
      lastMessageAt: (obj.lastMessageAt as string) ?? null,
      createdAt: obj.createdAt as string,
    };
  },
};
