import {
  Injectable,
  OnModuleInit,
  OnApplicationShutdown,
} from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import * as winston from 'winston';
import 'winston-daily-rotate-file';
import { AsyncLocalStorage } from 'async_hooks';
import * as fs from 'fs';
import * as path from 'path';
import { SensitiveFieldRedactor } from './sensitive-field-redactor';
import { LoggingConfigType } from './logging.config';

export interface LogMetadata {
  [key: string]: unknown;
}

export interface CorrelationStore {
  correlationId: string;
}

// Standard fields that metadata must NOT override
const STANDARD_FIELDS = new Set([
  'timestamp',
  'level',
  'context',
  'message',
  'service',
  'env',
]);

const MAX_MESSAGE_LENGTH = 10000;
const TRUNCATION_SUFFIX = '...[truncated]';

// Custom log levels (lower number = higher priority)
const CUSTOM_LEVELS: winston.config.AbstractConfigSetLevels = {
  error: 0,
  warn: 1,
  info: 2,
  http: 3,
  debug: 4,
  verbose: 5,
};

const CUSTOM_COLORS: Record<string, string> = {
  error: 'red',
  warn: 'yellow',
  info: 'green',
  http: 'magenta',
  debug: 'blue',
  verbose: 'cyan',
};

/**
 * Safe JSON replacer that handles circular references.
 */
function safeReplacer() {
  const seen = new WeakSet();
  return (_key: string, value: unknown): unknown => {
    if (typeof value === 'object' && value !== null) {
      if (seen.has(value)) {
        return '[Circular]';
      }
      seen.add(value);
    }
    return value;
  };
}

@Injectable()
export class LoggingService implements OnModuleInit, OnApplicationShutdown {
  private logger: winston.Logger;
  private context: string = 'Application';

  /** Shared AsyncLocalStorage for correlation ID propagation across the request lifecycle. */
  private static als = new AsyncLocalStorage<CorrelationStore>();

  constructor(
    private readonly configService: ConfigService,
    private readonly redactor: SensitiveFieldRedactor,
  ) {
    this.initLogger();
  }

  // ---------------------------------------------------------------------------
  // Static helpers
  // ---------------------------------------------------------------------------

  static getAsyncLocalStorage(): AsyncLocalStorage<CorrelationStore> {
    return LoggingService.als;
  }

  // ---------------------------------------------------------------------------
  // Lifecycle hooks
  // ---------------------------------------------------------------------------

  onModuleInit(): void {
    // Register global process-level error handlers so unhandled errors are
    // captured and logged before the process potentially exits.
    process.on('uncaughtException', (err: Error) => {
      this.error('Uncaught exception', {
        name: err.name,
        message: err.message,
        stack: err.stack,
      });
    });

    process.on('unhandledRejection', (reason: unknown) => {
      const err = reason instanceof Error ? reason : new Error(String(reason));
      this.error('Unhandled promise rejection', {
        name: err.name,
        message: err.message,
        stack: err.stack,
      });
    });
  }

  async onApplicationShutdown(): Promise<void> {
    await this.flush();
  }

  // ---------------------------------------------------------------------------
  // Public log methods
  // ---------------------------------------------------------------------------

  error(message: string, metadata?: LogMetadata): void {
    this.log('error', message, metadata);
  }

  warn(message: string, metadata?: LogMetadata): void {
    this.log('warn', message, metadata);
  }

  info(message: string, metadata?: LogMetadata): void {
    this.log('info', message, metadata);
  }

  http(message: string, metadata?: LogMetadata): void {
    this.log('http', message, metadata);
  }

  debug(message: string, metadata?: LogMetadata): void {
    this.log('debug', message, metadata);
  }

  verbose(message: string, metadata?: LogMetadata): void {
    this.log('verbose', message, metadata);
  }

  // ---------------------------------------------------------------------------
  // Context management
  // ---------------------------------------------------------------------------

  /**
   * Sets the context label for this LoggingService instance.
   * Mutates the current instance.
   */
  setContext(context: string): void {
    this.context = context;
  }

  /**
   * Returns a NEW LoggingService instance with the given context.
   * The new instance shares the same underlying Winston logger — it does NOT
   * create a new Winston logger or mutate the parent instance.
   */
  child(context: string): LoggingService {
    const childService = new LoggingService(
      this.configService,
      this.redactor,
    );
    // Replace the child's freshly-created logger with the parent's shared one
    childService.logger = this.logger;
    childService.context = context;
    return childService;
  }

  // ---------------------------------------------------------------------------
  // Flush
  // ---------------------------------------------------------------------------

  /**
   * Flushes all transports that support it.
   * Resolves once all transports have finished writing buffered records.
   */
  flush(): Promise<void> {
    return new Promise<void>((resolve) => {
      let resolved = false;
      const done = () => {
        if (!resolved) {
          resolved = true;
          resolve();
        }
      };

      // If there are no transports, resolve immediately
      if (!this.logger.transports || this.logger.transports.length === 0) {
        return resolve();
      }

      // Give transports up to 3 seconds to flush, then resolve anyway
      const timeout = setTimeout(done, 3000);

      this.logger.once('finish', () => {
        clearTimeout(timeout);
        done();
      });

      // Trigger drain by ending the logger stream
      this.logger.end();
    });
  }

  // ---------------------------------------------------------------------------
  // Private helpers
  // ---------------------------------------------------------------------------

  private log(level: string, message: string, metadata?: LogMetadata): void {
    // 1. Truncate message if too long
    const truncatedMessage =
      message.length > MAX_MESSAGE_LENGTH
        ? message.slice(0, MAX_MESSAGE_LENGTH) + TRUNCATION_SUFFIX
        : message;

    // 2. Redact sensitive fields from metadata
    const redactedMetadata = metadata
      ? (this.redactor.redact(metadata) as LogMetadata)
      : undefined;

    // 3. Build the extra fields to merge into the log record.
    //    Standard fields must NOT be overridden by metadata.
    const safeExtra: Record<string, unknown> = {};
    if (redactedMetadata) {
      for (const [key, value] of Object.entries(redactedMetadata)) {
        if (!STANDARD_FIELDS.has(key)) {
          safeExtra[key] = value;
        }
      }
    }

    // 4. Attach correlationId from AsyncLocalStorage if available
    const store = LoggingService.als.getStore();
    if (store?.correlationId) {
      safeExtra['correlationId'] = store.correlationId;
    }

    // 5. Serialize any circular references in safeExtra
    const serializedExtra = this.serializeSafe(safeExtra);

    // 6. Emit to Winston
    this.logger.log(level, truncatedMessage, {
      context: this.context,
      ...serializedExtra,
    });
  }

  /**
   * Serializes an object safely, replacing circular references with '[Circular]'.
   * Returns a plain object (parsed back from JSON) so Winston can handle it.
   */
  private serializeSafe(obj: Record<string, unknown>): Record<string, unknown> {
    try {
      return JSON.parse(JSON.stringify(obj, safeReplacer()));
    } catch {
      // Fallback: return empty object if serialization fails entirely
      return {};
    }
  }

  // ---------------------------------------------------------------------------
  // Winston initialization
  // ---------------------------------------------------------------------------

  private initLogger(): void {
    const config = this.configService.get<LoggingConfigType>('logging');

    const level = config?.level ?? 'info';
    const appEnv = config?.appEnv ?? 'development';
    const consoleEnabled = config?.consoleEnabled ?? true;
    const fileEnabled = config?.fileEnabled ?? false;
    const fileDir = config?.fileDir ?? './logs';
    const fileMaxSize = config?.fileMaxSize ?? '20m';
    const fileMaxFiles = config?.fileMaxDays ?? '14d';
    const serviceName = config?.serviceName ?? 'nestjs-base-v2';

    // Register custom colors with Winston
    winston.addColors(CUSTOM_COLORS);

    // Determine if we should use JSON format (production/staging) or colorized text (development)
    const isJsonFormat = appEnv !== 'development';

    // Base format: always include timestamp and standard fields
    const baseFormat = winston.format.combine(
      winston.format.timestamp({ format: () => new Date().toISOString() }),
      winston.format((info) => {
        // Inject service and env into every record
        info['service'] = serviceName;
        info['env'] = appEnv;
        return info;
      })(),
    );

    // Production/staging: JSON one-liner
    const jsonFormat = winston.format.combine(
      baseFormat,
      winston.format.json(),
    );

    // Development: colorized, human-readable text
    const devFormat = winston.format.combine(
      baseFormat,
      winston.format.colorize({ all: true }),
      winston.format.printf((info) => {
        const { timestamp, level, context, message, ...rest } = info;
        const contextStr = context ? `[${context}]` : '';
        const extras = Object.keys(rest).length
          ? ' ' + JSON.stringify(rest, safeReplacer())
          : '';
        return `${timestamp} ${level} ${contextStr} ${message}${extras}`;
      }),
    );

    const transports: winston.transport[] = [];

    // Console transport
    if (consoleEnabled) {
      const consoleTransport = new winston.transports.Console({
        format: isJsonFormat ? jsonFormat : devFormat,
      });
      consoleTransport.on('error', (err: Error) => {
        process.stderr.write(
          `[LoggingService] Console transport error: ${err.message}\n`,
        );
      });
      transports.push(consoleTransport);
    }

    // File transports
    if (fileEnabled) {
      const dirCreated = this.ensureLogDir(fileDir);
      if (dirCreated) {
        // All-levels rotating file transport
        const allLevelsTransport = new winston.transports.DailyRotateFile({
          dirname: fileDir,
          filename: 'app-%DATE%.log',
          datePattern: 'YYYY-MM-DD',
          maxSize: fileMaxSize,
          maxFiles: fileMaxFiles,
          format: jsonFormat,
          level: level,
        });
        allLevelsTransport.on('error', (err: Error) => {
          process.stderr.write(
            `[LoggingService] File transport error: ${err.message}\n`,
          );
        });
        transports.push(allLevelsTransport);

        // Error-only rotating file transport
        const errorTransport = new winston.transports.DailyRotateFile({
          dirname: fileDir,
          filename: 'app-error-%DATE%.log',
          datePattern: 'YYYY-MM-DD',
          maxSize: fileMaxSize,
          maxFiles: fileMaxFiles,
          format: jsonFormat,
          level: 'error',
        });
        errorTransport.on('error', (err: Error) => {
          process.stderr.write(
            `[LoggingService] Error file transport error: ${err.message}\n`,
          );
        });
        transports.push(errorTransport);
      }
    }

    this.logger = winston.createLogger({
      level,
      levels: CUSTOM_LEVELS,
      transports,
      exitOnError: false,
    });

    // Global transport error handler on the logger itself
    this.logger.on('error', (err: Error) => {
      process.stderr.write(
        `[LoggingService] Transport error: ${err.message}\n`,
      );
    });
  }

  /**
   * Ensures the log directory exists, creating it recursively if needed.
   * Returns true if the directory is ready to use, false if creation failed.
   */
  private ensureLogDir(dirPath: string): boolean {
    const resolvedDir = path.resolve(dirPath);
    try {
      if (!fs.existsSync(resolvedDir)) {
        fs.mkdirSync(resolvedDir, { recursive: true });
      }
      return true;
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      process.stderr.write(
        `[LoggingService] Failed to create log directory "${resolvedDir}": ${message}. File transport disabled.\n`,
      );
      return false;
    }
  }
}
