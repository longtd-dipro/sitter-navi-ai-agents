import { CacheClient, parseIfRequired } from '@type-cacheable/core';

interface CacheEntry {
  value: string;
  expiresAt: number; // 0 = no expiry
}

/**
 * In-memory cache adapter for @type-cacheable/core.
 *
 * Used as a fallback when Redis is unavailable. Provides basic TTL support
 * and LRU-style eviction when maxEntries is reached.
 *
 * Limitations vs Redis:
 * - Not shared across instances (single-node only)
 * - Lost on restart
 * - No pattern-based key scanning (delHash uses prefix match on local keys)
 */
export class InMemoryCacheAdapter implements CacheClient {
  private cache = new Map<string, CacheEntry>();
  private readonly maxEntries: number;
  private cleanupTimer: NodeJS.Timeout;

  constructor(options?: { maxEntries?: number; cleanupIntervalMs?: number }) {
    this.maxEntries = options?.maxEntries ?? 10000;
    const cleanupInterval = options?.cleanupIntervalMs ?? 60000;

    this.cleanupTimer = setInterval(() => this.cleanup(), cleanupInterval);
    this.cleanupTimer.unref();
  }

  async get(cacheKey: string): Promise<any> {
    const entry = this.cache.get(cacheKey);
    if (!entry) return null;

    if (entry.expiresAt > 0 && Date.now() > entry.expiresAt) {
      this.cache.delete(cacheKey);
      return null;
    }

    return parseIfRequired(entry.value);
  }

  async set(cacheKey: string, value: any, ttl?: number): Promise<any> {
    // Evict oldest entries if at capacity
    if (this.cache.size >= this.maxEntries) {
      const firstKey = this.cache.keys().next().value;
      if (firstKey) this.cache.delete(firstKey);
    }

    const expiresAt = ttl && ttl > 0 ? Date.now() + ttl * 1000 : 0;
    this.cache.set(cacheKey, {
      value: JSON.stringify(value),
      expiresAt,
    });
  }

  getClientTTL(): number {
    return 0;
  }

  async del(keyOrKeys: string | string[]): Promise<any> {
    const keys = Array.isArray(keyOrKeys) ? keyOrKeys : [keyOrKeys];
    let count = 0;
    for (const key of keys) {
      if (this.cache.delete(key)) count++;
    }
    return count;
  }

  async keys(pattern: string): Promise<string[]> {
    // Convert glob-like pattern to regex (simple * wildcard support)
    const regexStr = pattern.replace(/[.+^${}()|[\]\\]/g, '\\$&').replace(/\*/g, '.*');
    const regex = new RegExp(`^${regexStr}$`);

    const matched: string[] = [];
    for (const key of this.cache.keys()) {
      if (regex.test(key)) matched.push(key);
    }
    return matched;
  }

  async delHash(hashKeyOrKeys: string | string[]): Promise<any> {
    const patterns = Array.isArray(hashKeyOrKeys) ? hashKeyOrKeys : [hashKeyOrKeys];
    for (const pattern of patterns) {
      const matchedKeys = await this.keys(`*${pattern}*`);
      await this.del(matchedKeys);
    }
  }

  /**
   * Clear all entries.
   */
  clear(): void {
    this.cache.clear();
  }

  /**
   * Destroy the adapter (stop cleanup timer).
   */
  destroy(): void {
    clearInterval(this.cleanupTimer);
    this.cache.clear();
  }

  /**
   * Current number of entries in cache.
   */
  get size(): number {
    return this.cache.size;
  }

  private cleanup(): void {
    const now = Date.now();
    for (const [key, entry] of this.cache.entries()) {
      if (entry.expiresAt > 0 && entry.expiresAt <= now) {
        this.cache.delete(key);
      }
    }
  }
}
