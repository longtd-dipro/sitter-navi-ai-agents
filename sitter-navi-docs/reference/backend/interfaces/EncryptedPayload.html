import { createCipheriv, createDecipheriv, randomBytes, scryptSync } from 'crypto';

const ALGORITHM = 'aes-256-gcm';
const IV_LENGTH = 16;
const TAG_LENGTH = 16;
const KEY_LENGTH = 32;
const SALT_LENGTH = 16;

export interface EncryptedPayload {
  /** Base64-encoded encrypted data */
  encrypted: string;
  /** Base64-encoded initialization vector */
  iv: string;
  /** Base64-encoded auth tag */
  tag: string;
  /** Base64-encoded salt (if password-based) */
  salt?: string;
}

/**
 * Generate a random 256-bit encryption key.
 */
export function generateKey(): string {
  return randomBytes(KEY_LENGTH).toString('hex');
}

/**
 * Derive a 256-bit key from a password using scrypt.
 */
export function deriveKey(password: string, salt: Buffer): Buffer {
  return scryptSync(password, salt, KEY_LENGTH);
}

/**
 * Encrypt plaintext using AES-256-GCM with a hex key.
 */
export function aesEncrypt(plaintext: string, hexKey: string): EncryptedPayload {
  const key = Buffer.from(hexKey, 'hex');
  const iv = randomBytes(IV_LENGTH);

  const cipher = createCipheriv(ALGORITHM, key, iv);
  const encrypted = Buffer.concat([cipher.update(plaintext, 'utf-8'), cipher.final()]);
  const tag = cipher.getAuthTag();

  return {
    encrypted: encrypted.toString('base64'),
    iv: iv.toString('base64'),
    tag: tag.toString('base64'),
  };
}

/**
 * Decrypt ciphertext using AES-256-GCM with a hex key.
 */
export function aesDecrypt(payload: EncryptedPayload, hexKey: string): string {
  const key = Buffer.from(hexKey, 'hex');
  const iv = Buffer.from(payload.iv, 'base64');
  const encrypted = Buffer.from(payload.encrypted, 'base64');
  const tag = Buffer.from(payload.tag, 'base64');

  const decipher = createDecipheriv(ALGORITHM, key, iv);
  decipher.setAuthTag(tag);

  const decrypted = Buffer.concat([decipher.update(encrypted), decipher.final()]);
  return decrypted.toString('utf-8');
}

/**
 * Encrypt plaintext using a password (derives key via scrypt).
 * Includes salt in the payload for decryption.
 */
export function aesEncryptWithPassword(plaintext: string, password: string): EncryptedPayload {
  const salt = randomBytes(SALT_LENGTH);
  const key = deriveKey(password, salt);
  const iv = randomBytes(IV_LENGTH);

  const cipher = createCipheriv(ALGORITHM, key, iv);
  const encrypted = Buffer.concat([cipher.update(plaintext, 'utf-8'), cipher.final()]);
  const tag = cipher.getAuthTag();

  return {
    encrypted: encrypted.toString('base64'),
    iv: iv.toString('base64'),
    tag: tag.toString('base64'),
    salt: salt.toString('base64'),
  };
}

/**
 * Decrypt ciphertext using a password (derives key via scrypt).
 */
export function aesDecryptWithPassword(payload: EncryptedPayload, password: string): string {
  if (!payload.salt) {
    throw new Error('Salt is required for password-based decryption');
  }

  const salt = Buffer.from(payload.salt, 'base64');
  const key = deriveKey(password, salt);
  const iv = Buffer.from(payload.iv, 'base64');
  const encrypted = Buffer.from(payload.encrypted, 'base64');
  const tag = Buffer.from(payload.tag, 'base64');

  const decipher = createDecipheriv(ALGORITHM, key, iv);
  decipher.setAuthTag(tag);

  const decrypted = Buffer.concat([decipher.update(encrypted), decipher.final()]);
  return decrypted.toString('utf-8');
}
