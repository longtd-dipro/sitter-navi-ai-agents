import { Logger } from '@nestjs/common';
import { LockAdapter, LockOptions } from './interfaces';
import { LockNotAcquiredError } from './errors';

export interface LockManagerConfig {
  /**
   * Ordered list of adapters. The first healthy adapter wins.
   * If the primary fails, it falls back to the next one.
   */
  adapters: LockAdapter[];

  /**
   * Default TTL for locks in milliseconds.
   * @default 30000
   */
  defaultTtlMs?: number;

  /**
   * Default behavior when lock cannot be acquired.
   * @default 'skip'
   */
  defaultOnFail?: 'skip' | 'throw';
}

/**
 * LockManager orchestrates distributed lock acquisition across multiple
 * adapters with automatic fallback.
 *
 * Usage:
 *   const manager = new LockManager({
 *     adapters: [redisAdapter, postgresAdapter],
 *   });
 *
 *   // The manager tries Redis first; if Redis is down, falls back to PG.
 *   const released = await manager.withLock('my-job', async () => {
 *     // critical section
 *   });
 */
export class LockManager {
  private readonly logger = new Logger('LockManager');
  private readonly adapters: LockAdapter[];
  private readonly defaultTtlMs: number;
  private readonly defaultOnFail: 'skip' | 'throw';

  /** Track which adapter holds which lock for correct release */
  private lockOwnership = new Map<string, LockAdapter>();

  constructor(config: LockManagerConfig) {
    this.adapters = config.adapters;
    this.defaultTtlMs = config.defaultTtlMs ?? 30000;
    this.defaultOnFail = config.defaultOnFail ?? 'skip';

    if (this.adapters.length === 0) {
      throw new Error('LockManager requires at least one adapter');
    }
  }

  /**
   * Acquire a lock, execute the callback, then release.
   * Returns the callback result, or undefined if lock was not acquired (onFail: 'skip').
   */
  async withLock<T>(
    key: string,
    fn: () => Promise<T>,
    options?: Partial<LockOptions>,
  ): Promise<T | undefined> {
    const opts = this.resolveOptions(key, options);
    const acquired = await this.acquire(opts);

    if (!acquired) {
      if (opts.onFail === 'throw') {
        throw new LockNotAcquiredError(opts.key);
      }
      this.logger.warn(`Lock not acquired, skipping: ${opts.key}`);
      return undefined;
    }

    try {
      return await fn();
    } finally {
      await this.release(opts.key);
    }
  }

  /**
   * Acquire a lock using the first healthy adapter.
   * Falls back to subsequent adapters if the primary is unavailable.
   */
  async acquire(options: LockOptions): Promise<boolean> {
    const { key, ttlMs, waitMs, retryIntervalMs } = options;

    // Non-blocking mode (default)
    if (!waitMs || waitMs <= 0) {
      return this.tryAcquireOnce(key, ttlMs);
    }

    // Blocking mode with retry
    const deadline = Date.now() + waitMs;
    const interval = retryIntervalMs ?? 100;

    while (Date.now() < deadline) {
      const acquired = await this.tryAcquireOnce(key, ttlMs);
      if (acquired) return true;
      await this.sleep(Math.min(interval, deadline - Date.now()));
    }

    return false;
  }

  /**
   * Release a previously acquired lock.
   */
  async release(key: string): Promise<boolean> {
    const adapter = this.lockOwnership.get(key);
    if (!adapter) {
      this.logger.warn(`No lock ownership found for key: ${key}`);
      return false;
    }

    const released = await adapter.release(key);
    this.lockOwnership.delete(key);
    return released;
  }

  /**
   * Gracefully shut down all adapters.
   */
  async shutdown(): Promise<void> {
    for (const adapter of this.adapters) {
      await adapter.shutdown().catch((err) => {
        this.logger.error(`Error shutting down adapter ${adapter.name}`, (err as Error).stack);
      });
    }
    this.lockOwnership.clear();
  }

  // ---------------------------------------------------------------------------
  // Private
  // ---------------------------------------------------------------------------

  private async tryAcquireOnce(key: string, ttlMs?: number): Promise<boolean> {
    for (const adapter of this.adapters) {
      const healthy = await adapter.isHealthy().catch(() => false);
      if (!healthy) {
        this.logger.debug(`Adapter "${adapter.name}" unhealthy, trying next...`);
        continue;
      }

      const acquired = await adapter.acquire(key, ttlMs).catch((err) => {
        this.logger.error(
          `Adapter "${adapter.name}" acquire error for key "${key}"`,
          (err as Error).stack,
        );
        return false;
      });

      if (acquired) {
        this.lockOwnership.set(key, adapter);
        this.logger.debug(`Lock acquired via "${adapter.name}": ${key}`);
        return true;
      }

      // Lock is held by someone else on this adapter — don't fallback
      // (the resource is genuinely locked, not an adapter failure)
      return false;
    }

    // All adapters unhealthy
    this.logger.error(`All lock adapters unavailable for key: ${key}`);
    return false;
  }

  private resolveOptions(key: string, partial?: Partial<LockOptions>): LockOptions {
    return {
      key,
      ttlMs: partial?.ttlMs ?? this.defaultTtlMs,
      waitMs: partial?.waitMs ?? 0,
      retryIntervalMs: partial?.retryIntervalMs ?? 100,
      onFail: partial?.onFail ?? this.defaultOnFail,
    };
  }

  private sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}
