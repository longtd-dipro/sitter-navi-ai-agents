import { Injectable, Logger } from '@nestjs/common';
import { Interval } from '@nestjs/schedule';
import { LoggingService } from '@infrastructure/logging';

interface MemorySnapshot {
  timestamp: number;
  heapUsed: number;
  heapTotal: number;
  heapRatio: number;
  rss: number;
  external: number;
}

/**
 * Number of consecutive increasing samples to trigger a leak warning.
 * Memory naturally fluctuates — 1-2 increases are normal.
 * 5 consecutive increases = likely leak.
 */
const LEAK_DETECTION_WINDOW = 5;

/**
 * RSS increase ratio threshold (current / oldest in window).
 * If RSS has grown >20% across the detection window → potential leak.
 */
const LEAK_RSS_GROWTH_THRESHOLD = 1.2;

/**
 * Absolute RSS warning threshold (bytes).
 * Logs WARNING regardless of trend if RSS exceeds this.
 * Default: 400MB — early warning before the 450MB health check threshold.
 */
const RSS_WARN_THRESHOLD = 400 * 1024 * 1024;

@Injectable()
export class MemoryMonitorService {
  private readonly logger = new Logger(MemoryMonitorService.name);

  /** Sliding window of recent memory snapshots for trend detection */
  private readonly history: MemorySnapshot[] = [];
  private readonly maxHistory = LEAK_DETECTION_WINDOW + 5;

  /** Track consecutive growth samples */
  private consecutiveGrowth = 0;

  constructor(private readonly loggingService: LoggingService) {}

  // ──────────────────────────────────────────────────────────────────────────
  // Periodic sampling — every 30 seconds
  // ──────────────────────────────────────────────────────────────────────────

  @Interval(30_000)
  sampleMemory(): void {
    const snap = this.takeSnapshot();
    this.history.push(snap);

    // Keep history bounded
    if (this.history.length > this.maxHistory) {
      this.history.shift();
    }

    // Layer 1: Always log structured memory data (survives OOMKill)
    this.loggingService.info('Memory snapshot', {
      category: 'memory-monitor',
      heapUsedMB: this.toMB(snap.heapUsed),
      heapTotalMB: this.toMB(snap.heapTotal),
      heapUsagePercent: (snap.heapRatio * 100).toFixed(1),
      rssMB: this.toMB(snap.rss),
      externalMB: this.toMB(snap.external),
      consecutiveGrowth: this.consecutiveGrowth,
    });

    // Layer 2: Leak detection via trend analysis
    this.detectLeak(snap);

    // Layer 3: Absolute RSS threshold warning
    if (snap.rss > RSS_WARN_THRESHOLD) {
      this.loggingService.warn(
        `RSS (${this.toMB(snap.rss)} MB) exceeds warning threshold (${this.toMB(RSS_WARN_THRESHOLD)} MB)`,
        {
          category: 'memory-monitor',
          rssMB: this.toMB(snap.rss),
          rssWarnThresholdMB: this.toMB(RSS_WARN_THRESHOLD),
          heapUsagePercent: (snap.heapRatio * 100).toFixed(1),
        },
      );
    }
  }

  // ──────────────────────────────────────────────────────────────────────────
  // Public API — get recent history (exposed for health check / debugging)
  // ──────────────────────────────────────────────────────────────────────────

  /** Get the N most recent snapshots for trend analysis */
  getRecentHistory(count = LEAK_DETECTION_WINDOW): MemorySnapshot[] {
    return this.history.slice(-count);
  }

  /** Get current consecutive growth count */
  getConsecutiveGrowth(): number {
    return this.consecutiveGrowth;
  }

  // ──────────────────────────────────────────────────────────────────────────
  // Private helpers
  // ──────────────────────────────────────────────────────────────────────────

  private takeSnapshot(): MemorySnapshot {
    const mem = process.memoryUsage();
    return {
      timestamp: Date.now(),
      heapUsed: mem.heapUsed,
      heapTotal: mem.heapTotal,
      heapRatio: mem.heapUsed / mem.heapTotal,
      rss: mem.rss,
      external: mem.external,
    };
  }

  /**
   * Detect memory leak via trend: if RSS grows across N consecutive samples,
   * and the total growth exceeds threshold, log a WARNING.
   */
  private detectLeak(current: MemorySnapshot): void {
    const history = this.history;

    // Need at least 2 samples to detect growth
    if (history.length < 2) return;

    const previous = history[history.length - 2];

    // Track consecutive growth
    if (current.rss > previous.rss) {
      this.consecutiveGrowth++;
    } else {
      // Memory decreased (GC ran) — reset counter
      this.consecutiveGrowth = 0;
    }

    // Only trigger when we have enough consecutive growth
    if (this.consecutiveGrowth < LEAK_DETECTION_WINDOW) return;

    // Calculate growth ratio: oldest in window vs current
    const windowStart = history[history.length - LEAK_DETECTION_WINDOW];
    const growthRatio = current.rss / windowStart.rss;

    if (growthRatio >= LEAK_RSS_GROWTH_THRESHOLD) {
      this.loggingService.warn(
        `Potential memory leak detected: RSS grew ${((growthRatio - 1) * 100).toFixed(1)}% over ${LEAK_DETECTION_WINDOW} samples (${((LEAK_DETECTION_WINDOW * 30) / 60).toFixed(1)} min)`,
        {
          category: 'memory-monitor',
          alert: 'memory-leak-suspected',
          consecutiveGrowth: this.consecutiveGrowth,
          rssGrowthPercent: `${((growthRatio - 1) * 100).toFixed(1)}%`,
          rssMB: this.toMB(current.rss),
          rssStartMB: this.toMB(windowStart.rss),
          windowSamples: LEAK_DETECTION_WINDOW,
          windowMinutes: (LEAK_DETECTION_WINDOW * 30) / 60,
          heapUsagePercent: (current.heapRatio * 100).toFixed(1),
        },
      );

      // Reset counter after logging to avoid spam
      this.consecutiveGrowth = 0;
    }
  }

  private toMB(bytes: number): string {
    return (bytes / (1024 * 1024)).toFixed(1);
  }
}
