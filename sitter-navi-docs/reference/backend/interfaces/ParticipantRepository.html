import { ParticipantDto } from './dtos';
import { CreateParticipantInput, RestoreParticipantInput } from './inputs';
import { CursorPaginationInput, CursorPaginatedResult } from './pagination';

/**
 * Repository contract for the Participant aggregate.
 * All methods use only plain TypeScript types — no TypeORM types exposed.
 */
export interface ParticipantRepository {
  findByGroupAndUser(groupId: string, userId: string): Promise<ParticipantDto | null>;

  findActiveByGroup(groupId: string): Promise<ParticipantDto[]>;

  findActiveByUser(userId: string): Promise<ParticipantDto[]>;

  countActiveByGroup(groupId: string): Promise<number>;

  create(data: CreateParticipantInput): Promise<ParticipantDto>;

  createMany(data: CreateParticipantInput[]): Promise<ParticipantDto[]>;

  softDelete(groupId: string, userId: string): Promise<void>;

  restore(groupId: string, userId: string, data: RestoreParticipantInput): Promise<void>;

  updateRole(groupId: string, userId: string, role: string): Promise<void>;

  /** Increment unread_count for all active participants except the given user */
  incrementUnreadExcept(groupId: string, excludeUserId: string): Promise<void>;

  /** Reset unread_count to 0 for a specific user in a group */
  resetUnread(groupId: string, userId: string): Promise<void>;

  findSoftDeleted(groupId: string, userId: string): Promise<ParticipantDto | null>;

  findPaginated(
    groupId: string,
    pagination: CursorPaginationInput,
  ): Promise<CursorPaginatedResult<ParticipantDto>>;
}
