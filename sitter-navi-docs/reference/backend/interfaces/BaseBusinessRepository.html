import { EventEmitter2 } from '@nestjs/event-emitter';
import { WithIdAndTimestamp } from '@entities/base.entity';

/**
 * Minimal structural contract that {@link BaseBusinessService} relies on.
 *
 * The full {@link BaseRepository} contract (with `findById`, `softDelete`,
 * `count`, etc.) is defined separately by task 1.4 in
 * `src/common/repositories/base.repository.ts`. Any concrete repository that
 * implements that fuller contract is structurally assignable to this one,
 * which keeps `BaseBusinessService` decoupled from the repository module.
 */
export interface BaseBusinessRepository<TEntity extends WithIdAndTimestamp> {
  /**
   * Load an entity by id, throwing `NotFoundException` (with i18n key) when
   * the id is not found. Used by {@link BaseBusinessService.update} before
   * applying changes.
   */
  findByIdOrFail(id: string): Promise<TEntity>;

  /**
   * Persist the given (partial) entity and resolve with the saved entity,
   * including database-generated fields (id, timestamps).
   */
  save(data: Partial<TEntity>): Promise<TEntity>;
}

/**
 * Abstract base class for Complex Business Module services.
 *
 * Implements the Template Method pattern for `create` and `update`:
 *
 * - `create(dto)`:
 *   1. `validateCreate(dto)`            — business rule validation
 *   2. `mapToEntity(dto)`               — DTO → entity transformation
 *   3. `repository.save(entity)`        — persist
 *   4. `emitCreatedEvent(saved)`        — emit domain event (only on success)
 *
 * - `update(id, dto)`:
 *   1. `repository.findByIdOrFail(id)`  — load existing
 *   2. `validateUpdate(existing, dto)`  — business rule validation
 *   3. `applyChanges(existing, dto)`    — compute updated state
 *   4. `repository.save(updated)`       — persist
 *   5. `emitUpdatedEvent(before, after)` — emit domain event (only on success)
 *
 * Property 9 (event-after-persist): `emitCreatedEvent` / `emitUpdatedEvent`
 * are awaited *after* `repository.save` resolves. If save throws, the emit
 * step is never reached and no event is published. Subclasses MUST keep this
 * invariant — never emit inside `validateCreate`, `mapToEntity`,
 * `validateUpdate`, or `applyChanges`.
 *
 * @typeParam TEntity     - The aggregate entity (must extend
 *                          {@link WithIdAndTimestamp}).
 * @typeParam TCreateDto  - DTO accepted by `create`.
 * @typeParam TUpdateDto  - DTO accepted by `update`.
 */
export abstract class BaseBusinessService<
  TEntity extends WithIdAndTimestamp,
  TCreateDto,
  TUpdateDto,
> {
  constructor(
    protected readonly repository: BaseBusinessRepository<TEntity>,
    protected readonly eventEmitter: EventEmitter2,
  ) {}

  /**
   * Template method for the create flow.
   *
   * Order of operations is fixed by this base class to guarantee Property 9:
   * the domain event is emitted only after `repository.save` resolves
   * successfully.
   */
  async create(dto: TCreateDto): Promise<TEntity> {
    // Step 1: Validate business rules. Throws → no persist, no emit.
    await this.validateCreate(dto);

    // Step 2: Transform DTO to entity.
    const entity = this.mapToEntity(dto);

    // Step 3: Persist. Throws → no emit (Property 9).
    const saved = await this.repository.save(entity);

    // Step 4: Emit domain event AFTER successful persist.
    await this.emitCreatedEvent(saved);

    return saved;
  }

  /**
   * Template method for the update flow.
   *
   * Same Property 9 guarantee applies: `emitUpdatedEvent` runs only after
   * `repository.save` resolves successfully.
   */
  async update(id: string, dto: TUpdateDto): Promise<TEntity> {
    // Step 1: Load existing. Throws NotFoundException if id is missing.
    const existing = await this.repository.findByIdOrFail(id);

    // Step 2: Validate business rules against existing + dto.
    await this.validateUpdate(existing, dto);

    // Step 3: Compute updated state.
    const updated = this.applyChanges(existing, dto);

    // Step 4: Persist. Throws → no emit (Property 9).
    const saved = await this.repository.save(updated);

    // Step 5: Emit domain event AFTER successful persist.
    await this.emitUpdatedEvent(existing, saved);

    return saved;
  }

  // --- Hooks: subclasses implement module-specific logic. --------------------

  /**
   * Validate the create payload against business rules (uniqueness,
   * capacity, referenced ids existing, etc). Throw a NestJS HTTP exception
   * to abort persistence.
   */
  protected abstract validateCreate(dto: TCreateDto): Promise<void>;

  /**
   * Validate the update payload against the existing entity (state
   * transitions, ownership, etc). Throw a NestJS HTTP exception to abort
   * persistence.
   */
  protected abstract validateUpdate(existing: TEntity, dto: TUpdateDto): Promise<void>;

  /**
   * Map a create DTO to a partial entity ready for `repository.save`.
   * Implementations MUST NOT perform I/O here — keep this pure.
   */
  protected abstract mapToEntity(dto: TCreateDto): Partial<TEntity>;

  /**
   * Compute the next entity state by merging `dto` into `existing`. Should
   * return a partial entity that, when saved, produces the desired update.
   */
  protected abstract applyChanges(existing: TEntity, dto: TUpdateDto): Partial<TEntity>;

  /**
   * Emit the domain event corresponding to a successful create. Called
   * AFTER `repository.save` resolves (Property 9).
   */
  protected abstract emitCreatedEvent(entity: TEntity): Promise<void>;

  /**
   * Emit the domain event corresponding to a successful update. Called
   * AFTER `repository.save` resolves (Property 9). Receives both the
   * pre-update (`before`) and post-update (`after`) snapshots so subclasses
   * can include diff information in the event payload if needed.
   */
  protected abstract emitUpdatedEvent(before: TEntity, after: TEntity): Promise<void>;
}
