import { IoAdapter } from '@nestjs/platform-socket.io';
import { INestApplication, Logger } from '@nestjs/common';
import { createAdapter as createRedisAdapter } from '@socket.io/redis-adapter';
import { createAdapter as createPostgresAdapter } from '@socket.io/postgres-adapter';
import { Redis } from 'ioredis';
import { Pool } from 'pg';
import { ServerOptions } from 'socket.io';
import { getRedisKeyPrefix } from './redis.config';

interface RedisIoAdapterOptions {
  nodeEnv: string;
}

interface RedisConfig {
  host: string;
  port: number;
  username?: string;
  password?: string;
  tls?: boolean;
}

interface WebSocketAdapterConfig {
  redis: RedisConfig;
  databaseUrl?: string;
}

export class RedisIoAdapter extends IoAdapter {
  private adapterConstructor:
    | ReturnType<typeof createRedisAdapter>
    | ReturnType<typeof createPostgresAdapter>
    | undefined;
  private readonly logger = new Logger('RedisIoAdapter');
  private _activeAdapter: 'redis' | 'postgres' | 'in-memory' = 'in-memory';

  constructor(
    app: INestApplication,
    private options: RedisIoAdapterOptions,
  ) {
    super(app);
  }

  /**
   * Returns which adapter backend is currently active.
   */
  get activeAdapter(): string {
    return this._activeAdapter;
  }

  /**
   * Connect to Redis for Socket.IO pub/sub.
   * If Redis is unavailable, falls back to PostgreSQL LISTEN/NOTIFY.
   * If both are unavailable, uses the default in-memory adapter (single-node only).
   */
  async connectToRedis(config: WebSocketAdapterConfig): Promise<void> {
    // Try Redis first
    const redisConnected = await this.tryRedisAdapter(config.redis);
    if (redisConnected) return;

    // Fallback to PostgreSQL
    if (config.databaseUrl) {
      const pgConnected = await this.tryPostgresAdapter(config.databaseUrl);
      if (pgConnected) return;
    }

    // Final fallback: in-memory (default IoAdapter behavior)
    this._activeAdapter = 'in-memory';
    this.logger.warn(
      'Both Redis and PostgreSQL adapters unavailable. WebSocket will use in-memory adapter (single-node only).',
    );
  }

  createIOServer(port: number, options?: Partial<ServerOptions>) {
    const server = super.createIOServer(port, options);

    if (this.adapterConstructor) {
      server.adapter(this.adapterConstructor);
    }

    return server;
  }

  // ───────────────────────────────────────────────────────────────────────────
  // Private
  // ───────────────────────────────────────────────────────────────────────────

  private async tryRedisAdapter(redisConfig: RedisConfig): Promise<boolean> {
    try {
      const redisOptions = {
        host: redisConfig.host,
        port: redisConfig.port,
        ...(redisConfig.username && { username: redisConfig.username }),
        ...(redisConfig.password && { password: redisConfig.password }),
        ...(redisConfig.tls && { tls: {} }),
        keyPrefix: getRedisKeyPrefix(),
        connectTimeout: 5000,
        retryStrategy: () => null, // Don't auto-retry on initial connect
        lazyConnect: true,
      };

      const pubClient = new Redis(redisOptions);
      const subClient = new Redis(redisOptions);

      // Suppress unhandled error events
      pubClient.on('error', () => {});
      subClient.on('error', () => {});

      await pubClient.connect();
      await subClient.connect();

      // Re-enable retry after successful initial connection
      // so transient disconnects are handled gracefully
      pubClient.options.retryStrategy = (times) => Math.min(times * 200, 5000);
      subClient.options.retryStrategy = (times) => Math.min(times * 200, 5000);

      this.adapterConstructor = createRedisAdapter(pubClient, subClient);
      this._activeAdapter = 'redis';
      this.logger.log('WebSocket adapter: Redis');
      return true;
    } catch {
      this.logger.warn('Redis unavailable for WebSocket adapter, trying PostgreSQL fallback...');
      return false;
    }
  }

  private async tryPostgresAdapter(databaseUrl: string): Promise<boolean> {
    try {
      const pool = new Pool({
        connectionString: databaseUrl,
        // Minimal pool for the LISTEN/NOTIFY adapter
        max: 5,
        connectionTimeoutMillis: 5000,
        application_name: 'nestjs-base-v2:socket-io-adapter',
      });

      // Suppress pool error events
      pool.on('error', (err) => {
        this.logger.warn(`WebSocket PG pool error: ${err.message}`);
      });

      // Verify connectivity
      const client = await pool.connect();
      client.release();

      // Create the attachments table if it doesn't exist
      await pool.query(`
        CREATE TABLE IF NOT EXISTS socket_io_attachments (
          id          bigserial UNIQUE,
          created_at  timestamptz DEFAULT NOW(),
          payload     bytea
        );
      `);

      this.adapterConstructor = createPostgresAdapter(pool);
      this._activeAdapter = 'postgres';
      this.logger.log('WebSocket adapter: PostgreSQL (fallback)');
      return true;
    } catch (err) {
      this.logger.warn(`PostgreSQL adapter also unavailable: ${(err as Error).message}`);
      return false;
    }
  }
}
