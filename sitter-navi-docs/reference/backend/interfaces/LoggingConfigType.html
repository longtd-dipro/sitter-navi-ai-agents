import { registerAs } from '@nestjs/config';

export interface LoggingConfigType {
  level: string;
  consoleEnabled: boolean;
  fileEnabled: boolean;
  fileDir: string;
  fileMaxSize: string;
  fileMaxDays: string;
  httpIgnorePaths: string[];
  redactFields: string[];
  appEnv: string;
  serviceName: string;
  /** Whether to write TypeORM SQL queries to a dedicated file (logs/sql-YYYY-MM-DD.log). */
  sqlLogEnabled: boolean;
  /** Directory for the dedicated SQL log file (defaults to LOG_DIR). */
  sqlLogDir: string;
}

/**
 * Parse a boolean environment variable.
 * Accepts 'true'/'1' as true, 'false'/'0' as false (case-insensitive, trimmed).
 * Returns undefined if the value is absent or unrecognised.
 */
function parseBoolEnv(value: string | undefined): boolean | undefined {
  if (value === undefined || value === null) return undefined;
  const normalised = value.trim().toLowerCase();
  if (normalised === 'true' || normalised === '1') return true;
  if (normalised === 'false' || normalised === '0') return false;
  return undefined;
}

export const loggingConfig = registerAs('logging', (): LoggingConfigType => {
  const appEnv = process.env.NODE_ENV || 'development';
  const validLevels = ['error', 'warn', 'info', 'http', 'debug', 'verbose'];
  const envLevel = process.env.LOG_LEVEL;

  // Default level: 'info' for production/staging, 'debug' for development
  let level: string;
  if (envLevel && validLevels.includes(envLevel)) {
    level = envLevel;
  } else {
    level = ['production', 'staging'].includes(appEnv) ? 'info' : 'debug';
  }

  const isDevEnv = appEnv === 'development';

  // LOG_CONSOLE_ENABLED: default true always
  const consoleEnabledParsed = parseBoolEnv(process.env.LOG_CONSOLE_ENABLED);
  const consoleEnabled =
    consoleEnabledParsed !== undefined ? consoleEnabledParsed : true;

  // LOG_FILE_ENABLED: default false for development, true for staging/production
  const fileEnabledParsed = parseBoolEnv(process.env.LOG_FILE_ENABLED);
  const fileEnabled =
    fileEnabledParsed !== undefined ? fileEnabledParsed : !isDevEnv;

  // LOG_SQL_ENABLED: default true for development, false for staging/production
  const sqlLogEnabledParsed = parseBoolEnv(process.env.LOG_SQL_ENABLED);
  const sqlLogEnabled =
    sqlLogEnabledParsed !== undefined ? sqlLogEnabledParsed : isDevEnv;

  // LOG_SQL_DIR: defaults to LOG_DIR
  const sqlLogDir = process.env.LOG_SQL_DIR || process.env.LOG_DIR || './logs';

  // Default sensitive fields to redact
  const defaultRedactFields = [
    'password',
    'currentPassword',
    'newPassword',
    'passwordConfirm',
    'token',
    'accessToken',
    'refreshToken',
    'authorization',
    'cookie',
    'setCookie',
    'secret',
    'apiKey',
    'creditCard',
  ];

  // Merge extra fields from LOG_REDACT_FIELDS (comma-separated)
  const extraFields = process.env.LOG_REDACT_FIELDS
    ? process.env.LOG_REDACT_FIELDS.split(',')
        .map((f) => f.trim())
        .filter(Boolean)
    : [];

  return {
    level,
    consoleEnabled,
    fileEnabled,
    fileDir: process.env.LOG_DIR || './logs',
    fileMaxSize: process.env.LOG_FILE_MAX_SIZE || '20m',
    fileMaxDays: process.env.LOG_FILE_MAX_DAYS || '14d',
    httpIgnorePaths: process.env.LOG_HTTP_IGNORE_PATHS
      ? process.env.LOG_HTTP_IGNORE_PATHS.split(',').map((p) => p.trim())
      : ['/api/health', '/swagger'],
    redactFields: [...defaultRedactFields, ...extraFields],
    appEnv,
    serviceName: 'nestjs-base-v2',
    sqlLogEnabled,
    sqlLogDir,
  };
});
