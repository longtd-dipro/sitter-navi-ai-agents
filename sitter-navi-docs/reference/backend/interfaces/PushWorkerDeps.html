import { Job } from 'bullmq';
import { QUEUE } from '@constants';
import { PushTemplateService } from '../../modules/push-notification/push-template.service';
import { PushRendererService } from '../../modules/push-notification/push-renderer.service';
import { OutboxPushService } from '../../modules/push-notification/outbox-push.service';
import { UserDeviceService } from '../../modules/push-notification/user-device.service';
import { FcmDispatcher } from '../../modules/push-notification/fcm-dispatcher.service';
import { PushJob } from '../../modules/push-notification/push-dispatcher.interface';
import { bootstrapWorker, WorkerContext, WorkerDefinition } from '../base/worker-bootstrap';

// ──────────────────────────────────────────────────────────────────
// Worker config
// ──────────────────────────────────────────────────────────────────

const CONCURRENCY = 5;
const RATE_MAX_PER_SEC = 10;
const ONE_SECOND_MS = 1000;

export interface PushWorkerDeps {
  templateService: PushTemplateService;
  renderer: PushRendererService;
  outboxService: OutboxPushService;
  userDeviceService: UserDeviceService;
  fcmDispatcher: FcmDispatcher;
}

/** Reusable worker definition — used by both standalone & all-in-one entry points. */
export const pushWorkerDef: WorkerDefinition<PushJob, PushWorkerDeps> = {
  options: {
    queueName: QUEUE.PUSH,
    concurrency: CONCURRENCY,
    limiter: { max: RATE_MAX_PER_SEC, duration: ONE_SECOND_MS },
  },
  processor: async (
    job: Job<PushJob>,
    { templateService, renderer, outboxService, userDeviceService, fcmDispatcher },
  ) => {
    const pushJob = job.data;

    if (pushJob.type === 'template') {
      // 1. Look up template
      const template = await templateService.findByTemplateName(pushJob.templateName);
      if (!template || !template.isActive) {
        await outboxService.markFailed(
          pushJob.outboxId,
          `Template "${pushJob.templateName}" not found or inactive`,
          true,
        );
        return;
      }

      // 2. Get user's active device tokens
      const deviceTokens = await userDeviceService.getActiveDevices(pushJob.userId);
      if (deviceTokens.length === 0) {
        await outboxService.markFailed(
          pushJob.outboxId,
          `No active devices for user ${pushJob.userId}`,
          true,
        );
        return;
      }

      // 3. Render EJS
      const rendered = await renderer.render(template, pushJob.data);

      // 4. Dispatch via FCM
      const result = await fcmDispatcher.dispatch({
        deviceTokens,
        title: rendered.title,
        body: rendered.body,
        imageUrl: rendered.imageUrl,
        dataPayload: rendered.dataPayload,
      });

      if (result.success) {
        await outboxService.markSent(pushJob.outboxId);
      } else {
        throw new Error(result.error ?? 'FCM dispatch failed');
      }
    } else {
      // Custom push — send directly
      const result = await fcmDispatcher.dispatch({
        deviceTokens: pushJob.deviceTokens,
        title: pushJob.title,
        body: pushJob.body,
        imageUrl: pushJob.imageUrl,
        dataPayload: pushJob.dataPayload,
      });

      if (result.success) {
        await outboxService.markSent(pushJob.outboxId);
      } else {
        throw new Error(result.error ?? 'FCM dispatch failed');
      }
    }
  },
  getDeps: (ctx: WorkerContext): PushWorkerDeps => ({
    templateService: ctx.app.get(PushTemplateService),
    renderer: ctx.app.get(PushRendererService),
    outboxService: ctx.app.get(OutboxPushService),
    userDeviceService: ctx.app.get(UserDeviceService),
    fcmDispatcher: ctx.app.get(FcmDispatcher),
  }),
};

// ── Standalone entry point ──
async function bootstrap() {
  await bootstrapWorker<PushJob, PushWorkerDeps>(
    pushWorkerDef.options,
    pushWorkerDef.processor,
    pushWorkerDef.getDeps,
  );
}

bootstrap().catch((err) => {
  console.error('Push worker failed to start:', err);
  process.exit(1);
});
