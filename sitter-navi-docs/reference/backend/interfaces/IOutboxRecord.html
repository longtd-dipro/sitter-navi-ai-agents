export enum OutboxStatus {
  PENDING = 'pending',
  SENDING = 'sending',
  SENT = 'sent',
  FAILED = 'failed',
  PERMANENTLY_FAILED = 'permanently_failed',
}

/**
 * Contract that every outbox entity must satisfy to work with BaseOutboxService.
 *
 * Each channel (Email, Push) has its own entity that extends their base entity
 * and also implements this interface.
 */
export interface IOutboxRecord {
  id: string;

  /** Current status of this outbox record */
  status: OutboxStatus;

  /** Number of dispatch attempts so far */
  tryCount: number;

  /** Maximum number of retries before marking as PERMANENTLY_FAILED */
  maxRetries: number;

  /** Error message from the last failed attempt */
  lastError: string | null;

  /** Timestamp of the last dispatch attempt */
  lastAttemptAt: Date | null;

  /**
   * When the next retry should occur.
   * Calculated by exponential backoff.
   * null means it's ready for immediate dispatch.
   */
  nextRetryAt: Date | null;

  /** Record creation timestamp */
  createdAt: Date;
}
