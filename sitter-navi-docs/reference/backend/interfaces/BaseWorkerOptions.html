import { NestFactory } from '@nestjs/core';
import { Worker, Job, WorkerOptions } from 'bullmq';
import { ConfigService } from '@nestjs/config';
import { AppModule } from '../../app.module';
import { QUEUE, getBullMqPrefix } from '@constants';

// ──────────────────────────────────────────────────────────────────
// Type for the processor function
// ──────────────────────────────────────────────────────────────────

// eslint-disable-next-line @typescript-eslint/no-explicit-any
export type WorkerProcessor<TData = unknown, TDeps = any> = (
  job: Job<TData>,
  deps: TDeps,
) => Promise<void>;

// ──────────────────────────────────────────────────────────────────
// Worker options (without connection — auto-filled from config)
// ──────────────────────────────────────────────────────────────────

export interface BaseWorkerOptions {
  queueName: string;
  concurrency: number;
  limiter: { max: number; duration: number };
}

// ──────────────────────────────────────────────────────────────────
// Bootstrap
// ──────────────────────────────────────────────────────────────────

export interface WorkerContext {
  app: Awaited<ReturnType<typeof NestFactory.createApplicationContext>>;
  configService: ConfigService;
}

// ──────────────────────────────────────────────────────────────────
// Multi-worker definition (for all-in-one process)
// ──────────────────────────────────────────────────────────────────

// eslint-disable-next-line @typescript-eslint/no-explicit-any
export interface WorkerDefinition<TData = unknown, TDeps = any> {
  options: BaseWorkerOptions;
  processor: WorkerProcessor<TData, TDeps>;
  getDeps: (ctx: WorkerContext) => TDeps;
}

// eslint-disable-next-line @typescript-eslint/no-explicit-any
export async function bootstrapAllWorkers(workers: WorkerDefinition<any, any>[]): Promise<void> {
  const HEARTBEAT_INTERVAL_MS = 5 * 60 * 1000;

  const app = await NestFactory.createApplicationContext(AppModule);
  const configService = app.get(ConfigService);

  const connection = {
    host: configService.get('redis.host'),
    port: configService.get('redis.port'),
    password: configService.get('redis.password'),
  };

  const ctx: WorkerContext = { app, configService };

  // Create all workers
  const instances: Worker[] = [];

  for (const def of workers) {
    const deps = def.getDeps(ctx);
    const { queueName, concurrency, limiter } = def.options;

    const worker = new Worker(
      queueName,
      async (job: Job) => {
        await def.processor(job, deps);
      },
      { connection, concurrency, limiter, prefix: getBullMqPrefix() },
    );

    instances.push(worker);
    console.log(
      `${queueName} worker started (PID: ${process.pid}, concurrency: ${concurrency}, limiter: ${limiter.max}/${limiter.duration}ms)`,
    );
  }

  const allLabels = workers.map((w) => w.options.queueName).join(' + ');
  console.log(`All workers ready: ${allLabels} (PID: ${process.pid})`);

  // ── Graceful shutdown ──
  const shutdown = async (signal: string) => {
    console.log(`All workers shutting down (${signal})...`);
    await Promise.all(instances.map((w) => w.close()));
    await app.close();
    process.exit(0);
  };

  process.on('SIGTERM', () => shutdown('SIGTERM'));
  process.on('SIGINT', () => shutdown('SIGINT'));

  // ── Heartbeat ──
  setInterval(() => {
    console.log(`[Heartbeat] All workers alive (PID: ${process.pid})`);
  }, HEARTBEAT_INTERVAL_MS);
}

// eslint-disable-next-line @typescript-eslint/no-explicit-any
export async function bootstrapWorker<TData = unknown, TDeps = any>(
  workerOptions: BaseWorkerOptions,
  processor: WorkerProcessor<TData, TDeps>,
  getDeps?: (ctx: WorkerContext) => TDeps,
): Promise<void> {
  const HEARTBEAT_INTERVAL_MS = 5 * 60 * 1000;

  // Create NestJS app context WITHOUT HTTP listener — DI only
  const app = await NestFactory.createApplicationContext(AppModule);
  const configService = app.get(ConfigService);

  // Resolve extra dependencies if provided
  const extraDeps = (getDeps?.({ app, configService }) ?? {}) as TDeps;

  const connection = {
    host: configService.get('redis.host'),
    port: configService.get('redis.port'),
    password: configService.get('redis.password'),
  };

  const worker = new Worker<TData>(
    workerOptions.queueName,
    async (job: Job<TData>) => {
      await processor(job, extraDeps);
    },
    {
      connection,
      concurrency: workerOptions.concurrency,
      limiter: workerOptions.limiter,
      prefix: getBullMqPrefix(),
    },
  );

  const queueLabel = workerOptions.queueName;
  console.log(
    `${queueLabel} worker started (PID: ${process.pid}, concurrency: ${workerOptions.concurrency}, limiter: ${workerOptions.limiter.max}/${workerOptions.limiter.duration}ms)`,
  );

  // ── Graceful shutdown ──
  const shutdown = async (signal: string) => {
    console.log(`${queueLabel} worker shutting down (${signal})...`);
    await worker.close();
    await app.close();
    process.exit(0);
  };

  process.on('SIGTERM', () => shutdown('SIGTERM'));
  process.on('SIGINT', () => shutdown('SIGINT'));

  // ── Heartbeat ──
  setInterval(() => {
    console.log(`[Heartbeat] ${queueLabel} worker alive (PID: ${process.pid})`);
  }, HEARTBEAT_INTERVAL_MS);
}
