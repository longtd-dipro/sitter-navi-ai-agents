import { Column, Entity, JoinColumn, ManyToOne } from 'typeorm';
import { CrudValidationGroups } from '@dataui/crud';
import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
import {
  IsBoolean,
  IsDateString,
  IsDefined,
  IsEnum,
  IsNotEmpty,
  IsNumber,
  IsOptional,
  IsString,
  MaxLength,
} from 'class-validator';

import { WithIdAndTimestamp } from '@entities/base.entity';
import { UserEntity } from '@modules/user/user.entity';
import { decimalToNumber } from '@utils/typeorm.transformer';

export enum Gender {
  MALE = 'male',
  FEMALE = 'female',
}

export enum ChildStatus {
  DRAFT = 'draft',
  COMPLETE = 'complete',
}

export const COMPLETE = 'CHILD-COMPLETE';

const { CREATE, UPDATE } = CrudValidationGroups;
const DRAFT_GROUPS = [CREATE, UPDATE];

@Entity('children')
export class Child extends WithIdAndTimestamp {
  @ApiPropertyOptional({ description: 'ID phụ huynh sở hữu — tự gán từ JWT khi tạo' })
  @IsOptional({ always: true })
  @Column({ type: 'bigint', comment: 'ID phụ huynh sở hữu (users.id)' })
  userId: string;

  @ApiPropertyOptional({ type: () => UserEntity, description: 'Phụ huynh sở hữu' })
  @ManyToOne(() => UserEntity, { onDelete: 'CASCADE' })
  @JoinColumn()
  user?: UserEntity;

  @ApiProperty({ enum: ChildStatus, default: ChildStatus.DRAFT })
  @IsEnum(ChildStatus)
  @IsOptional({ always: true })
  @Column({
    type: 'enum',
    enum: ChildStatus,
    default: ChildStatus.DRAFT,
    comment: 'Trạng thái hồ sơ',
  })
  status: ChildStatus;

  // --- THÔNG TIN CƠ BẢN ---
  // Mọi cột data đều nullable để lưu được bản nháp thiếu; field bắt buộc chỉ
  // được enforce ở group COMPLETE (lúc submit).

  @ApiPropertyOptional({ description: 'Họ (性) — bắt buộc khi hoàn tất', maxLength: 50 })
  @IsString({ always: true })
  @IsNotEmpty({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @MaxLength(50, { always: true })
  @Column({ type: 'varchar', length: 50, nullable: true, comment: 'Họ (性)' })
  lastName: string;

  @ApiPropertyOptional({ description: 'Tên (名) — bắt buộc khi hoàn tất', maxLength: 50 })
  @IsString({ always: true })
  @IsNotEmpty({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @MaxLength(50, { always: true })
  @Column({ type: 'varchar', length: 50, nullable: true, comment: 'Tên (名)' })
  firstName: string;

  @ApiPropertyOptional({
    description: 'Họ Furigana (せい) — bắt buộc khi hoàn tất',
    maxLength: 100,
  })
  @IsString({ always: true })
  @IsNotEmpty({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @MaxLength(100, { always: true })
  @Column({ type: 'varchar', length: 100, nullable: true, comment: 'Họ Furigana (せい)' })
  lastNameFurigana: string;

  @ApiPropertyOptional({
    description: 'Tên Furigana (めい) — bắt buộc khi hoàn tất',
    maxLength: 100,
  })
  @IsString({ always: true })
  @IsNotEmpty({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @MaxLength(100, { always: true })
  @Column({ type: 'varchar', length: 100, nullable: true, comment: 'Tên Furigana (めい)' })
  firstNameFurigana: string;

  @ApiPropertyOptional({
    description: 'Ngày sinh (生年月日) — bắt buộc khi hoàn tất',
    example: '2020-01-01',
  })
  @IsDateString(undefined, { always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({ type: 'date', nullable: true, comment: 'Ngày sinh (生年月日)' })
  dateOfBirth: Date;

  @ApiPropertyOptional({ description: 'Giới tính (性別) — bắt buộc khi hoàn tất', enum: Gender })
  @IsEnum(Gender, { always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({ type: 'enum', enum: Gender, nullable: true, comment: 'Giới tính (性別)' })
  gender: Gender;

  @ApiPropertyOptional({
    description: 'Có dị ứng hay không (アレルギーの有無) — bắt buộc khi hoàn tất',
  })
  @IsBoolean({ always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({ type: 'boolean', nullable: true, comment: 'Có dị ứng hay không (アレルギーの有無)' })
  hasAllergy: boolean;

  // --- THÔNG TIN CHI TIẾT / Y TẾ ---

  @ApiPropertyOptional({
    description: 'Phòng khám nhi thường trực (かかりつけの小児科) — bắt buộc khi hoàn tất',
  })
  @IsString({ always: true })
  @IsNotEmpty({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({
    type: 'text',
    nullable: true,
    comment: 'Phòng khám nhi thường trực (かかりつけの小児科)',
  })
  pediatricianDetails: string;

  @ApiPropertyOptional({
    description:
      'Có bệnh bẩm sinh/tiền sử bệnh không (先天性疾患や既往歴など) — bắt buộc khi hoàn tất',
  })
  @IsBoolean({ always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({
    type: 'boolean',
    nullable: true,
    comment: 'Có bệnh bẩm sinh/tiền sử bệnh không (先天性疾患や既往歴など)',
  })
  hasCongenitalDisease: boolean;

  @ApiPropertyOptional({ description: 'Chi tiết bệnh bẩm sinh (詳細入力)' })
  @IsString({ always: true })
  @IsOptional({ always: true })
  @Column({ type: 'text', nullable: true, comment: 'Chi tiết bệnh bẩm sinh (詳細入力)' })
  congenitalDiseaseDetails: string;

  @ApiPropertyOptional({
    description: 'Có lịch sử nhập viện không (入院歴) — bắt buộc khi hoàn tất',
  })
  @IsBoolean({ always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({ type: 'boolean', nullable: true, comment: 'Có lịch sử nhập viện không (入院歴)' })
  hasHospitalizationHistory: boolean;

  @ApiPropertyOptional({ description: 'Chi tiết lịch sử nhập viện (詳細入力)' })
  @IsString({ always: true })
  @IsOptional({ always: true })
  @Column({ type: 'text', nullable: true, comment: 'Chi tiết lịch sử nhập viện (詳細入力)' })
  hospitalizationHistoryDetails: string;

  @ApiPropertyOptional({
    description: 'Có tiền sử co giật do sốt không (ひきつけ 熱性けいれん) — bắt buộc khi hoàn tất',
  })
  @IsBoolean({ always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({
    type: 'boolean',
    nullable: true,
    comment: 'Có tiền sử co giật do sốt không (ひきつけ 熱性けいれん)',
  })
  hasConvulsions: boolean;

  @ApiPropertyOptional({ description: 'Chi tiết tiền sử co giật (詳細入力)' })
  @IsString({ always: true })
  @IsOptional({ always: true })
  @Column({ type: 'text', nullable: true, comment: 'Chi tiết tiền sử co giật (詳細入力)' })
  convulsionsDetails: string;

  @ApiPropertyOptional({
    description:
      'Có đang đi học tại cơ sở giáo dục đặc biệt/phục hồi chức năng không (療育施設への通所) — bắt buộc khi hoàn tất',
  })
  @IsBoolean({ always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({
    type: 'boolean',
    nullable: true,
    comment:
      'Có đang đi học tại cơ sở giáo dục đặc biệt/phục hồi chức năng không (療育施設への通所)',
  })
  isAttendingRehabilitation: boolean;

  @ApiPropertyOptional({ description: 'Chi tiết cơ sở rehabilitation (詳細入力)' })
  @IsString({ always: true })
  @IsOptional({ always: true })
  @Column({ type: 'text', nullable: true, comment: 'Chi tiết cơ sở rehabilitation (詳細入力)' })
  rehabilitationDetails: string;

  @ApiPropertyOptional({
    description: 'Có sổ tay khuyết tật/hỗ trợ không (手帳の有無) — bắt buộc khi hoàn tất',
  })
  @IsBoolean({ always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({
    type: 'boolean',
    nullable: true,
    comment: 'Có sổ tay khuyết tật/hỗ trợ không (手帳の有無)',
  })
  hasDisabilityHandbook: boolean;

  @ApiPropertyOptional({ description: 'Cấp độ/Hạng của sổ tay (等級)', maxLength: 50 })
  @IsString({ always: true })
  @IsOptional({ always: true })
  @MaxLength(50, { always: true })
  @Column({ type: 'varchar', length: 50, nullable: true, comment: 'Cấp độ/Hạng của sổ tay (等級)' })
  disabilityHandbookLevel: string;

  // --- ĐẶC ĐIỂM CÁ NHÂN ---

  @ApiPropertyOptional({
    description: 'Nhiệt độ cơ thể bình thường (平熱 °C) — bắt buộc khi hoàn tất',
    example: 36.5,
  })
  @IsNumber({ maxDecimalPlaces: 1 }, { always: true })
  @IsDefined({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({
    type: 'decimal',
    precision: 3,
    scale: 1,
    nullable: true,
    transformer: decimalToNumber,
    comment: 'Nhiệt độ cơ thể bình thường (平熱 °C)',
  })
  normalBodyTemperature: number;

  @ApiPropertyOptional({ description: 'Tính cách (性格) — bắt buộc khi hoàn tất' })
  @IsString({ always: true })
  @IsNotEmpty({ groups: [COMPLETE] })
  @IsOptional({ groups: DRAFT_GROUPS })
  @Column({ type: 'text', nullable: true, comment: 'Tính cách (性格)' })
