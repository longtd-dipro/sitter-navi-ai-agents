import { ApiProperty } from '@nestjs/swagger';
import { IsBoolean } from 'class-validator';
import { Column, Entity } from 'typeorm';
import { WithIdAndTimestamp } from '@entities/base.entity';
import { NumberField, StringField } from '@decorators/common.decorator';

export enum OtpPurpose {
  PASSWORD_RESET = 'password_reset',
  LOGIN_VERIFY = 'login_verify',
  EMAIL_VERIFY = 'email_verify',
}

@Entity('otp')
export class OTPEntity extends WithIdAndTimestamp {
  // NOTE: Cross-module reference to User module decoupled per Requirements 5.1-5.3, 11.2-11.3.
  // Previous `@ManyToOne(() => UserEntity)` + `@JoinColumn({ name: 'user_id' })` removed.
  // The `user_id` DB column is preserved via the `userId` column below (same nullability + unique).
  // Lookup of the related user must be performed at the service layer via DI (UserService/UserRepository).
  @NumberField({ optional: false })
  @Column({ nullable: true, unique: true })
  userId: string;

  @StringField({ optional: false })
  @Column({ type: 'text', nullable: false })
  otp: string;

  @ApiProperty()
  @IsBoolean()
  @Column({
    type: 'timestamptz',
    nullable: false,
  })
  expiredAt: Date;

  @Column({ type: 'varchar', unique: true })
  identifier: string;

  @Column({ type: 'varchar', nullable: true, default: OtpPurpose.LOGIN_VERIFY })
  purpose: OtpPurpose;

  @Column({ type: 'int', nullable: true, default: 0 })
  retry: number;

  @Column({ type: 'boolean', nullable: true, default: false })
  isUsed: boolean;

  @Column({ type: 'jsonb' })
  meta: Record<string, any>;
}
