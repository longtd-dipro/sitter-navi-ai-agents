import { Column, Entity } from 'typeorm';
import { WithIdAndTimestamp } from '../../entities/base.entity';
import { IOutboxRecord, OutboxStatus } from '@messaging';

@Entity('outbox_push')
export class OutboxPush extends WithIdAndTimestamp implements IOutboxRecord {
  // ──── IOutboxRecord contract ────

  @Column({ type: 'varchar', default: OutboxStatus.PENDING })
  status: OutboxStatus;

  @Column({ type: 'int', default: 0 })
  tryCount: number;

  @Column({ type: 'int', default: 3 })
  maxRetries: number;

  @Column({ type: 'varchar', nullable: true })
  lastError: string | null;

  @Column({ type: 'timestamptz', nullable: true })
  lastAttemptAt: Date | null;

  @Column({ type: 'timestamptz', nullable: true })
  nextRetryAt: Date | null;

  // ──── Push-specific fields ────

  /** User ID for traceability (nullable — raw send may not have userId) */
  @Column({ type: 'varchar', nullable: true })
  userId: string | null;

  /** Array of FCM device tokens to send to */
  @Column({ type: 'jsonb' })
  deviceTokens: string[];

  @Column({ type: 'varchar' })
  title: string;

  @Column({ type: 'text' })
  body: string;

  @Column({ type: 'varchar', nullable: true })
  imageUrl: string | null;

  /** Custom data payload for deep linking */
  @Column({ type: 'jsonb', nullable: true })
  dataPayload: Record<string, string> | null;

  /** Template name used to render this push (for traceability) */
  @Column({ type: 'varchar', nullable: true })
  templateName: string | null;

  /** Event name that triggered this push (for traceability) */
  @Column({ type: 'varchar', nullable: true })
  eventName: string | null;
}
