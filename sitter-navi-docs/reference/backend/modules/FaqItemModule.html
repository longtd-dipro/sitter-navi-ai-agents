import { Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';

import { FaqSubCategoryModule } from '../faq-sub-category/faq-sub-category.module';
import { AdminFaqItemController } from './faq-item.controller.admin';
import { FaqItem } from './faq-item.entity';
import { FaqItemService } from './faq-item.service';

@Module({
  imports: [
    TypeOrmModule.forFeature([FaqItem]),
    FaqSubCategoryModule, // cross-module DI cho FaqSubCategoryService
  ],
  providers: [FaqItemService],
  controllers: [AdminFaqItemController],
  exports: [FaqItemService],
})
export class FaqItemModule {}
