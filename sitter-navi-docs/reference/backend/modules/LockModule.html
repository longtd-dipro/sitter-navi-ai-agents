Description

LockModule provides distributed locking infrastructure.

Default priority: Redis (primary) → PostgreSQL (fallback). If Redis is unavailable, all lock operations automatically fall back to PG advisory locks.

Example :
// Default — Redis primary + PG fallback
LockModule.forRoot()


// Custom adapters
LockModule.forRoot({
  adapters: [redisLockAdapter, postgresLockAdapter],
})

import { Global, Module, OnModuleInit, OnApplicationShutdown, Logger } from '@nestjs/common';
import { LockManager, LockManagerConfig } from './lock-manager';
import { PostgresLockAdapter } from './adapters/postgres-lock.adapter';
import { RedisLockAdapter } from './adapters/redis-lock.adapter';
import { DistributedLockConfig } from './distributed-lock.decorator';
import { LockAdapter } from './interfaces';

export const LOCK_MANAGER = 'LOCK_MANAGER';

/**
 * LockModule provides distributed locking infrastructure.
 *
 * Default priority: Redis (primary) → PostgreSQL (fallback).
 * If Redis is unavailable, all lock operations automatically fall back to PG advisory locks.
 *
 * @example
 * ```ts
 * // Default — Redis primary + PG fallback
 * LockModule.forRoot()
 *
 * // Custom adapters
 * LockModule.forRoot({
 *   adapters: [redisLockAdapter, postgresLockAdapter],
 * })
 * ```
 */
@Global()
@Module({})
export class LockModule implements OnModuleInit, OnApplicationShutdown {
  private static lockManager: LockManager;
  private static config: LockManagerConfig | null = null;
  private static redisLockAdapter: RedisLockAdapter | null = null;
  private static readonly logger = new Logger('LockModule');

  /**
   * Configure the lock module with custom adapters and options.
   * Default: Redis primary → PostgreSQL fallback.
   */
  static forRoot(config?: Partial<LockManagerConfig>) {
    this.config = config as LockManagerConfig | null;

    return {
      module: LockModule,
      providers: [
        {
          provide: LOCK_MANAGER,
          useFactory: () => {
            if (!LockModule.lockManager) {
              const adapters = config?.adapters ?? LockModule.createDefaultAdapters();
              LockModule.lockManager = new LockManager({
                adapters,
                defaultTtlMs: config?.defaultTtlMs,
                defaultOnFail: config?.defaultOnFail,
              });
            }
            return LockModule.lockManager;
          },
        },
      ],
      exports: [LOCK_MANAGER],
    };
  }

  /**
   * Get the RedisLockAdapter instance (used to configure it with a Redis client after DI is ready).
   */
  static getRedisLockAdapter(): RedisLockAdapter | null {
    return this.redisLockAdapter;
  }

  /**
   * Get the active adapter status for startup logging.
   */
  static async getAdapterStatus(): Promise<{ primary: string; healthy: boolean }> {
    if (!LockModule.lockManager) {
      return { primary: 'none', healthy: false };
    }

    // Check Redis first (it's the primary)
    if (LockModule.redisLockAdapter) {
      const redisHealthy = await LockModule.redisLockAdapter.isHealthy().catch(() => false);
      if (redisHealthy) {
        return { primary: 'redis', healthy: true };
      }
    }

    // Fallback to PG
    return { primary: 'postgres (fallback)', healthy: true };
  }

  onModuleInit() {
    // If forRoot() was not called, create default Redis + PG adapters
    if (!LockModule.lockManager) {
      const adapters = LockModule.createDefaultAdapters();
      LockModule.lockManager = new LockManager({ adapters });
    }

    // Wire the global config so @DistributedLock decorator can access it
    DistributedLockConfig.setManager(LockModule.lockManager);
  }

  async onApplicationShutdown() {
    if (LockModule.lockManager) {
      await LockModule.lockManager.shutdown();
    }
  }

  /**
   * Creates the default adapter chain: Redis (primary) → PostgreSQL (fallback).
   */
  private static createDefaultAdapters(): LockAdapter[] {
    const redisAdapter = new RedisLockAdapter();
    LockModule.redisLockAdapter = redisAdapter;

    const pgAdapter = new PostgresLockAdapter();

    LockModule.logger.log('Lock adapters configured: Redis (primary) → PostgreSQL (fallback)');

    return [redisAdapter, pgAdapter];
  }
}
