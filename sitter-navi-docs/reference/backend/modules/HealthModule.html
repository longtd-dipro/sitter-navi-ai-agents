// ─────────────────────────────────────────────────────────────────────────────
// Health Module.
//
// Pluggable health check system using token-based multi-provider injection.
// Each dependency has its own IHealthIndicator implementation.
// New indicators can be added without modifying any existing code.
//
// The HEALTH_INDICATOR multi-provider token collects all indicators
// into an array and injects them into HealthService — same pattern
// used by EMAIL_DISPATCHER and PUSH_DISPATCHER.
// ─────────────────────────────────────────────────────────────────────────────

import { Module } from '@nestjs/common';
import { HealthController } from './health.controller';
import { HealthService } from './health.service';
import { DatabaseHealthIndicator } from './indicators/database.health-indicator';
import { RedisHealthIndicator } from './indicators/redis.health-indicator';
import { CacheHealthIndicator } from './indicators/cache.health-indicator';
import { LockHealthIndicator } from './indicators/lock.health-indicator';
import { DiskHealthIndicator } from './indicators/disk.health-indicator';
import { MemoryHealthIndicator } from './indicators/memory.health-indicator';
import { MemoryMonitorService } from './memory-monitor.service';
import { HEALTH_INDICATOR, IHealthIndicator } from './indicators/health-indicator.interface';

@Module({
  controllers: [HealthController],
  providers: [
    HealthService,

    // Background services
    MemoryMonitorService,

    // Individual indicators
    DatabaseHealthIndicator,
    RedisHealthIndicator,
    CacheHealthIndicator,
    LockHealthIndicator,
    DiskHealthIndicator,
    MemoryHealthIndicator,

    // Collect all indicators into an array via useFactory.
    // useExisting with the same token does NOT auto-collect in NestJS —
    // useFactory with inject is the correct approach for multi-provider patterns.
    {
      provide: HEALTH_INDICATOR,
      useFactory: (
        db: IHealthIndicator,
        redis: IHealthIndicator,
        cache: IHealthIndicator,
        lock: IHealthIndicator,
        disk: IHealthIndicator,
        memory: IHealthIndicator,
      ) => [db, redis, cache, lock, disk, memory],
      inject: [
        DatabaseHealthIndicator,
        RedisHealthIndicator,
        CacheHealthIndicator,
        LockHealthIndicator,
        DiskHealthIndicator,
        MemoryHealthIndicator,
      ],
    },
  ],
})
export class HealthModule {}
