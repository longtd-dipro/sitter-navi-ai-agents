import { Global, Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';
import { APP_GUARD } from '@nestjs/core';
import { DiscoveryModule } from '@nestjs/core';

import {
  Permission,
  PermissionGroup,
  PermissionGroupItem,
  RolePermission,
  UserPermission,
  UserPermissionGroup,
} from './entities';
import { PermissionService } from './services/permission.service';
import { PermissionCacheService } from './services/permission-cache.service';
import { PermissionRegistryService } from './services/permission-registry.service';
import { PermissionGuard } from './guards/permission.guard';
import { AdminPermissionController } from './permission.controller.admin';

@Global()
@Module({
  imports: [
    DiscoveryModule,
    TypeOrmModule.forFeature([
      Permission,
      PermissionGroup,
      PermissionGroupItem,
      RolePermission,
      UserPermission,
      UserPermissionGroup,
    ]),
  ],
  providers: [
    PermissionService,
    PermissionCacheService,
    PermissionRegistryService,
    PermissionGuard,
    {
      provide: APP_GUARD,
      useExisting: PermissionGuard,
    },
  ],
  controllers: [AdminPermissionController],
  exports: [PermissionService, PermissionCacheService, PermissionRegistryService, PermissionGuard],
})
export class PermissionModule {}
