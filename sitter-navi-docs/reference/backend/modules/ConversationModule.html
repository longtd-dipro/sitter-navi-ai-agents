import { Module } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { JwtModule } from '@nestjs/jwt';
import { TypeOrmModule } from '@nestjs/typeorm';
import { EventEmitterModule } from '@nestjs/event-emitter';
import {
  Conversation,
  Participant,
  Message,
  MessageAttachment,
  MessageStatus,
  UploadedFile,
  UserEntity,
} from '@entities';
import { CacheModule } from '@infrastructure/cache';
import { StorageModule } from '../storage/storage.module';
import { ConversationService } from './conversation.service';
import { ConversationController } from './conversation.controller.client';
import { ConversationGateway } from './conversation.gateway';
import { ParticipantGuard } from './guards/participant.guard';
import { GroupRoleGuard } from './guards/group-role.guard';
import {
  CONVERSATION_REPOSITORY,
  PARTICIPANT_REPOSITORY,
  MESSAGE_REPOSITORY,
  MESSAGE_ATTACHMENT_REPOSITORY,
} from './repositories/tokens';
import {
  TypeOrmConversationRepository,
  TypeOrmParticipantRepository,
  TypeOrmMessageRepository,
  TypeOrmMessageAttachmentRepository,
} from './repositories/typeorm';

@Module({
  imports: [
    TypeOrmModule.forFeature([
      Conversation,
      Participant,
      Message,
      MessageAttachment,
      MessageStatus,
      UploadedFile,
      UserEntity,
    ]),
    JwtModule.registerAsync({
      useFactory: (config: ConfigService) => ({
        secret: config.get<string>('jwt.secret'),
        signOptions: {
          expiresIn: config.get('jwt.expireTime') as any,
        },
      }),
      inject: [ConfigService],
    }),
    StorageModule,
    EventEmitterModule,
    CacheModule,
  ],
  providers: [
    ConversationService,
    ConversationGateway,
    ParticipantGuard,
    GroupRoleGuard,
    {
      provide: CONVERSATION_REPOSITORY,
      useClass: TypeOrmConversationRepository,
    },
    { provide: PARTICIPANT_REPOSITORY, useClass: TypeOrmParticipantRepository },
    { provide: MESSAGE_REPOSITORY, useClass: TypeOrmMessageRepository },
    {
      provide: MESSAGE_ATTACHMENT_REPOSITORY,
      useClass: TypeOrmMessageAttachmentRepository,
    },
  ],
  controllers: [ConversationController],
  exports: [ConversationService, ConversationGateway],
})
export class ConversationModule {}
