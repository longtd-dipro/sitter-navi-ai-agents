@Module({
  imports: [
    // ScheduleModule.forRoot(),
    EventEmitterModule.forRoot({
      wildcard: true,
      delimiter: '.',
      verboseMemoryLeak: true,
    }),
    ConfigModule.forRoot({
      load: [configuration, configurationRedis, businessHoursConfig, loggingConfig],
      isGlobal: true,
    }),
    TypeOrmModule.forRootAsync({
      imports: [ConfigModule],
      useFactory: (configService: ConfigService, loggingService: LoggingService) => {
        const logConfig = configService.get<LoggingConfigType>('logging');
        const typeOrmLogger = new TypeOrmLoggerService(loggingService, {
          sqlFileEnabled: logConfig?.sqlLogEnabled ?? false,
          sqlFileDir: logConfig?.sqlLogDir ?? './logs',
          sqlFileMaxSize: logConfig?.fileMaxSize ?? '20m',
          sqlFileMaxDays: logConfig?.fileMaxDays ?? '14d',
        });
        return {
          type: 'postgres',
          url: configService.get<string>('database.url'),
          entities: [
            join(__dirname, '/entities/**/*.entity.{ts,js}'),
            join(__dirname, '/modules/**/*.entity.{ts,js}'),
          ],
          logger: typeOrmLogger,
          synchronize: configService.get<boolean>('database.synDB'),
          namingStrategy: new SnakeNamingStrategy(),
          ssl:
            configService.get<string>('database.ssl') === 'true'
              ? { rejectUnauthorized: false }
              : false,
          maxQueryExecutionTime: 1000,
          applicationName: `nestjs-base-v2-${configService.get<string>('appEnv') || 'development'}`,
          extra: {
            max: 50, // Pool max connections for high-concurrency chat workloads
          },
        };
      },
      inject: [ConfigService, LoggingService],
    }),
    I18nModule.forRootAsync({
      useFactory: () => ({
        fallbackLanguage: 'jp',
        loaderOptions: {
          path: join(__dirname, 'i18n'),
          watch: false,
        },
      }),
      inject: [ConfigService],
      resolvers: [
        { use: QueryResolver, options: ['lang', 'locale', 'l'] },
        new HeaderResolver(['x-lang']),
        AcceptLanguageResolver,
        new CookieResolver(['lang', 'locale', 'l']),
      ],
    }),
    AuthModule,
    SharedModule,
    UploadModule,
    LoggingModule,
    CacheModule,
    HttpClientModule.register(),
    LockModule.forRoot(),
    PermissionModule,

export class AppModule implements OnModuleInit, NestModule {
  constructor(private dataSource: DataSource) {}

  onModuleInit() {
    DataSourceHolder.set(this.dataSource);
  }

  configure(consumer: MiddlewareConsumer) {
    consumer.apply(CorrelationIdMiddleware).forRoutes('*');
  }
}
