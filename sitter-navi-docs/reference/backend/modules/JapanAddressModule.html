import { Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';

import { Prefecture } from './prefecture.entity';
import { Municipality } from './municipality.entity';
import { PostalCode } from './postal-code.entity';
import { PrefectureService } from './prefecture.service';
import { MunicipalityService } from './municipality.service';
import { PostalCodeService } from './postal-code.service';
import { AdminPrefectureController } from './prefecture.controller.admin';
import { ClientPrefectureController } from './prefecture.controller.client';
import { AdminMunicipalityController } from './municipality.controller.admin';
import { ClientMunicipalityController } from './municipality.controller.client';
import { AdminPostalCodeController } from './postal-code.controller.admin';
import { ClientPostalCodeController } from './postal-code.controller.client';

@Module({
  imports: [TypeOrmModule.forFeature([Prefecture, Municipality, PostalCode])],
  providers: [PrefectureService, MunicipalityService, PostalCodeService],
  controllers: [
    AdminPrefectureController,
    ClientPrefectureController,
    AdminMunicipalityController,
    ClientMunicipalityController,
    AdminPostalCodeController,
    ClientPostalCodeController,
  ],
  exports: [PrefectureService, MunicipalityService, PostalCodeService],
})
export class JapanAddressModule {}
