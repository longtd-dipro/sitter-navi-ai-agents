import { Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';

import { EmailTemplate } from './email-template.entity';
import { OutboxEmail } from './outbox-email.entity';
import { EmailTemplateService } from './email-template.service';
import { EmailRendererService } from './email-renderer.service';
import { OutboxEmailService } from './outbox-email.service';
import { SmtpEmailDispatcher } from './smtp-email-dispatcher.service';
import { EMAIL_DISPATCHER } from './email-dispatcher.interface';
import { MailService } from './mail.service';
import { MailScheduler } from './mail.scheduler';
import { AdminEmailTemplateController } from './email-template.controller.admin';

@Module({
  imports: [TypeOrmModule.forFeature([EmailTemplate, OutboxEmail])],
  providers: [
    EmailTemplateService,
    EmailRendererService,
    OutboxEmailService,
    SmtpEmailDispatcher,
    /**
     * Token-based injection for IEmailDispatcher.
     *
     * To swap to BullMQ later:
     *   Change useClass to BullMqEmailDispatcher.
     *   No other code changes needed.
     */
    { provide: EMAIL_DISPATCHER, useClass: SmtpEmailDispatcher },
    MailService,
    MailScheduler,
  ],
  controllers: [AdminEmailTemplateController],
  /**
   * Export MailService so other modules can inject it for direct
   * email sending (sendByTemplate, sendCustomEmail).
   */
  exports: [MailService, EMAIL_DISPATCHER],
})
export class EmailModule {}
