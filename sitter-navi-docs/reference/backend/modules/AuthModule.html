import { OTPEntity } from './otp.entity';
import { RefreshTokenEntity } from './refresh-token.entity';
import { Module } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { JwtModule } from '@nestjs/jwt';
import { PassportModule } from '@nestjs/passport';
import { TypeOrmModule } from '@nestjs/typeorm';
import { UserModule } from '@modules/user/user.module';
import { ParentProfileModule } from '@modules/parent-profile/parent-profile.module';
import { AuthController } from './controller/auth.controller';
import { AdminAuthController } from './controller/admin-auth.controller';
import { AuthOtpController } from './controller/auth.otp.controller';
import { RefreshTokenController } from './controller/refresh-token.controller';
import { AuthService } from './service/auth.service';
import { AdminAuthService } from './service/admin-auth.service';
import { OtpService } from './service/otp.service';
import { SocialService } from './service/social.service';
import { RefreshTokenService } from './service/refresh-token.service';
import { AuthRepository } from './auth.repository';
import { RefreshTokenRepository } from './refresh-token.repository';
import { JwtStrategy } from './strategies/jwt.strategy';
import { RefreshJwtStrategy } from './strategies/refresh-jwt.strategy';
import type { JwtModuleOptions } from '@nestjs/jwt';

@Module({
  imports: [
    // OTPEntity stays registered for AuthRepository's @InjectRepository.
    // UserModule is imported to receive UserRepository via cross-module DI
    // (UserModule exports UserRepository per task 4.3).
    TypeOrmModule.forFeature([OTPEntity, RefreshTokenEntity]),
    UserModule,
    // Exports ParentProfileService — the parent onboarding step persists the
    // parent_profile row via it (see AuthService.registerParentInfo).
    ParentProfileModule,
    PassportModule,
    JwtModule.registerAsync({
      useFactory: (config: ConfigService): JwtModuleOptions => ({
        secret: config.get<string>('jwt.secret'),
        signOptions: {
          expiresIn: config.get('jwt.expireTime') as any,
        },
      }),
      inject: [ConfigService],
    }),
  ],
  providers: [
    JwtStrategy,
    AuthService,
    AdminAuthService,
    OtpService,
    SocialService,
    RefreshTokenService,
    AuthRepository,
    RefreshTokenRepository,
    RefreshJwtStrategy,
  ],
  exports: [AuthService, AdminAuthService, OtpService, SocialService, RefreshTokenService],
  controllers: [AuthController, AdminAuthController, AuthOtpController, RefreshTokenController],
})
export class AuthModule {}
