import { Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';

import { PushTemplate } from './push-template.entity';
import { OutboxPush } from './outbox-push.entity';
import { UserDevice } from './user-device.entity';
import { PushTemplateService } from './push-template.service';
import { PushRendererService } from './push-renderer.service';
import { OutboxPushService } from './outbox-push.service';
import { UserDeviceService } from './user-device.service';
import { FcmDispatcher } from './fcm-dispatcher.service';
import { PUSH_DISPATCHER } from './push-dispatcher.interface';
import { PushService } from './push.service';
import { PushScheduler } from './push.scheduler';
import { AdminPushTemplateController } from './push-template.controller.admin';
import { ClientDeviceController } from './client-device.controller';
import { PushTestController } from './push-test.controller';

@Module({
  imports: [TypeOrmModule.forFeature([PushTemplate, OutboxPush, UserDevice])],
  providers: [
    PushTemplateService,
    PushRendererService,
    OutboxPushService,
    UserDeviceService,
    FcmDispatcher,
    /**
     * Token-based injection for IPushDispatcher.
     *
     * To swap to BullMQ later:
     *   Change useClass to BullMqPushDispatcher.
     */
    { provide: PUSH_DISPATCHER, useClass: FcmDispatcher },
    PushService,
    PushScheduler,
  ],
  controllers: [AdminPushTemplateController, ClientDeviceController, PushTestController],
  /**
   * Export PushService and UserDeviceService so other modules can use them.
   */
  exports: [PushService, PUSH_DISPATCHER, UserDeviceService],
})
export class PushNotificationModule {}
