Description

Abstract base class for outbox services.

Provides CRUD + recovery operations for the transactional outbox pattern. Retry timing is handled externally:

  • BullMQ mode: BullMQ built-in retry + backoff.
  • Direct mode: Scheduler polls getStuckPending() every 30s.
  • Permanent errors (invalid email, unregistered token) skip retry entirely.

Usage: @Injectable() class OutboxEmailService extends BaseOutboxService { constructor(@InjectRepository(OutboxEmail) repo: Repository) { super(repo); } }

import { Logger } from '@nestjs/common';
import { Repository, ObjectLiteral } from 'typeorm';
import { IOutboxRecord, OutboxStatus } from './outbox-record.interface';

const DEFAULT_BATCH_SIZE = 100;
const MAX_ERROR_LENGTH = 1000;
const MAX_LOG_ERROR_LENGTH = 200;
const DEFAULT_CLEANUP_DAYS = 30;

/**
 * Abstract base class for outbox services.
 *
 * Provides CRUD + recovery operations for the transactional outbox pattern.
 * Retry timing is handled externally:
 *   - BullMQ mode: BullMQ built-in retry + backoff.
 *   - Direct mode: Scheduler polls getStuckPending() every 30s.
 *   - Permanent errors (invalid email, unregistered token) skip retry entirely.
 *
 * Usage:
 *   @Injectable()
 *   class OutboxEmailService extends BaseOutboxService<OutboxEmail> {
 *     constructor(@InjectRepository(OutboxEmail) repo: Repository<OutboxEmail>) {
 *       super(repo);
 *     }
 *   }
 *
 * @template T - The entity type, must extend ObjectLiteral (TypeORM) & IOutboxRecord.
 */
export abstract class BaseOutboxService<T extends ObjectLiteral & IOutboxRecord> {
  protected readonly logger = new Logger(this.constructor.name);

  constructor(protected readonly repo: Repository<T>) {}

  // ──────────────────────────────────────────────────────────────────
  // Public API — CRUD
  // ──────────────────────────────────────────────────────────────────

  /**
   * Create a new pending outbox record.
   */
  async createPending(data: Partial<T>): Promise<T> {
    const entity = this.repo.create({
      ...data,
      status: OutboxStatus.PENDING,
      tryCount: 0,
    } as unknown as T);

    return this.repo.save(entity);
  }

  /**
   * Get a batch of pending records ready for dispatch.
   * Filters records where nextRetryAt is null or in the past.
   */
  async getPendingBatch(limit = DEFAULT_BATCH_SIZE): Promise<T[]> {
    return this.repo
      .createQueryBuilder('o')
      .where('o.status = :status', { status: OutboxStatus.PENDING })
      .orderBy('o.createdAt', 'ASC')
      .limit(DEFAULT_BATCH_SIZE)
      .getMany();
  }

  /**
   * Get records stuck in PENDING state for longer than the specified duration.
   * Used by the scheduler for recovery when BullMQ/Redis is unavailable.
   *
   * @param minutes - Records pending for more than this many minutes.
   */
  async getStuckPending(minutes: number): Promise<T[]> {
    const cutoff = new Date(Date.now() - minutes * 60 * 1000);

    return this.repo
      .createQueryBuilder('o')
      .where('o.status = :status', { status: OutboxStatus.PENDING })
      .andWhere('o.createdAt < :cutoff', { cutoff })
      .orderBy('o.createdAt', 'ASC')
      .limit(100)
      .getMany();
  }

  /**
   * Mark a record as successfully sent.
   * Uses TypeORM increment() — safe from SQL injection.
   *
   * @param id    - Outbox record ID.
   * @param patch - Optional extra fields to persist (e.g. rendered subject/body).
   */
  async markSent(id: string, patch?: Partial<T>): Promise<void> {
    await this.repo.increment({ id } as any, 'tryCount', 1);
    await this.repo.update(id, {
      ...patch,
      status: OutboxStatus.SENT,
      lastAttemptAt: new Date(),
    } as unknown as Partial<T>);
  }

  /**
   * Mark a record as failed.
   *
   * @param id        - Outbox record ID.
   * @param error     - Error message from dispatcher.
   * @param permanent - If true, mark PERMANENTLY_FAILED immediately (no retry).
   *                    Use for permanent errors: invalid email, token not registered.
   *                    Default false → keeps PENDING for scheduler retry.
   * @param patch     - Optional extra fields to persist (e.g. rendered subject/body),
   *                    saved even though dispatch failed so the row reflects what was built.
   */
  async markFailed(
    id: string,
    error: string,
    permanent = false,
    patch?: Partial<T>,
  ): Promise<void> {
    const record = await this.repo.findOneBy({ id } as any);
    if (!record) {
      this.logger.warn(`markFailed: record not found id=${id}`);
      return;
    }

    const newTryCount = record.tryCount + 1;
    const exhausted = newTryCount >= record.maxRetries;
    const isPermanent = permanent || exhausted;

    await this.repo.increment({ id } as any, 'tryCount', 1);
    await this.repo.update(id, {
      ...patch,
      status: isPermanent ? OutboxStatus.PERMANENTLY_FAILED : OutboxStatus.PENDING,
      lastError: error.slice(0, MAX_ERROR_LENGTH),
      lastAttemptAt: new Date(),
    } as unknown as Partial<T>);

    if (isPermanent) {
      this.logger.error(
        `Record permanently failed id=${id} (attempts=${newTryCount}, permanent=${permanent}): ${error.slice(0, MAX_LOG_ERROR_LENGTH)}`,
      );
    }
  }

  /**
   * Delete old sent or permanently-failed records.
   *
   * @param days - Delete records older than this many days.
   * @returns Number of deleted records.
   */
  async cleanup(days = DEFAULT_CLEANUP_DAYS): Promise<number> {
    const result = await this.repo
      .createQueryBuilder()
      .delete()
      .where('status IN (:...statuses)', {
        statuses: [OutboxStatus.SENT, OutboxStatus.PERMANENTLY_FAILED],
      })
      .andWhere(`createdAt < NOW() - INTERVAL :days DAY`, { days })
      .execute();

    const count = result.affected ?? 0;
    if (count > 0) {
      this.logger.log(`Cleaned up ${count} old outbox records`);
    }
    return count;
  }
}
