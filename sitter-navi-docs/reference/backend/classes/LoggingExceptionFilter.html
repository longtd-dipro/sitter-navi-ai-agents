import { ArgumentsHost, Catch, ExceptionFilter, HttpException, HttpStatus } from '@nestjs/common';
import { I18nService } from 'nestjs-i18n';
import { LoggingService } from './logging.service';

@Catch()
export class LoggingExceptionFilter implements ExceptionFilter {
  constructor(
    private readonly logger: LoggingService,
    private readonly i18n?: I18nService,
  ) {
    this.logger.setContext('ExceptionFilter');
  }

  async catch(exception: unknown, host: ArgumentsHost): Promise<void> {
    const ctx = host.switchToHttp();
    const request = ctx.getRequest<{
      method: string;
      url: string;
      i18nLang?: string;
    }>();
    const response = ctx.getResponse();

    // If request is not available (e.g. non-HTTP context), skip
    if (!request?.method) {
      throw exception;
    }

    const method = request.method;
    const url = request.url;

    const correlationId = LoggingService.getAsyncLocalStorage().getStore()?.correlationId;

    const status =
      exception instanceof HttpException ? exception.getStatus() : HttpStatus.INTERNAL_SERVER_ERROR;

    const name = exception instanceof Error ? exception.name : 'UnknownError';

    // Original (untranslated) message — kept as `errorCode` for debugging and
    // used as the i18n key for translation.
    const errorCode = this.extractMessage(exception);

    // Translated message for the client; falls back to the original when there
    // is no i18n service or no matching translation key.
    const message = await this.translate(errorCode, request.i18nLang);

    // Log the ORIGINAL message + errorCode so logs stay debuggable regardless
    // of the request locale.
    if (status >= 500) {
      const metadata: Record<string, unknown> = {
        name,
        errorCode,
        correlationId,
        method,
        url,
        statusCode: status,
      };

      if (exception instanceof Error && exception.stack) {
        metadata.stack = exception.stack;
      }

      this.logger.error(errorCode, metadata);
    } else if (status >= 400) {
      this.logger.warn(errorCode, {
        name,
        errorCode,
        correlationId,
        method,
        url,
        statusCode: status,
      });
    }

    // Send JSON response
    if (response && !response.headersSent) {
      const responseBody =
        exception instanceof HttpException
          ? exception.getResponse()
          : {
              statusCode: status,
              message,
              error: name,
            };

      // Normalize response to always be an object, with the translated message
      // and the original code attached for debugging.
      const base =
        typeof responseBody === 'string'
          ? { statusCode: status, message, error: name }
          : { ...responseBody, message };

      const body = {
        ...base,
        errorCode,
        ...(correlationId && { correlationId }),
      };

      response.status(status).json(body);
    }
  }

  /**
   * Extracts the raw, untranslated message from any thrown value. This value
   * doubles as the i18n key and the `errorCode` returned to the client.
   */
  private extractMessage(exception: unknown): string {
    if (exception instanceof HttpException) {
      const res = exception.getResponse() as {
        message?: string | string[];
      };
      if (Array.isArray(res?.message)) {
        return res.message[0];
      }
      return res?.message || exception.message;
    }
    if (exception instanceof Error) {
      return exception.message;
    }
    if (typeof exception === 'string') {
      return exception;
    }
    return 'An unexpected error occurred';
  }

  private async translate(key: string, lang?: string): Promise<string> {
    if (!this.i18n || !key) {
      return key;
    }
    try {
      return await this.i18n.translate(key, { lang });
    } catch {
      return key;
    }
  }
}
