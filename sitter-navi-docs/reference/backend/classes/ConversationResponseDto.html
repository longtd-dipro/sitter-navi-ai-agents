import { ApiProperty } from '@nestjs/swagger';
import { ConversationType } from '../../enums/conversation-type.enum';

export class ConversationResponseDto {
  @ApiProperty({ example: '1234567890123456789', description: 'Group conversation Snowflake ID' })
  id: string;

  @ApiProperty({ example: 'Project Team', description: 'Group name' })
  name: string;

  @ApiProperty({
    enum: ConversationType,
    example: ConversationType.GROUP,
    description: 'Conversation type: direct or group',
  })
  type: ConversationType;

  @ApiProperty({
    example: 'https://s3.amazonaws.com/bucket/avatar.png',
    description: 'Avatar presigned URL',
    nullable: true,
  })
  avatarUrl: string | null;

  @ApiProperty({
    example: 'Hello everyone! Let me share...',
    description: 'Last message preview (max 100 chars)',
    nullable: true,
  })
  lastMessagePreview: string | null;

  @ApiProperty({ example: 3, description: 'Unread message count for the requesting user' })
  unreadCount: number;

  @ApiProperty({
    example: '2025-01-15T10:30:00.000Z',
    description: 'Timestamp of the last message',
    nullable: true,
  })
  lastMessageAt: Date | null;
}
