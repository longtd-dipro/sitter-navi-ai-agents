import { UserEntity } from '@modules/user/user.entity';
import { UserRepository } from '@modules/user/user.repository';
import { HttpException, HttpStatus, Injectable, UnauthorizedException } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { JwtService } from '@nestjs/jwt';
import { EncryptionUtil } from '@utils';
import { AdminLoginRequestDto, LoginRequestDto, LoginResponseDto } from '../dto/login';
import { RegisterDto } from '../dto/register';
import { SocialLoginDto, SocialLoginResponseDto } from '../dto/social';
import { MyInfoResponseDto } from '../dto/me';
import { ResetPasswordDto, ResetPasswordResponseDto } from '../dto/reset-password';
import { OtpSentResponseDto } from '../dto/otp';
import { SocialService } from './social.service';
import { RefreshTokenService } from './refresh-token.service';
import { OtpService } from './otp.service';
import { RequireAction } from '../enum';
import { SocialProvider } from '@modules/user/social-account.entity';
import { RoleType } from '@common/enums';
import { ParentProfileService } from '@modules/parent-profile/parent-profile.service';
import { UpsertParentProfileInfoDto } from '@modules/parent-profile/dtos/request';

export class IToken {
  accessToken: string;
}

/**
 * Provider identity carried by the short-lived token issued from
 * {@link AuthService.socialLogin} for an unknown social account. The
 * register flow trusts these claims (the token is server-signed) instead
 * of re-verifying the provider ID token.
 */
export interface SocialRegisterIdentity {
  provider: SocialProvider;
  providerUserId: string;
}

/**
 * AuthService — Complex_Business_Service for the Auth module.
 *
 * Owns registration, login (user + admin), JWT issuance and account-state
 * validation. OTP concerns have been extracted to {@link OtpService}.
 *
 * Persistence is delegated to {@link UserRepository} — owns the `users`
 * table (cross-module DI; `UserModule` exports `UserRepository`).
 */
@Injectable()
export class AuthService {
  constructor(
    public readonly jwtService: JwtService,
    public readonly config: ConfigService,
    private readonly userRepository: UserRepository,
    private readonly socialService: SocialService,
    private readonly refreshTokenService: RefreshTokenService,
    private readonly otpService: OtpService,
    private readonly parentProfileService: ParentProfileService,
  ) {}

  async userRegister(dto: RegisterDto, social?: SocialRegisterIdentity): Promise<LoginResponseDto> {
    const existUser = await this.userRepository.findByEmail(dto.email);
    if (existUser) {
      throw new HttpException(
        {
          status: HttpStatus.CONFLICT,
          messageCode: 'auth.email_used',
        },
        HttpStatus.CONFLICT,
      );
    }
    const createdUser = await this.createUser(dto, social);
    const { accessToken } = this.createToken(createdUser);
    const { refreshToken, refreshTokenExpiresAt } = await this.refreshTokenService.issueForUser(
      createdUser.id,
    );

    return {
      accessToken,
      refreshToken,
      refreshTokenExpiresAt,
      data: createdUser,
    };
  }

  async createUser(
    userRegisterDto: RegisterDto,
    social?: SocialRegisterIdentity,
  ): Promise<UserEntity> {
    if (userRegisterDto.password) {
      userRegisterDto.password = EncryptionUtil.generateHash(userRegisterDto.password);
    }

    if (social) {
      const { provider, providerUserId } = social;
      const existUser = await this.userRepository.findSocialProfile(provider, providerUserId);
      if (existUser) {
        throw new HttpException(
          {
            status: HttpStatus.CONFLICT,
            messageCode: 'auth.social_account_used',
          },
          HttpStatus.CONFLICT,
        );
      }
      return this.userRepository.createUserWithSocialAccount(
        userRegisterDto,
        provider,
        providerUserId,
      );
    }

    return this.userRepository.createUser(userRegisterDto);
  }

  createToken(data: UserEntity): IToken {
    const { id, role, email } = data;

    return {
      accessToken: this.jwtService.sign({
        id,
        email,
        role,
      }),
    };
  }

  /**
   * Login step 1. Verifies the credentials, then branches by role:
   *  - **parent**: an e-mail OTP second factor is required, so instead of a
   *    session this returns the OTP challenge `{ identifier, expiredAt }`. The
   *    client completes login via `POST /auth/otp/verify` (see
   *    {@link OtpService.verifyOtp} → `LOGIN_VERIFY` branch).
   *  - **other roles** (admin/caregiver): a session is issued immediately.
   */
  async login(dto: LoginRequestDto): Promise<OtpSentResponseDto | LoginResponseDto> {
    const user = await this.userRepository.findActiveLoginCandidate(dto.email);
    const isPasswordValid = await EncryptionUtil.validateHash(dto.password, user?.password || '');
    if (!isPasswordValid || !user) {
      throw new UnauthorizedException('auth.user_not_valid');
    }

    if (user.isLocked) {
      throw new UnauthorizedException('auth.account_locked');
    }

    if (user.role === RoleType.PARENT) {
      return this.otpService.sendLoginOtp(user);
    }

    return this.issueSession(user);
  }

  /**
   * Mint a full login session (access + refresh) for an already-validated
   * user and shape it as a {@link LoginResponseDto}.
   */
  private async issueSession(user: UserEntity): Promise<LoginResponseDto> {
    const { accessToken } = this.createToken(user);
    const { refreshToken, refreshTokenExpiresAt } = await this.refreshTokenService.issueForUser(
      user.id,
    );

    return {
      accessToken,
      refreshToken,
      refreshTokenExpiresAt,
      data: user,
    };
  }

  async adminLogin(dto: AdminLoginRequestDto): Promise<LoginResponseDto> {
    const userAdmin = await this.userRepository.findActiveAdminByEmail(dto.email);

    const isPasswordValid = await EncryptionUtil.validateHash(
      dto.password,
      userAdmin?.password || '',
    );

    if (!isPasswordValid || !userAdmin) {
      throw new UnauthorizedException('auth.admin_not_valid');
    }

    if (userAdmin.isLocked) {
      throw new UnauthorizedException('auth.account_locked');
    }

    // Return JWT directly (OTP flow disabled for now)
    const { accessToken } = this.createToken(userAdmin);
    const { refreshToken, refreshTokenExpiresAt } = await this.refreshTokenService.issueForUser(
      userAdmin.id,
    );
    return {
      accessToken,
      refreshToken,
      refreshTokenExpiresAt,
      data: userAdmin,
    };
  }

  /**
   * Final step of the "forgot password" flow. Trusts the short-lived
   * `token` issued by {@link OtpService.verifyOtp} (server-signed, carries
   * the OTP-verified e-mail) instead of re-checking the OTP. Sets the new
   * password and revokes every refresh token so all existing sessions are
   * logged out.
   *
   * Returns `success: true` regardless of whether the e-mail maps to a real
   * account, so the endpoint does not leak account existence.
   */
  async resetPassword(dto: ResetPasswordDto): Promise<ResetPasswordResponseDto> {
    let email: string | undefined;
    try {
      const payload = this.jwtService.verify<{ email?: string }>(dto.token);
      email = payload?.email;
    } catch {
      throw new UnauthorizedException('auth.reset_token_invalid');
    }

    if (!email) {
      throw new UnauthorizedException('auth.reset_token_invalid');
    }

    const user = await this.userRepository.findByEmail(email);
    if (user) {
      const hashedPassword = EncryptionUtil.generateHash(dto.password);
      await this.userRepository.updatePassword(user.id, hashedPassword);
      await this.refreshTokenService.revokeAllForUser(user.id);
    }

    return { success: true };
  }

  /**
   * Parent onboarding step ("đăng ký thông tin trẻ"). The registration flow
   * only creates the `user`; the `parent_profile` row is born here, the first
   * place that owns the children data. Idempotent — resubmitting updates the
   * same row. Returns the refreshed {@link getUserInfo} so the client sees
   * `requireAction` flip to `NONE` once the profile exists.
   */
  async registerParentInfo(userId: string, dto: UpsertParentProfileInfoDto): Promise<MyInfoResponseDto> {
    await this.parentProfileService.upsertByUserId(userId, dto);
    return this.getUserInfo(userId);
  }

  async getUserInfo(id: string | null): Promise<MyInfoResponseDto> {
    let user: UserEntity | null = null;
    if (id) {
      user = await this.userRepository.findById(id);
    }

    const requireAction = await this.resolveRequireAction(id, user);
    let userInfo: MyInfoResponseDto['info'] = null;
    if (user) {
      userInfo = {
        id: user.id,
        role: user.role,
        email: user.email,
      };
    }

    return {
      info: userInfo,
      requireAction,
    };
  }

  /**
   * Decide whether the client still owes an onboarding step:
   *  - no account yet (anonymous / social sign-up not finished) → must register;
   *  - a **parent** without a `parent_profile` row → must complete the children
   *    info step (see {@link registerParentInfo});
   *  - otherwise nothing is required.
   *
   * Sitter onboarding has its own (later) flow, so other roles return `NONE`.
   */
  private async resolveRequireAction(
    id: string | null,
    user: UserEntity | null,
  ): Promise<RequireAction> {
    if (!id || !user) {
      return RequireAction.REGISTER_INFO;
    }

    if (user.role === RoleType.PARENT) {
      const hasProfile = await this.parentProfileService.existsByUserId(user.id);
      return hasProfile ? RequireAction.NONE : RequireAction.REGISTER_INFO;
    }

    return RequireAction.NONE;
  }

  async socialLogin(dto: SocialLoginDto): Promise<SocialLoginResponseDto> {
    const { provider, idToken } = dto;
    const { providerUserId } = await this.socialService.verifyIdToken(provider, idToken);

    const user = await this.userRepository.findSocialProfile(provider, providerUserId);
    const info = await this.getUserInfo(user?.id || null);
    const isAuthenticated = info.requireAction === RequireAction.NONE && !!user;

    // Unknown social account: issue a short-lived token carrying the provider
    // identity so the client can complete registration (bypass OTP for social).
    if (!isAuthenticated) {
      return {
        accessToken: this.jwtService.sign({ provider, providerUserId }),
        refreshToken: null,
        refreshTokenExpiresAt: null,
        ...info,
      };
    }

    const { accessToken } = this.createToken(user!);
    const { refreshToken, refreshTokenExpiresAt } = await this.refreshTokenService.issueForUser(
      user!.id,
    );

    return {
      accessToken,
      refreshToken,
      refreshTokenExpiresAt,
      ...info,
    };
  }
}
