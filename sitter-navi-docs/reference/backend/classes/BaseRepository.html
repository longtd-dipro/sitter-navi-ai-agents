Description

Optional abstract base class that module-level repositories MAY extend to inherit a default implementation of the IBaseRepository contract. Subclasses only have to provide:

  1. A TypeORM Repository via the repo property (typically injected with @InjectRepository(Entity)).
  2. An entityName string that becomes the i18n prefix for findByIdOrFail errors (for example, 'user' produces the i18n key 'user.not_found').

Module repositories are free to override any method to add module-specific behavior, or to skip this base class entirely and implement IBaseRepository directly.

Example:

Example :
@Injectable()
export class UserRepository extends BaseRepository<UserEntity> {
  protected readonly entityName = 'user';

  constructor(
    @InjectRepository(UserEntity)
    protected readonly repo: Repository<UserEntity>,
  ) {
    super();
  }

  findByEmail(email: string) {
    return this.repo.findOneBy({ email });
  }
}

import { NotFoundException } from '@nestjs/common';
import { DeepPartial, FindManyOptions, FindOptionsWhere, ObjectLiteral, Repository } from 'typeorm';

/**
 * Standard contract that every Complex_Business_Module repository must
 * satisfy. The methods listed here are the minimum surface area that
 * service layers can rely on when interacting with persistence.
 *
 * Implementations:
 *  - MUST NOT enforce authorization (that is the service/guard layer's job).
 *  - MUST NOT contain business validation logic.
 *  - MUST NOT emit domain events.
 *
 * See: Requirements 12.1, 12.2, 12.3, 12.5
 */
export interface IBaseRepository<TEntity extends ObjectLiteral> {
  /**
   * Look up an entity by its primary id. Returns `null` when the id does
   * not exist (no exception thrown).
   */
  findById(id: string): Promise<TEntity | null>;

  /**
   * Same as {@link findById} but throws {@link NotFoundException} with an
   * i18n key (`{entity}.not_found`) when the id does not exist.
   */
  findByIdOrFail(id: string): Promise<TEntity>;

  /**
   * Persist a partial entity. Implementations may either `INSERT` a new
   * row (when `id` is absent) or `UPDATE` an existing row.
   */
  save(data: Partial<TEntity>): Promise<TEntity>;

  /**
   * Soft-delete the entity with the given id (sets `deleted_at`).
   */
  softDelete(id: string): Promise<void>;

  /**
   * Count rows that match the given options. The options shape mirrors
   * TypeORM's {@link FindManyOptions}.
   */
  count(options?: FindManyOptions<TEntity>): Promise<number>;
}

/**
 * Optional abstract base class that module-level repositories MAY extend
 * to inherit a default implementation of the {@link IBaseRepository}
 * contract. Subclasses only have to provide:
 *
 *   1. A TypeORM {@link Repository} via the `repo` property
 *      (typically injected with `@InjectRepository(Entity)`).
 *   2. An `entityName` string that becomes the i18n prefix for
 *      `findByIdOrFail` errors (for example, `'user'` produces the
 *      i18n key `'user.not_found'`).
 *
 * Module repositories are free to override any method to add
 * module-specific behavior, or to skip this base class entirely and
 * implement {@link IBaseRepository} directly.
 *
 * Example:
 *
 * ```ts
 * @Injectable()
 * export class UserRepository extends BaseRepository<UserEntity> {
 *   protected readonly entityName = 'user';
 *
 *   constructor(
 *     @InjectRepository(UserEntity)
 *     protected readonly repo: Repository<UserEntity>,
 *   ) {
 *     super();
 *   }
 *
 *   findByEmail(email: string) {
 *     return this.repo.findOneBy({ email });
 *   }
 * }
 * ```
 */
export abstract class BaseRepository<
  TEntity extends ObjectLiteral & { id: string },
> implements IBaseRepository<TEntity> {
  /** TypeORM repository injected by the subclass. */
  protected abstract readonly repo: Repository<TEntity>;

  /**
   * i18n prefix used for not-found errors. Subclasses MUST override this
   * with the aggregate name (e.g. `'user'`, `'insurance_contract'`).
   */
  protected abstract readonly entityName: string;

  async findById(id: string): Promise<TEntity | null> {
    return this.repo.findOneBy({ id } as FindOptionsWhere<TEntity>);
  }

  async findByIdOrFail(id: string): Promise<TEntity> {
    const entity = await this.findById(id);
    if (!entity) {
      throw new NotFoundException(`${this.entityName}.not_found`);
    }
    return entity;
  }

  async save(data: Partial<TEntity>): Promise<TEntity> {
    const entity = this.repo.create(data as DeepPartial<TEntity>);
    return this.repo.save(entity);
  }

  async softDelete(id: string): Promise<void> {
    await this.repo.softDelete(id);
  }

  async count(options?: FindManyOptions<TEntity>): Promise<number> {
    return this.repo.count(options);
  }
}
