import { Logger } from '@nestjs/common';
import { LockAdapter } from '../interfaces';

/**
 * Redis distributed lock adapter.
 *
 * Uses SET key value NX PX ttl pattern (single-instance Redlock).
 * For production multi-node Redis, consider extending with full Redlock algorithm.
 *
 * Requires an ioredis instance to be provided via configure().
 */
export class RedisLockAdapter implements LockAdapter {
  readonly name = 'redis';
  private readonly logger = new Logger('RedisLockAdapter');

  /** Unique value per lock to ensure only the holder can release */
  private activeLocks = new Map<string, string>();

  private redisClient: any | null = null;

  /**
   * Configure the adapter with an ioredis client instance.
   * Must be called before acquire/release.
   */
  configure(redisClient: any): void {
    this.redisClient = redisClient;
  }

  async acquire(key: string, ttlMs = 30000): Promise<boolean> {
    if (!this.redisClient) return false;

    const lockKey = this.buildKey(key);
    const lockValue = this.generateValue();

    try {
      // SET key value NX PX ttlMs — atomic set-if-not-exists with expiry
      const result = await this.redisClient.set(
        lockKey,
        lockValue,
        'PX',
        ttlMs,
        'NX',
      );

      if (result === 'OK') {
        this.activeLocks.set(key, lockValue);
        this.logger.debug(`Lock acquired: ${key} (ttl: ${ttlMs}ms)`);
        return true;
      }

      return false;
    } catch (error) {
      this.logger.error(`Failed to acquire lock: ${key}`, (error as Error).stack);
      return false;
    }
  }

  async release(key: string): Promise<boolean> {
    if (!this.redisClient) return false;

    const lockValue = this.activeLocks.get(key);
    if (!lockValue) return false;

    const lockKey = this.buildKey(key);

    try {
      // Lua script: only delete if the value matches (prevents releasing someone else's lock)
      const script = `
        if redis.call("get", KEYS[1]) == ARGV[1] then
          return redis.call("del", KEYS[1])
        else
          return 0
        end
      `;

      const result = await this.redisClient.eval(script, 1, lockKey, lockValue);
      this.activeLocks.delete(key);

      if (result === 1) {
        this.logger.debug(`Lock released: ${key}`);
        return true;
      }

      // Lock was already expired or taken by someone else
      this.logger.warn(`Lock release failed (not held): ${key}`);
      return false;
    } catch (error) {
      this.logger.error(`Failed to release lock: ${key}`, (error as Error).stack);
      this.activeLocks.delete(key);
      return false;
    }
  }

  async isHealthy(): Promise<boolean> {
    if (!this.redisClient) return false;

    try {
      const status = this.redisClient.status;
      if (status !== 'ready') return false;
      await this.redisClient.ping();
      return true;
    } catch {
      return false;
    }
  }

  async shutdown(): Promise<void> {
    // Release all active locks
    for (const key of this.activeLocks.keys()) {
      await this.release(key);
    }
  }

  private buildKey(key: string): string {
    return `dlock:${key}`;
  }

  private generateValue(): string {
    // Unique value = timestamp + random to prevent collisions
    return `${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
  }
}
