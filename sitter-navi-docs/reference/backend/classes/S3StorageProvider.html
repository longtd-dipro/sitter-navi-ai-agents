import {
  DeleteObjectCommand,
  DeleteObjectsCommand,
  GetObjectCommand,
  PutObjectCommand,
  S3Client,
  S3ClientConfig,
} from '@aws-sdk/client-s3';
import { createPresignedPost } from '@aws-sdk/s3-presigned-post';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';
import { Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { generateUUID } from '@foundation';
import { StorageProvider } from '../interfaces/storage-provider.interface';

export class S3StorageProvider implements StorageProvider {
  private logger = new Logger(S3StorageProvider.name);
  private s3: S3Client;
  private bucket: string;
  private region: string;
  /** Prefix for private objects — all presigned upload/get/delete flows use this. */
  private prefix: string;
  /** Prefix for public objects — readable directly via {@link getPublicUrl} without signing. */
  private publicPrefix: string;

  constructor(private readonly configService: ConfigService) {
    this.region = this.configService.get<string>('aws.region') ?? 'ap-northeast-1';
    this.s3 = new S3Client(this.buildClientConfig());
    this.prefix = this.configService.get<string>('aws.privatePrefix') ?? 'pr';
    this.publicPrefix = this.configService.get<string>('aws.publicPrefix') ?? 'pu';

    this.bucket = this.getRequiredConfig('aws.bucket');
    this.logger.debug(
      `S3StorageProvider initialized with bucket: ${this.bucket}, region: ${this.region}`,
    );
  }

  /**
   * Builds a direct, unsigned, permanent URL for a public object.
   *
   * Only objects stored under the public prefix (and covered by the bucket's
   * public-read policy) are actually reachable this way. Use this instead of a
   * presigned URL for files that don't need access control.
   */
  getPublicUrl(key: string): string {
    const configuredBase = this.configService.get<string>('aws.publicBaseUrl');
    const endpoint = this.configService.get<string>('aws.endpoint');
    // Local/S3-compatible (e.g. RustFS) uses path-style: <endpoint>/<bucket>/<key>.
    // Real AWS S3 uses virtual-hosted style: https://<bucket>.s3.<region>.amazonaws.com/<key>.
    const base =
      configuredBase ??
      (endpoint
        ? `${endpoint}/${this.bucket}`
        : `https://${this.bucket}.s3.${this.region}.amazonaws.com`);
    return `${base.replace(/\/+$/, '')}/${key}`;
  }

  /**
   * Builds the S3 client config.
   *
   * When `aws.endpoint` is set (local S3-compatible storage such as RustFS or
   * MinIO), it points the client at that endpoint, forces path-style URLs
   * (compatible servers don't support virtual-hosted-style buckets) and passes
   * explicit credentials. When unset, the client falls back to the standard AWS
   * default credential chain + real S3 endpoint — so production is unchanged.
   */
  private buildClientConfig(): S3ClientConfig {
    const endpoint = this.configService.get<string>('aws.endpoint');
    if (!endpoint) {
      return { region: this.region };
    }

    this.logger.debug(`S3StorageProvider using custom endpoint: ${endpoint}`);
    return {
      region: this.region,
      endpoint,
      forcePathStyle: true,
      credentials: {
        accessKeyId: this.getRequiredConfig('aws.accessKeyId'),
        secretAccessKey: this.getRequiredConfig('aws.secretAccessKey'),
      },
    };
  }

  private getRequiredConfig(key: string): string {
    const value = this.configService.get<string>(key);
    if (!value) {
      throw new Error(`Missing required config: ${key}`);
    }
    return value;
  }

  async getPresignedUploadUrl(
    filename: string,
    options?: { contentType?: string; expiresIn?: number },
  ): Promise<{
    uploadUrl: string;
    fileUrl: string;
    fields?: Record<string, string>;
  }> {
    return this.getPresignedPostUrl(filename, options);
  }

  async getPresignedPutUrl(
    filename: string,
    options?: { contentType?: string; expiresIn?: number },
  ): Promise<{ uploadUrl: string; fileUrl: string }> {
    const id = generateUUID();
    const fileExtension = filename.includes('.') ? filename.split('.').pop() : '';

    const key = fileExtension ? `${this.prefix}/${id}.${fileExtension}` : `${this.prefix}/${id}`;

    const expiresIn = options?.expiresIn ?? 3600;

    const command = new PutObjectCommand({
      Bucket: this.bucket,
      Key: key,
      ContentType: options?.contentType ?? 'application/octet-stream',
    });

    const uploadUrl = await getSignedUrl(this.s3, command, { expiresIn });

    return { uploadUrl, fileUrl: key };
  }

  async getPresignedPostUrl(
    filename: string,
    options?: {
      contentType?: string;
      expiresIn?: number;
    },
  ): Promise<{
    uploadUrl: string;
    fields: Record<string, string>;
    fileUrl: string;
  }> {
    const id = generateUUID();
    const fileExtension = filename.includes('.') ? filename.split('.').pop() : '';

    const key = fileExtension ? `${this.prefix}/${id}.${fileExtension}` : `${this.prefix}/${id}`;

    // Rule: max 10MB for images, 100MB for others
    let maxFileSize = 100 * 1024 * 1024;
    if (options?.contentType?.startsWith('image/')) {
      maxFileSize = 10 * 1024 * 1024;
    }

    const presignedPost = await createPresignedPost(this.s3, {
      Bucket: this.bucket,
      Key: key,
      Conditions: [
        ['content-length-range', 0, maxFileSize],
        ['starts-with', '$Content-Type', options?.contentType ?? ''],
      ],
      Fields: {
        'Content-Type': options?.contentType ?? 'application/octet-stream',
      },
      Expires: options?.expiresIn ?? 3600,
    });

    return { ...presignedPost, fileUrl: key, uploadUrl: presignedPost.url };
  }

  async deleteFile(key: string): Promise<void> {
    const command = new DeleteObjectCommand({
      Bucket: this.bucket,
      Key: key,
    });
    await this.s3.send(command);
  }

  async deleteFiles(keys: string[]): Promise<void> {
    if (keys.length === 0) return;

    const command = new DeleteObjectsCommand({
      Bucket: this.bucket,
      Delete: {
        Objects: keys.map((key) => ({ Key: key })),
      },
    });

    await this.s3.send(command);
  }

  async getFileUrl(key: string): Promise<string> {
    const command = new GetObjectCommand({
      Bucket: this.bucket,
      Key: key,
    });

    return await getSignedUrl(this.s3, command, { expiresIn: 60 * 5 });
  }

  async getPresignedGetUrls(
    keys: string[],
    expiresIn: number = 300,
  ): Promise<Record<string, string>> {
    const urls: Record<string, string> = {};
    await Promise.all(
      keys.map(async (key) => {
        const command = new GetObjectCommand({
          Bucket: this.bucket,
          Key: key,
        });
        urls[key] = await getSignedUrl(this.s3, command, { expiresIn });
      }),
    );
    return urls;
  }
}
