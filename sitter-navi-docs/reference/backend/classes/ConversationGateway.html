import {
  WebSocketGateway,
  WebSocketServer,
  OnGatewayConnection,
  OnGatewayDisconnect,
  SubscribeMessage,
  MessageBody,
  ConnectedSocket,
} from '@nestjs/websockets';
import { Inject, Logger } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { OnEvent } from '@nestjs/event-emitter';
import { Server, Socket } from 'socket.io';

import { RedisCacheService } from '@infrastructure/cache';

import { ConversationService } from './conversation.service';
import {
  GroupCreatedEvent,
  GroupDissolvedEvent,
  GroupUpdatedEvent,
  MemberAddedEvent,
  MemberLeftEvent,
  MemberRemovedEvent,
  MessageDeliveredEvent,
  MessageReadEvent,
  MessageSentEvent,
  NewMessageEvent,
} from './events/conversation.events';
import { PARTICIPANT_REPOSITORY, ParticipantRepository } from './repositories';

@WebSocketGateway({ namespace: '/conversations', cors: true })
export class ConversationGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer()
  server: Server;

  private readonly logger = new Logger(ConversationGateway.name);

  /** Maximum concurrent connections per user */
  private static readonly MAX_CONNECTIONS_PER_USER = 5;

  /** Maximum number of group rooms to join on initial connection */
  private static readonly MAX_INITIAL_ROOMS = 100;

  constructor(
    private readonly jwtService: JwtService,
    @Inject(PARTICIPANT_REPOSITORY)
    private readonly participantRepo: ParticipantRepository,
    private readonly redisCacheService: RedisCacheService,
    private readonly ConversationService: ConversationService,
  ) {}

  /**
   * Handle new WebSocket connection.
   *
   * 1. Validate JWT from auth handshake
   * 2. Check connection count (max 5, disconnect oldest if exceeded)
   * 3. Store connection in Redis hash
   * 4. Join all group rooms where user is an active participant
   *
   * Requirements: 8.1, 8.2, 8.3, 8.4, 8.5
   */
  async handleConnection(client: Socket): Promise<void> {
    try {
      // 1. Extract and validate JWT token from handshake
      const token = client.handshake?.auth?.token;
      if (!token) {
        this.logger.warn(`Connection rejected: no token provided (socket: ${client.id})`);
        client.emit('group:error', {
          code: 'UNAUTHORIZED',
          message: 'Authentication token is required',
        });
        client.disconnect(true);
        return;
      }

      let payload: any;
      try {
        payload = this.jwtService.verify(token);
      } catch (error) {
        this.logger.warn(`Connection rejected: invalid/expired token (socket: ${client.id})`);
        client.emit('group:error', {
          code: 'UNAUTHORIZED',
          message: 'Invalid or expired authentication token',
        });
        client.disconnect(true);
        return;
      }

      const userId = payload.id;
      if (!userId) {
        this.logger.warn(`Connection rejected: token missing user ID (socket: ${client.id})`);
        client.emit('group:error', {
          code: 'UNAUTHORIZED',
          message: 'Invalid token payload',
        });
        client.disconnect(true);
        return;
      }

      // Attach userId to socket data for later use
      client.data.userId = userId;

      // 2. Check connection count and enforce max 5 limit
      const redisKey = `user:${userId}:connections`;
      await this.enforceConnectionLimit(client, userId, redisKey);

      // 3. Store this connection in Redis hash (field: socketId, value: timestamp)
      const now = Date.now().toString();
      try {
        await this.redisHSet(redisKey, client.id, now);
      } catch (error) {
        this.logger.warn(
          `Failed to store connection in Redis for user ${userId}: ${(error as Error).message}`,
        );
      }

      // 4. Join group rooms where user is an active participant (limit to most recent)
      const participantRecords = await this.participantRepo.findActiveByUser(userId);
      const roomsToJoin = participantRecords.slice(0, ConversationGateway.MAX_INITIAL_ROOMS);

      await Promise.all(roomsToJoin.map((record) => client.join(`group:${record.conversationId}`)));

      this.logger.log(
        `User ${userId} connected (socket: ${client.id}, totalGroups: ${participantRecords.length}, roomsJoined: ${roomsToJoin.length})`,
      );
    } catch (error) {
      this.logger.error(
        `Unexpected error during connection: ${(error as Error).message}`,
        (error as Error).stack,
      );
      client.emit('group:error', {
        code: 'INTERNAL_ERROR',
        message: 'Connection failed due to an internal error',
      });
      client.disconnect(true);
    }
  }

  /**
   * Handle WebSocket disconnection.
   *
   * Remove the connection from Redis hash.
   * Socket.IO automatically handles leaving all rooms.
   *
   * Requirements: 8.4, 8.5
   */
  async handleDisconnect(client: Socket): Promise<void> {
    const userId = client.data.userId;
    if (!userId) {
      return;
    }

    const redisKey = `user:${userId}:connections`;
    try {
      await this.redisHDel(redisKey, client.id);
    } catch (error) {
      this.logger.warn(
        `Failed to remove connection from Redis for user ${userId}: ${(error as Error).message}`,
      );
    }

    this.logger.log(`User ${userId} disconnected (socket: ${client.id})`);
  }

  // ─────────────────────────────────────────────────────────────────────────
  // Server-side emit methods (called from service/event handlers)
  // ─────────────────────────────────────────────────────────────────────────

  /**
   * Emit an event to all sockets in a group room.
   */
  emitToGroup(groupId: string, event: string, payload: any): void {
    this.server.to(`group:${groupId}`).emit(event, payload);
  }

  /**
   * Emit an event to all sockets of a specific user.
   */
  async emitToUser(userId: string, event: string, payload: any): Promise<void> {
    const redisKey = `user:${userId}:connections`;
    try {
      const connections = await this.redisHGetAll(redisKey);
      if (connections) {
        const socketIds = Object.keys(connections);
        for (const socketId of socketIds) {
          this.server.to(socketId).emit(event, payload);
        }
      }
    } catch {
      // Fallback: if Redis is unavailable, we can't target specific user sockets
      this.logger.warn(`Failed to emit to user ${userId}: Redis unavailable`);
    }
  }

  /**
   * Join a specific user's sockets to a group room.
   * Used when a user is added to a group while already connected.
   */
  async joinUserToRoom(userId: string, groupId: string): Promise<void> {
    const redisKey = `user:${userId}:connections`;
    try {
      const connections = await this.redisHGetAll(redisKey);
      if (connections) {
        const socketIds = Object.keys(connections);
        for (const socketId of socketIds) {
          const sockets = await this.server.in(socketId).fetchSockets();
          for (const socket of sockets) {
            await socket.join(`group:${groupId}`);
          }
        }
      }
    } catch {
      this.logger.warn(`Failed to join user ${userId} to room group:${groupId}`);
    }
  }

  /**
   * Remove a specific user's sockets from a group room.
   * Used when a user is removed from or leaves a group.
   */
  async removeUserFromRoom(userId: string, groupId: string): Promise<void> {
    const redisKey = `user:${userId}:connections`;
    try {
      const connections = await this.redisHGetAll(redisKey);
      if (connections) {
        const socketIds = Object.keys(connections);
        for (const socketId of socketIds) {
          const sockets = await this.server.in(socketId).fetchSockets();
          for (const socket of sockets) {
            socket.leave(`group:${groupId}`);
          }
        }
      }
    } catch {
      this.logger.warn(`Failed to remove user ${userId} from room group:${groupId}`);
    }
  }

  // ─────────────────────────────────────────────────────────────────────────
  // Socket.IO event handlers (Client → Server)
  // ─────────────────────────────────────────────────────────────────────────

  /**
   * Handle conversation-opened event from client.
   * Resets the unread count for the user in the specified group.
   *
   * Requirements: 9.6
   */
  @SubscribeMessage('group:conversation-opened')
  async handleConversationOpened(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { groupId: string },
  ): Promise<void> {
    const userId = client.data.userId;
    if (!userId || !payload?.groupId) {
      return;
    }

    try {
      await this.ConversationService.resetUnreadCount(userId, payload.groupId);
      // Also mark all unread messages in this conversation as read
      await this.ConversationService.markRead(userId, payload.groupId);

      // Emit updated total unread count to this user
      const { totalUnreadCount } = await this.ConversationService.getTotalUnreadCount(userId);
      await this.emitToUser(userId, 'message:total-unread-updated', {
        totalUnreadCount,
      });
    } catch (error) {
      this.logger.warn(
        `Failed to reset unread count for user ${userId} in group ${payload.groupId}: ${(error as Error).message}`,
      );
    }
  }

  /**
   * Handle message:delivered event from client.
   * Called when a recipient's device acknowledges receiving a message.
   *
   * Requirements: real-time delivery acknowledgment
   */
  @SubscribeMessage('message:delivered')
  async handleMessageDelivered(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { messageId: string; groupId: string },
  ): Promise<void> {
    const userId = client.data.userId;
    if (!userId || !payload?.messageId || !payload?.groupId) {
      return;
    }

    try {
      await this.ConversationService.markDelivered(payload.messageId, userId, payload.groupId);
    } catch (error) {
      this.logger.warn(
        `Failed to mark message ${payload.messageId} as delivered for user ${userId}: ${(error as Error).message}`,
      );
    }
  }

  /**
   * Handle message:read event from client.
   * Called when a user reads one or more messages in a conversation.
   *
   * Requirements: real-time read acknowledgment
   */
  @SubscribeMessage('message:read')
  async handleMessageRead(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { groupId: string; messageIds?: string[] },
  ): Promise<void> {
    const userId = client.data.userId;
    if (!userId || !payload?.groupId) {
      return;
    }

    try {
      await this.ConversationService.markRead(userId, payload.groupId, payload.messageIds);
    } catch (error) {
      this.logger.warn(
        `Failed to mark messages as read for user ${userId} in group ${payload.groupId}: ${(error as Error).message}`,
      );
    }
  }

  // ─────────────────────────────────────────────────────────────────────────
  // Domain event listeners (Service → Gateway via EventEmitter)
  // ─────────────────────────────────────────────────────────────────────────

  /**
   * Handle GroupCreatedEvent.
   * Emit group:created to all participants and join their sockets to the new room.
   *
   * Requirements: 1.5
   */
  @OnEvent(GroupCreatedEvent.eventName, { async: true })
  async handleGroupCreated(event: GroupCreatedEvent): Promise<void> {
    this.logger.debug(`Handling GroupCreatedEvent for group ${event.groupId}`);

    // Join all participants' sockets to the new room (parallel)
    await Promise.all(event.participants.map((p) => this.joinUserToRoom(p.userId, event.groupId)));

    // Emit group:created to all participants (parallel)
    await Promise.all(
      event.participants.map((participant) =>
        this.emitToUser(participant.userId, 'group:created', {
          groupId: event.groupId,
          groupName: event.groupName,
          participants: event.participants,
        }),
      ),
    );
  }

  /**
   * Handle MemberAddedEvent.
   * Emit group:member-added to the room and join the new member's sockets.
   *
   * Requirements: 2.4, 2.5
   */
  @OnEvent(MemberAddedEvent.eventName, { async: true })
  async handleMemberAdded(event: MemberAddedEvent): Promise<void> {
    this.logger.debug(
      `Handling MemberAddedEvent: user ${event.userId} added to group ${event.groupId}`,
    );

    // Join the new member's sockets to the room
    await this.joinUserToRoom(event.userId, event.groupId);

    // Emit group:member-added to the room (including the new member)
    this.emitToGroup(event.groupId, 'group:member-added', {
      groupId: event.groupId,
      userId: event.userId,
      displayName: event.displayName,
      addedBy: event.addedBy,
    });

    // Emit group:conversation-list-updated to all participants
    this.emitToGroup(event.groupId, 'group:conversation-list-updated', {
      groupId: event.groupId,
    });
  }

  /**
   * Handle MemberRemovedEvent.
   * Emit group:member-removed to the room and remove the member's sockets from the room.
   *
   * Requirements: 2.4
   */
  @OnEvent(MemberRemovedEvent.eventName, { async: true })
  async handleMemberRemoved(event: MemberRemovedEvent): Promise<void> {
    this.logger.debug(
      `Handling MemberRemovedEvent: user ${event.userId} removed from group ${event.groupId}`,
    );

    // Emit group:member-removed to the room (including the removed member, before leaving)
    this.emitToGroup(event.groupId, 'group:member-removed', {
      groupId: event.groupId,
      userId: event.userId,
      removedBy: event.removedBy,
    });

    // Also notify the removed user directly (in case they're not in the room anymore)
    await this.emitToUser(event.userId, 'group:member-removed', {
      groupId: event.groupId,
      userId: event.userId,
      removedBy: event.removedBy,
    });

    // Remove the member's sockets from the room
    await this.removeUserFromRoom(event.userId, event.groupId);

    // Emit group:conversation-list-updated to remaining participants
    this.emitToGroup(event.groupId, 'group:conversation-list-updated', {
      groupId: event.groupId,
    });
  }

  /**
   * Handle MemberLeftEvent.
   * Emit group:member-left to the room and remove the member's sockets from the room.
   *
   * Requirements: 7.2
   */
  @OnEvent(MemberLeftEvent.eventName, { async: true })
  async handleMemberLeft(event: MemberLeftEvent): Promise<void> {
    this.logger.debug(`Handling MemberLeftEvent: user ${event.userId} left group ${event.groupId}`);

    // Emit group:member-left to the room (before removing the member)
    this.emitToGroup(event.groupId, 'group:member-left', {
      groupId: event.groupId,
      userId: event.userId,
    });

    // Also notify the leaving user directly
    await this.emitToUser(event.userId, 'group:member-left', {
      groupId: event.groupId,
      userId: event.userId,
    });

    // Remove the member's sockets from the room
    await this.removeUserFromRoom(event.userId, event.groupId);

    // Emit group:conversation-list-updated to remaining participants
    this.emitToGroup(event.groupId, 'group:conversation-list-updated', {
      groupId: event.groupId,
    });
  }

  /**
   * Handle GroupUpdatedEvent.
   * Emit group:updated to the room.
   *
   * Requirements: 6.1, 6.2
   */
  @OnEvent(GroupUpdatedEvent.eventName, { async: true })
  async handleGroupUpdated(event: GroupUpdatedEvent): Promise<void> {
    this.logger.debug(`Handling GroupUpdatedEvent for group ${event.groupId}`);

    this.emitToGroup(event.groupId, 'group:updated', {
      groupId: event.groupId,
      name: event.name,
      avatarFileKey: event.avatarFileKey,
    });

    // Emit group:conversation-list-updated to all participants
    this.emitToGroup(event.groupId, 'group:conversation-list-updated', {
      groupId: event.groupId,
    });
  }

  /**
   * Handle GroupDissolvedEvent.
   * Emit group:dissolved to all participants.
   *
   * Requirements: 7.5
   */
  @OnEvent(GroupDissolvedEvent.eventName, { async: true })
  async handleGroupDissolved(event: GroupDissolvedEvent): Promise<void> {
    this.logger.debug(`Handling GroupDissolvedEvent for group ${event.groupId}`);

    // Emit group:dissolved to all participants (parallel)
    await Promise.all(
      event.participantIds.map((participantId) =>
        this.emitToUser(participantId, 'group:dissolved', {
          groupId: event.groupId,
        }),
      ),
    );
  }

  /**
   * Handle NewMessageEvent.
   * Broadcast group:new-message to the room and emit group:conversation-list-updated.
   *
   * Requirements: 3.4, 9.7
   */
  @OnEvent(NewMessageEvent.eventName, { async: true })
  async handleNewMessage(event: NewMessageEvent): Promise<void> {
    this.logger.debug(
      `Handling NewMessageEvent: message ${event.messageId} in group ${event.groupId}`,
    );

    // Broadcast group:new-message to the room
    this.emitToGroup(event.groupId, 'group:new-message', {
      messageId: event.messageId,
      groupId: event.groupId,
      senderId: event.senderId,
      senderName: event.senderName,
      content: event.content,
      messageType: event.messageType,
      attachments: event.attachments,
      timestamp: event.timestamp,
    });

    // Emit group:conversation-list-updated to all participants in the group
    // This notifies clients to refresh their conversation list (new unread count, preview)
    const lastMessagePreview = event.content
      ? event.content.length > 100
        ? event.content.substring(0, 100)
        : event.content
      : null;

    this.emitToGroup(event.groupId, 'group:conversation-list-updated', {
      groupId: event.groupId,
      lastMessagePreview,
      senderId: event.senderId,
      senderName: event.senderName,
      messageType: event.messageType,
      timestamp: event.timestamp,
    });

    // Emit message:total-unread-updated to all recipients (not sender)
    // Uses single batch query instead of N individual queries
    try {
      const participants = await this.participantRepo.findActiveByGroup(event.groupId);
      const recipientIds = participants
        .filter((p) => p.userId !== event.senderId)
        .map((p) => p.userId);

      if (recipientIds.length > 0) {
        // Batch calculate total unread for all recipients in one query via service
        const unreadMap = await this.ConversationService.getBatchTotalUnreadCounts(recipientIds);

        // Emit in parallel
        await Promise.all(
          recipientIds.map(async (userId) => {
            const totalUnreadCount = unreadMap.get(userId) ?? 0;
            await this.emitToUser(userId, 'message:total-unread-updated', {
              totalUnreadCount,
            });
          }),
        );
      }
    } catch (error) {
      this.logger.warn(
        `Failed to emit total-unread-updated for group ${event.groupId}: ${(error as Error).message}`,
      );
    }
  }

  /**

  // ─────────────────────────────────────────────────────────────────────────
  // Private helpers
  // ─────────────────────────────────────────────────────────────────────────

  /**
   * Enforce the max connection limit per user.
   * If the user already has MAX_CONNECTIONS_PER_USER connections,
   * disconnect the oldest one (earliest timestamp).
   */
  private async enforceConnectionLimit(
    client: Socket,
    userId: string,
    redisKey: string,
  ): Promise<void> {
    try {
      const connections = await this.redisHGetAll(redisKey);
      if (!connections) return;

      const entries = Object.entries(connections);
      if (entries.length < ConversationGateway.MAX_CONNECTIONS_PER_USER) {
        return;
      }

      // Sort by timestamp ascending (oldest first)
      entries.sort((a, b) => parseInt(a[1], 10) - parseInt(b[1], 10));

      // Disconnect oldest connections until we're under the limit
      const toDisconnect = entries.slice(
        0,
        entries.length - ConversationGateway.MAX_CONNECTIONS_PER_USER + 1,
      );

      for (const [socketId] of toDisconnect) {
        // Remove from Redis
        await this.redisHDel(redisKey, socketId);

        // Disconnect the socket
        const sockets = await this.server.in(socketId).fetchSockets();
        for (const oldSocket of sockets) {
          oldSocket.emit('group:error', {
            code: 'CONNECTION_REPLACED',
            message: 'Connection replaced by a newer connection',
          });
          oldSocket.disconnect(true);
        }

        this.logger.log(
          `Disconnected oldest socket ${socketId} for user ${userId} (connection limit exceeded)`,
        );
      }
    } catch (error) {
      this.logger.warn(
        `Failed to enforce connection limit for user ${userId}: ${(error as Error).message}`,
      );
    }
  }

  // ─────────────────────────────────────────────────────────────────────────
  // Redis Hash helpers (wrapping RedisCacheService for hash operations)
  // ─────────────────────────────────────────────────────────────────────────

  /**
   * Set a field in a Redis hash.
   * Uses the underlying Redis instance via eval-like approach with setValue.
   * Since RedisCacheService doesn't expose hash commands directly,
   * we store as a JSON-encoded value under the key.
   */
  private async redisHSet(key: string, field: string, value: string): Promise<void> {
    const existing = await this.getConnectionsMap(key);
    existing[field] = value;
    await this.redisCacheService.setValue(key, JSON.stringify(existing));
  }

  /**
   * Delete a field from a Redis hash (stored as JSON).
   */
  private async redisHDel(key: string, field: string): Promise<void> {
    const existing = await this.getConnectionsMap(key);
    delete existing[field];
    if (Object.keys(existing).length === 0) {
      await this.redisCacheService.clearCacheByKey(key);
    } else {
      await this.redisCacheService.setValue(key, JSON.stringify(existing));
    }
  }

  /**
   * Get all fields from a Redis hash (stored as JSON).
   */
  private async redisHGetAll(key: string): Promise<Record<string, string> | null> {
    const data = await this.getConnectionsMap(key);
    return Object.keys(data).length > 0 ? data : null;
  }

  /**
   * Get the connections map from Redis (JSON-encoded).
   */
  private async getConnectionsMap(key: string): Promise<Record<string, string>> {
    try {
      const raw = await this.redisCacheService.getValue(key);
      if (raw) {
        return JSON.parse(raw);
      }
    } catch {
      // If parsing fails or Redis is unavailable, return empty
    }
    return {};
  }
