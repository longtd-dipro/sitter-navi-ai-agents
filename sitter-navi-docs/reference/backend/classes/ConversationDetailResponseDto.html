import { ApiProperty } from '@nestjs/swagger';

import { ConversationType, ParticipantRole } from '../../enums';

export class ParticipantResponseDto {
  @ApiProperty({ example: '1234567890123456789', description: 'User ID' })
  userId: string;

  @ApiProperty({ example: 'John Doe', description: 'Display name of the participant' })
  displayName: string;

  @ApiProperty({
    enum: ParticipantRole,
    example: ParticipantRole.MEMBER,
    description: 'Role in the group',
  })
  role: ParticipantRole;

  @ApiProperty({
    example: '2025-01-10T08:00:00.000Z',
    description: 'When the participant joined the group',
  })
  joinedAt: Date;
}

export class ConversationDetailResponseDto {
  @ApiProperty({ example: '1234567890123456789', description: 'Group conversation Snowflake ID' })
  id: string;

  @ApiProperty({ example: 'Project Team', description: 'Group name' })
  name: string;

  @ApiProperty({
    enum: ConversationType,
    example: ConversationType.GROUP,
    description: 'Conversation type: direct or group',
  })
  type: ConversationType;

  @ApiProperty({
    example: 'https://s3.amazonaws.com/bucket/avatar.png',
    description: 'Avatar presigned URL',
    nullable: true,
  })
  avatarUrl: string | null;

  @ApiProperty({ type: [ParticipantResponseDto], description: 'List of active participants' })
  participants: ParticipantResponseDto[];

  @ApiProperty({ example: '2025-01-10T08:00:00.000Z', description: 'Group creation timestamp' })
  createdAt: Date;
}
