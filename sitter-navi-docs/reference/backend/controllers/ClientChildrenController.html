import {
  Crud,
  CrudAuth,
  CrudController,
  CrudRequest,
  Override,
  ParsedBody,
  ParsedRequest,
} from '@dataui/crud';
import { Controller, Param, Patch, Query } from '@nestjs/common';
import { ApiOkResponse, ApiOperation, ApiTags } from '@nestjs/swagger';
import { ApiAudience } from '@constants';

import { Auth, CurrentUserId } from '@decorators';
import { Child, ChildStatus } from './children.entity';
import { ChildrenService } from './children.service';
import { RequestGetManyChildrenDto } from './dtos/request';

@Auth()
@Crud({
  model: {
    type: Child,
  },
  params: {
    id: {
      type: 'string',
      primary: true,
      field: 'id',
    },
  },
  query: {
    alwaysPaginate: true,
    softDelete: true,
    maxLimit: 100,
  },
  routes: {
    only: ['getOneBase', 'getManyBase', 'createOneBase', 'updateOneBase', 'deleteOneBase'],
    updateOneBase: {
      allowParamsOverride: true,
      returnShallow: true,
    },
  },
})
@CrudAuth({
  property: 'user',
  filter: (user: { id: string }) => ({ userId: user.id }),
  persist: (user: { id: string }) => ({ userId: user.id }),
})
@ApiTags(ApiAudience.SMILE_FAMILY, 'client/children')
@Controller('client/children')
export class ClientChildrenController implements CrudController<Child> {
  constructor(public service: ChildrenService) {}

  get base(): CrudController<Child> {
    return this;
  }

  @Override()
  getMany(@ParsedRequest() req: CrudRequest, @Query() query: RequestGetManyChildrenDto) {
    return this.service.getMany(req, query);
  }

  // Tạo mới luôn ở trạng thái nháp — không cho client tự set status=complete để
  // bỏ qua validate. Chuyển COMPLETE chỉ qua endpoint submit bên dưới.
  @Override('createOneBase')
  createOne(@ParsedRequest() req: CrudRequest, @ParsedBody() dto: Child) {
    dto.status = ChildStatus.DRAFT;
    return this.service.createOne(req, dto);
  }

  // Client không được tự đổi status. Hồ sơ COMPLETE thì sửa vẫn phải đầy đủ
  // (logic trong service.updateForOwner), hồ sơ DRAFT sửa tự do.
  @Override('updateOneBase')
  updateOne(
    @ParsedRequest() req: CrudRequest,
    @ParsedBody() dto: Child,
    @Param('id') id: string,
    @CurrentUserId() userId: string,
  ) {
    delete (dto as Partial<Child>).status;
    return this.service.updateForOwner(req, dto, id, userId);
  }

  @Patch(':id/submit')
  @ApiOperation({
    summary: 'Hoàn tất hồ sơ trẻ — validate đầy đủ field bắt buộc rồi chuyển sang COMPLETE',
  })
  @ApiOkResponse({ type: Child, description: 'Hồ sơ sau khi chuyển trạng thái COMPLETE' })
  submit(@Param('id') id: string, @CurrentUserId() userId: string) {
    return this.service.submit(id, userId);
  }
}
