import { Body, Controller, Delete, Get, Param, Post, Put, Query } from '@nestjs/common';
import { ApiTags, ApiOperation } from '@nestjs/swagger';
import { ApiAudience } from '@constants';

import { Auth } from '@decorators';
import { RoleType } from '@enums';
import { PermissionService } from './services/permission.service';
import { UserPermissionType } from './entities/user-permission.entity';

// @Auth(RoleType.ADMIN)
@ApiTags(ApiAudience.ADMIN, 'admin/permissions')
@Controller('admin/permissions')
export class AdminPermissionController {
  constructor(private readonly service: PermissionService) {}

  // ─── Permissions ──────────────────────────────────────────────────────────

  @Get()
  @ApiOperation({ summary: 'Get all permissions in the system' })
  getAllPermissions() {
    return this.service.getAllPermissions();
  }

  @Get('discovered')
  @ApiOperation({ summary: 'Get all permissions discovered from code (registry)' })
  getDiscoveredPermissions() {
    return this.service.getDiscoveredPermissions();
  }

  // ─── Permission Groups ────────────────────────────────────────────────────

  @Get('groups')
  @ApiOperation({ summary: 'Get all permission groups' })
  getAllGroups() {
    return this.service.getAllGroups();
  }

  @Post('groups')
  @ApiOperation({ summary: 'Create a permission group' })
  createGroup(
    @Body() body: { name: string; slug: string; description?: string; permissionIds: string[] },
  ) {
    return this.service.createGroup(body);
  }

  @Put('groups/:id/permissions')
  @ApiOperation({ summary: 'Update permissions in a group' })
  updateGroupPermissions(@Param('id') id: string, @Body() body: { permissionIds: string[] }) {
    return this.service.updateGroupPermissions(id, body.permissionIds);
  }

  @Delete('groups/:id')
  @ApiOperation({ summary: 'Delete a permission group' })
  deleteGroup(@Param('id') id: string) {
    return this.service.deleteGroup(id);
  }

  // ─── Role Permissions ─────────────────────────────────────────────────────

  @Get('roles/:role')
  @ApiOperation({ summary: 'Get permissions assigned to a role' })
  getRolePermissions(@Param('role') role: string) {
    return this.service.getRolePermissions(role);
  }

  @Put('roles/:role')
  @ApiOperation({ summary: 'Set permissions for a role (replace all)' })
  setRolePermissions(@Param('role') role: string, @Body() body: { permissionIds: string[] }) {
    return this.service.setRolePermissions(role, body.permissionIds);
  }

  // ─── User Permissions ─────────────────────────────────────────────────────

  @Get('users/:userId')
  @ApiOperation({ summary: 'Get permission details for a user' })
  getUserPermissions(@Param('userId') userId: string) {
    return this.service.getUserPermissionDetails(userId);
  }

  @Get('users/:userId/effective')
  @ApiOperation({ summary: 'Get resolved effective permissions for a user' })
  getEffectivePermissions(@Param('userId') userId: string, @Query('role') role: string) {
    return this.service.getEffectivePermissions(userId, role);
  }

  @Post('users/:userId/grant')
  @ApiOperation({ summary: 'Grant individual permissions to a user' })
  grantPermissions(@Param('userId') userId: string, @Body() body: { permissionIds: string[] }) {
    return this.service.assignPermissionsToUser(
      userId,
      body.permissionIds,
      UserPermissionType.GRANTED,
    );
  }

  @Post('users/:userId/revoke')
  @ApiOperation({ summary: 'Revoke specific permissions from a user' })
  revokePermissions(@Param('userId') userId: string, @Body() body: { permissionIds: string[] }) {
    return this.service.revokePermissionsFromUser(userId, body.permissionIds);
  }

  @Delete('users/:userId/permissions')
  @ApiOperation({ summary: 'Remove permission overrides for a user' })
  removeUserPermissions(
    @Param('userId') userId: string,
    @Body() body: { permissionIds: string[] },
  ) {
    return this.service.removeUserPermissions(userId, body.permissionIds);
  }

  // ─── User Permission Groups ───────────────────────────────────────────────

  @Post('users/:userId/groups')
  @ApiOperation({ summary: 'Assign permission groups to a user' })
  assignGroups(@Param('userId') userId: string, @Body() body: { groupIds: string[] }) {
    return this.service.assignGroupsToUser(userId, body.groupIds);
  }

  @Delete('users/:userId/groups')
  @ApiOperation({ summary: 'Remove permission groups from a user' })
  removeGroups(@Param('userId') userId: string, @Body() body: { groupIds: string[] }) {
    return this.service.removeGroupsFromUser(userId, body.groupIds);
  }
}
