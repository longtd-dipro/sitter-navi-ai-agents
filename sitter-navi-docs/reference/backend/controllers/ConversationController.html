import {
  Body,
  Controller,
  Delete,
  Get,
  HttpCode,
  HttpStatus,
  Param,
  Patch,
  Post,
  Query,
  Res,
  UseGuards,
} from '@nestjs/common';
import {
  ApiBearerAuth,
  ApiOperation,
  ApiParam,
  ApiResponse,
  ApiTags,
  ApiUnauthorizedResponse,
} from '@nestjs/swagger';
import { ApiAudience } from '@constants';
import { Response } from 'express';

import { CurrentUserId } from '@decorators';
import { AuthGuard } from '../../guards/auth.guard';
import { ConversationService } from './conversation.service';
import { ParticipantGuard } from './guards/participant.guard';
import { GroupRoleGuard } from './guards/group-role.guard';
import { GroupRoles } from './decorators/group-roles.decorator';
import { ParticipantRole } from './enums/participant-role.enum';

import { CreateGroupDto } from './dtos/request/create-group.dto';
import { UpdateGroupDto } from './dtos/request/update-group.dto';
import { AddMemberDto } from './dtos/request/add-member.dto';
import { SendMessageDto } from './dtos/request/send-message.dto';
import { GetMessagesDto } from './dtos/request/get-messages.dto';
import { GetGroupsDto } from './dtos/request/get-groups.dto';
import { UploadFileDto } from './dtos/request/upload-file.dto';

import { ConversationResponseDto } from './dtos/response/conversation.response.dto';
import { ConversationDetailResponseDto } from './dtos/response/conversation-detail.response.dto';
import { MessageResponseDto } from './dtos/response/message.response.dto';
import { PaginatedConversationsResponseDto } from './dtos/response/paginated-conversations.response.dto';
import { PaginatedMessagesResponseDto } from './dtos/response/paginated-messages.response.dto';
import { PresignedUploadResponseDto } from './dtos/response/presigned-upload.response.dto';

@ApiTags(ApiAudience.SHARED_CLIENT, 'Conversations')
@ApiBearerAuth()
@ApiUnauthorizedResponse({ description: 'Unauthorized' })
@UseGuards(AuthGuard)
@Controller('conversations')
export class ConversationController {
  constructor(private readonly ConversationService: ConversationService) {}

  @Post()
  @ApiOperation({ summary: 'Create a new group conversation' })
  @ApiResponse({
    status: 201,
    description: 'Group created successfully',
    type: ConversationResponseDto,
  })
  @ApiResponse({ status: 400, description: 'Validation error' })
  async createGroup(
    @CurrentUserId() userId: string,
    @Body() dto: CreateGroupDto,
  ): Promise<ConversationResponseDto> {
    return this.ConversationService.createGroup(userId, dto);
  }

  @Get()
  @ApiOperation({
    summary: 'Get list of group conversations for the current user',
  })
  @ApiResponse({
    status: 200,
    description: 'Group list retrieved successfully',
    type: PaginatedConversationsResponseDto,
  })
  async getUserGroups(
    @CurrentUserId() userId: string,
    @Query() dto: GetGroupsDto,
  ): Promise<PaginatedConversationsResponseDto> {
    return this.ConversationService.getUserGroups(userId, dto);
  }

  @Get('unread-count')
  @ApiOperation({ summary: 'Get total unread message count across all conversations' })
  @ApiResponse({
    status: 200,
    description: 'Total unread count retrieved successfully',
    schema: {
      type: 'object',
      properties: {
        totalUnreadCount: { type: 'number', example: 5 },
      },
    },
  })
  async getTotalUnreadCount(
    @CurrentUserId() userId: string,
  ): Promise<{ totalUnreadCount: number }> {
    return this.ConversationService.getTotalUnreadCount(userId);
  }

  @Get(':id')
  @UseGuards(ParticipantGuard)
  @ApiOperation({ summary: 'Get group conversation details' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiResponse({
    status: 200,
    description: 'Group details retrieved successfully',
    type: ConversationDetailResponseDto,
  })
  @ApiResponse({ status: 403, description: 'Forbidden - not a participant' })
  @ApiResponse({ status: 404, description: 'Group not found' })
  async getGroupDetail(
    @CurrentUserId() userId: string,
    @Param('id') groupId: string,
  ): Promise<ConversationDetailResponseDto> {
    return this.ConversationService.getGroupDetail(userId, groupId);
  }

  @Patch(':id')
  @UseGuards(ParticipantGuard, GroupRoleGuard)
  @GroupRoles(ParticipantRole.OWNER, ParticipantRole.ADMIN)
  @ApiOperation({ summary: 'Update group conversation information' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiResponse({
    status: 200,
    description: 'Group updated successfully',
    type: ConversationResponseDto,
  })
  @ApiResponse({ status: 400, description: 'Validation error' })
  @ApiResponse({ status: 403, description: 'Forbidden - insufficient role' })
  @ApiResponse({ status: 404, description: 'Group not found' })
  async updateGroup(
    @CurrentUserId() userId: string,
    @Param('id') groupId: string,
    @Body() dto: UpdateGroupDto,
  ): Promise<ConversationResponseDto> {
    return this.ConversationService.updateGroup(userId, groupId, dto);
  }

  @Delete(':id')
  @UseGuards(ParticipantGuard, GroupRoleGuard)
  @GroupRoles(ParticipantRole.OWNER)
  @HttpCode(HttpStatus.NO_CONTENT)
  @ApiOperation({ summary: 'Dissolve (delete) a group conversation' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiResponse({ status: 204, description: 'Group dissolved successfully' })
  @ApiResponse({
    status: 403,
    description: 'Forbidden - only owner can dissolve',
  })
  @ApiResponse({ status: 404, description: 'Group not found' })
  async dissolveGroup(
    @CurrentUserId() userId: string,
    @Param('id') groupId: string,
  ): Promise<void> {
    return this.ConversationService.dissolveGroup(userId, groupId);
  }

  @Post(':id/members')
  @UseGuards(ParticipantGuard, GroupRoleGuard)
  @GroupRoles(ParticipantRole.OWNER, ParticipantRole.ADMIN)
  @HttpCode(HttpStatus.NO_CONTENT)
  @ApiOperation({ summary: 'Add a member to the group' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiResponse({ status: 204, description: 'Member added successfully' })
  @ApiResponse({
    status: 400,
    description: 'Validation error (user not found, already member, group full)',
  })
  @ApiResponse({ status: 403, description: 'Forbidden - insufficient role' })
  async addMember(
    @CurrentUserId() userId: string,
    @Param('id') groupId: string,
    @Body() dto: AddMemberDto,
  ): Promise<void> {
    return this.ConversationService.addMember(userId, groupId, dto);
  }

  @Delete(':id/members/:userId')
  @UseGuards(ParticipantGuard, GroupRoleGuard)
  @GroupRoles(ParticipantRole.OWNER, ParticipantRole.ADMIN)
  @HttpCode(HttpStatus.NO_CONTENT)
  @ApiOperation({ summary: 'Remove a member from the group' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiParam({ name: 'userId', description: 'User ID of the member to remove' })
  @ApiResponse({ status: 204, description: 'Member removed successfully' })
  @ApiResponse({ status: 400, description: 'Validation error' })
  @ApiResponse({
    status: 403,
    description: 'Forbidden - insufficient role or hierarchy violation',
  })
  async removeMember(
    @CurrentUserId() userId: string,
    @Param('id') groupId: string,
    @Param('userId') targetUserId: string,
  ): Promise<void> {
    return this.ConversationService.removeMember(userId, groupId, targetUserId);
  }

  @Post(':id/leave')
  @UseGuards(ParticipantGuard)
  @HttpCode(HttpStatus.NO_CONTENT)
  @ApiOperation({ summary: 'Leave a group conversation' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiResponse({ status: 204, description: 'Left group successfully' })
  @ApiResponse({ status: 403, description: 'Forbidden - not a participant' })
  async leaveGroup(@CurrentUserId() userId: string, @Param('id') groupId: string): Promise<void> {
    return this.ConversationService.leaveGroup(userId, groupId);
  }

  @Get(':id/messages')
  @UseGuards(ParticipantGuard)
  @ApiOperation({ summary: 'Get message history for a group conversation' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiResponse({
    status: 200,
    description: 'Messages retrieved successfully',
    type: PaginatedMessagesResponseDto,
  })
  @ApiResponse({ status: 403, description: 'Forbidden - not a participant' })
  async getMessages(
    @CurrentUserId() userId: string,
    @Param('id') groupId: string,
    @Query() dto: GetMessagesDto,
  ): Promise<PaginatedMessagesResponseDto> {
    return this.ConversationService.getMessages(userId, groupId, dto);
  }

  @Post(':id/messages')
  @UseGuards(ParticipantGuard)
  @ApiOperation({ summary: 'Send a message to a group conversation' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiResponse({
    status: 201,
    description: 'Message sent successfully',
    type: MessageResponseDto,
  })
  @ApiResponse({
    status: 200,
    description: 'Duplicate message detected - returning existing message',
    type: MessageResponseDto,
  })
  @ApiResponse({ status: 400, description: 'Validation error' })
  @ApiResponse({ status: 403, description: 'Forbidden - not a participant' })
  async sendMessage(
    @CurrentUserId() userId: string,
    @Res({ passthrough: true }) res: Response,
    @Param('id') groupId: string,
    @Body() dto: SendMessageDto,
  ): Promise<MessageResponseDto> {
    const result = await this.ConversationService.sendMessage(userId, groupId, dto);

    if (result.isNew) {
      res.status(HttpStatus.CREATED);
    } else {
      res.status(HttpStatus.OK);
    }

    // eslint-disable-next-line @typescript-eslint/no-unused-vars
    const { isNew, ...messageResponse } = result;
    return messageResponse as MessageResponseDto;
  }

  @Post(':id/upload-url')
  @UseGuards(ParticipantGuard)
  @ApiOperation({ summary: 'Get a presigned upload URL for file attachment' })
  @ApiParam({ name: 'id', description: 'Group conversation ID' })
  @ApiResponse({
    status: 201,
    description: 'Upload URL generated successfully',
    type: PresignedUploadResponseDto,
  })
  @ApiResponse({ status: 403, description: 'Forbidden - not a participant' })
  async getUploadUrl(
    @CurrentUserId() userId: string,
    @Param('id') groupId: string,
    @Body() dto: UploadFileDto,
  ): Promise<PresignedUploadResponseDto> {
    return this.ConversationService.getUploadUrl(userId, groupId, dto);
  }
}
